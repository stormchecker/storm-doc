52template<typename ValueType, typename StateType>
53StateBehavior<ValueType, StateType>& StateBehavior<ValueType, StateType>::operator=(StateBehavior const& other) {
54 if (this != &other) {
55 // Copy the active choices into the already existing choices (this reuses their memory)
56 if (choices.size() < other.numberOfChoices) {
57 choices.resize(other.numberOfChoices);
58 }
59 std::copy(other.choices.begin(), other.choices.begin() + other.numberOfChoices, choices.begin());
60 numberOfChoices = other.numberOfChoices;
61 stateRewards = other.stateRewards;
62 expanded = other.expanded;
63 }
64 return *this;
65}
