48boost::any TraverseFormulaVisitor::visit(BoundedUntilFormula const& f, boost::any const& data) const {
49 if (callback(f)) {
50 if (f.hasMultiDimensionalSubformulas()) {
51 for (uint64_t i = 0; i < f.getDimension(); ++i) {
52 f.getLeftSubformula(i).accept(*this, data);
53 f.getRightSubformula(i).accept(*this, data);
54 }
55 } else {
56 // All dimensions share the same subformulas, which we thus only visit once.
57 f.getLeftSubformula().accept(*this, data);
58 f.getRightSubformula().accept(*this, data);
59 }
60 }
61 return boost::any();
