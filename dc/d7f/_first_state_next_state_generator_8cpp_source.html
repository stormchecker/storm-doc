46 auto result = pomdp.getTransitionMatrix().getRowGroupSize(belief.representativeState());

55template<typename PomdpType, typename BeliefType>
56typename PomdpType::ValueType FirstStateNextStateGenerator<PomdpType, BeliefType>::getBeliefActionReward(BeliefType const& belief,
57 uint64_t const& localActionIndex) const {
58 using PomdpValueType = typename PomdpType::ValueType;
59 using BeliefValueType = typename BeliefType::ValueType;
60
61 STORM_LOG_ASSERT(hasRewardModel(), "Requested a reward although no reward model was specified.");
62 STORM_LOG_ASSERT(localActionIndex < getBeliefNumberOfActions(belief), "Invalid action index " << localActionIndex << ".");
63 auto result = storm::utility::zero<PomdpValueType>();
64 auto const& actionIndices = pomdp.getTransitionMatrix().getRowGroupIndices();
65 belief.forEach([&localActionIndex, &actionIndices, &result, this](BeliefStateType const& state, BeliefValueType const& val) {
66 uint64_t const actionIndex = actionIndices[state] + localActionIndex;
67 result += storm::utility::convertNumber<PomdpValueType>(val) * this->actionRewards[actionIndex];
68 });
69 return result;
70}
