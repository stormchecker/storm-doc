65storm::Environment EnvironmentBuilder::buildEnvironment() {
66 storm::Environment env;
67
68 applyIfRegistered<storm::settings::modules::GeneralSettings>(
69 [&](auto const& generalSettings) { applyIfSet(generalSettings.isToleranceSet(), [&]() { env.setModelTolerance(generalSettings.getTolerance()); }); });
70
71 setSolverEnvironment(env.solver());
72 setModelcheckerEnvironment(env.modelchecker());
73 setDdEnvironment(env.dd());
74 setExplorationEnvironment(env.exploration());
75
76 return env;
77}

254 }
255 if (multiobjectiveSettings.isAutoEncodingSet()) {

257 } else if (multiobjectiveSettings.isClassicEncodingSet()) {
258 multiobjective.setEncodingType(MultiObjectiveModelCheckerEnvironment::EncodingType::Classic);

265 multiobjective.setUseIndicatorConstraints(multiobjectiveSettings.isIndicatorConstraintsSet());
266 multiobjective.setUseRedundantBsccConstraints(multiobjectiveSettings.isRedundantBsccConstraintsSet());
267 applyIfSet(multiobjectiveSettings.isWeightedSumApproximationTradeoffSet(), [&]() {
268 multiobjective.setApproximationTradeoff(
269 storm::utility::convertNumber<storm::RationalNumber>(multiobjectiveSettings.getWeightedSumApproximationTradeoff()));
270 });
271 applyIfSet(multiobjectiveSettings.isMaxStepsSet(), [&]() { multiobjective.setMaxSteps(multiobjectiveSettings.getMaxSteps()); });
272 applyIfSet(multiobjectiveSettings.hasSchedulerRestriction(),
273 [&]() { multiobjective.setSchedulerRestriction(multiobjectiveSettings.getSchedulerRestriction()); });
274 multiobjective.setPrintResults(multiobjectiveSettings.isPrintResultsSet());
