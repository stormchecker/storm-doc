23 std::vector<ConstantType> const maxValues) const {
24 std::vector<std::pair<Assumption, AssumptionStatus>> result;
25 STORM_LOG_INFO("Creating assumptions for " << val1 << " and " << val2);
26 STORM_LOG_ASSERT(order->compare(val1, val2) == Order::UNKNOWN, "Expected the given pair to indeed be unordered.");
27 auto assumption = createAndCheckAssumption(val1, val2, AssumptionRelation::Greater, order, region, minValues, maxValues);
28 if (assumption.second != AssumptionStatus::INVALID) {
29 result.push_back(assumption);
30 if (assumption.second == AssumptionStatus::VALID) {
31 STORM_LOG_ASSERT(
32 createAndCheckAssumption(val2, val1, AssumptionRelation::Greater, order, region, minValues, maxValues).second != AssumptionStatus::VALID &&
33 createAndCheckAssumption(val1, val2, AssumptionRelation::Equal, order, region, minValues, maxValues).second != AssumptionStatus::VALID,
34 "At most one of the three candidate assumptions may be valid.");
35 STORM_LOG_INFO("Assumption " << assumption.first << "is valid\n");
36 return result;
37 }
38 }
39 STORM_LOG_ASSERT(order->compare(val1, val2) == Order::UNKNOWN, "Expected the given pair to indeed be unordered.");
40 assumption = createAndCheckAssumption(val2, val1, AssumptionRelation::Greater, order, region, minValues, maxValues);
41 if (assumption.second != AssumptionStatus::INVALID) {
42 if (assumption.second == AssumptionStatus::VALID) {
43 result.clear();
44 result.push_back(assumption);
45 STORM_LOG_ASSERT(
46 createAndCheckAssumption(val1, val2, AssumptionRelation::Equal, order, region, minValues, maxValues).second != AssumptionStatus::VALID,
47 "At most one of the three candidate assumptions may be valid.");
48 STORM_LOG_INFO("Assumption " << assumption.first << "is valid\n");
49 return result;
50 }
51 result.push_back(assumption);
52 }
53 STORM_LOG_ASSERT(order->compare(val1, val2) == Order::UNKNOWN, "Expected the given pair to indeed be unordered.");
54 assumption = createAndCheckAssumption(val1, val2, AssumptionRelation::Equal, order, region, minValues, maxValues);
55 if (assumption.second != AssumptionStatus::INVALID) {
56 if (assumption.second == AssumptionStatus::VALID) {
57 result.clear();
58 result.push_back(assumption);
