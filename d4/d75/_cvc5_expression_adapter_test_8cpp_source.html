17void assertEquivalent(storm::expressions::Expression const& expression, cvc5::Term const& term, cvc5::TermManager& termManager, cvc5::Solver& solver,
18 storm::adapters::Cvc5ExpressionAdapter& adapter) {
19 // Note that the variables occurring in the term have to be the ones that the adapter uses. Two constants with the
20 // same name are distinct variables for cvc5.
21 cvc5::Term equivalence = termManager.mkTerm(cvc5::Kind::EQUAL, {adapter.translateExpression(expression), term});
22 solver.assertFormula(termManager.mkTerm(cvc5::Kind::NOT, {equivalence}));
23 ASSERT_TRUE(solver.checkSat().isUnsat()) << "Not equivalent: " << expression << " (" << adapter.translateExpression(expression) << ") vs " << term;
24 solver.resetAssertions();
25}

114 // Dividing an integer by a rational requires the integer to be converted into a rational.

116

119 cvc5::Term cvc5IntVar2 = adapter.translateExpression(intVar2);
120 assertEquivalent(intVar / intVar2, termManager.mkTerm(cvc5::Kind::DIVISION, {cvc5ToReal, termManager.mkTerm(cvc5::Kind::TO_REAL, {cvc5IntVar2})}),
121 termManager, solver, adapter);
122
123 // Storm types the quotient of two integers as an integer, whereas cvc5 divides over the rational numbers.
124 // Surrounding integer terms must therefore still be translated into a term cvc5 accepts.
125 storm::expressions::Variable intVar3 = manager->declareIntegerVariable("intVar3");
126 cvc5::Term cvc5IntVar3 = adapter.translateExpression(intVar3);
127 assertEquivalent(intVar / intVar2 + intVar3 == manager->integer(0),

150TEST(Cvc5ExpressionAdapter, StormToCvc5FloorAndCeil) {
151 cvc5::TermManager termManager;
152 cvc5::Solver solver(termManager);
153
154 std::shared_ptr<storm::expressions::ExpressionManager> manager(new storm::expressions::ExpressionManager());
155 storm::adapters::Cvc5ExpressionAdapter adapter(*manager, solver);
156
157 storm::expressions::Variable x = manager->declareRationalVariable("x");
158 cvc5::Term cvc5X = adapter.translateExpression(x);
159
160 cvc5::Term cvc5Floor = adapter.translateExpression(storm::expressions::floor(x));
161 cvc5::Term cvc5Ceil = adapter.translateExpression(storm::expressions::ceil(x));
162
163 // The floor and the ceiling of a real number are integers, which we check by looking at the model.
164 solver.assertFormula(termManager.mkTerm(cvc5::Kind::EQUAL, {cvc5X, termManager.mkReal(7, 3)}));
165 solver.assertFormula(termManager.mkTerm(cvc5::Kind::EQUAL, {cvc5Floor, termManager.mkInteger(2)}));
166 solver.assertFormula(termManager.mkTerm(cvc5::Kind::EQUAL, {cvc5Ceil, termManager.mkInteger(3)}));
167 ASSERT_TRUE(solver.checkSat().isSat());
168
169 // Storm's floor is cvc5's to_int, whereas the ceiling is the negation of the floor of the negation.
170 assertEquivalent(storm::expressions::floor(x), termManager.mkTerm(cvc5::Kind::TO_INTEGER, {cvc5X}), termManager, solver, adapter);
171 assertEquivalent(storm::expressions::ceil(x),
172 termManager.mkTerm(cvc5::Kind::NEG, {termManager.mkTerm(cvc5::Kind::TO_INTEGER, {termManager.mkTerm(cvc5::Kind::NEG, {cvc5X})})}),
173 termManager, solver, adapter);
174
175 // The floor and the ceiling of an integer are the integer itself.
176 storm::expressions::Variable intVar = manager->declareIntegerVariable("intVar");
177 cvc5::Term cvc5IntVar = adapter.translateExpression(intVar);
178 assertEquivalent(storm::expressions::floor(intVar), cvc5IntVar, termManager, solver, adapter);
179 assertEquivalent(storm::expressions::ceil(intVar), cvc5IntVar, termManager, solver, adapter);
180}

207TEST(Cvc5ExpressionAdapter, Cvc5ToStorm) {
208 cvc5::TermManager termManager;
209 cvc5::Solver solver(termManager);
210
211 std::shared_ptr<storm::expressions::ExpressionManager> manager(new storm::expressions::ExpressionManager());
212 storm::adapters::Cvc5ExpressionAdapter adapter(*manager, solver);
213
214 // Boolean constants can be translated back to storm.
215 storm::expressions::Expression translatedTrue = adapter.translateExpression(termManager.mkBoolean(true));
216 ASSERT_TRUE(translatedTrue.isTrue());
217 ASSERT_TRUE(adapter.translateExpression(termManager.mkBoolean(false)).isFalse());
218
219 // cvc5 prints negative integers in the SMT-LIB notation, i.e., as "(- 42)".
220 auto const assertIntegerLiteral = [&adapter, &termManager](int64_t value) {
221 storm::expressions::Expression translated = adapter.translateExpression(termManager.mkInteger(value));
222 auto const* integerLiteral = dynamic_cast<storm::expressions::IntegerLiteralExpression const*>(&translated.getBaseExpression());
223 EXPECT_TRUE(integerLiteral != nullptr);
224 if (integerLiteral != nullptr) {
225 EXPECT_EQ(value, integerLiteral->getValue());
226 }
227 };
228 assertIntegerLiteral(-42);
229 assertIntegerLiteral(0);
230 assertIntegerLiteral(42);
231
232 // cvc5 prints non-integral rationals in the SMT-LIB notation, e.g., 1/2 is printed as "(/ 1 2)" and -3/4 as
233 // "(/ (- 3) 4)", whereas integral rationals are printed in decimal notation.
234 auto const assertRationalLiteral = [&adapter, &termManager](int numerator, int denominator) {
235 storm::expressions::Expression translated = adapter.translateExpression(termManager.mkReal(numerator, denominator));
236 auto const* rationalLiteral = dynamic_cast<storm::expressions::RationalLiteralExpression const*>(&translated.getBaseExpression());
237 EXPECT_TRUE(rationalLiteral != nullptr);
238 if (rationalLiteral != nullptr) {
239 EXPECT_EQ(storm::RationalNumber(numerator, denominator), rationalLiteral->getValue());
240 }
241 };
242 assertRationalLiteral(1, 2);
243 assertRationalLiteral(-3, 4);
244 assertRationalLiteral(7, 3);
245 assertRationalLiteral(2, 1);
246
247 // Free constants and the usual operators can be translated back to storm.
248 storm::expressions::Variable xVar = manager->declareIntegerVariable("x");
249 storm::expressions::Variable yVar = manager->declareIntegerVariable("y");
250 cvc5::Term cvc5X = adapter.translateExpression(xVar);
251 cvc5::Term cvc5Y = adapter.translateExpression(yVar);
252 cvc5::Term cvc5Sum = termManager.mkTerm(cvc5::Kind::ADD, {cvc5X, cvc5Y});
253 cvc5::Term cvc5Less = termManager.mkTerm(cvc5::Kind::LT, {cvc5Sum, termManager.mkInteger(0)});
254
255 storm::expressions::Expression translatedSum = adapter.translateExpression(cvc5Sum);
256 storm::expressions::Expression translatedLess = adapter.translateExpression(cvc5Less);
257
258 ASSERT_EQ(storm::expressions::OperatorType::Plus, translatedSum.getOperator());
259 ASSERT_TRUE(translatedSum.getOperand(0).isVariable());
260 ASSERT_EQ("x", translatedSum.getOperand(0).getIdentifier());
261 ASSERT_TRUE(translatedSum.getOperand(1).isVariable());
262 ASSERT_EQ("y", translatedSum.getOperand(1).getIdentifier());
263
264 ASSERT_EQ(storm::expressions::OperatorType::Less, translatedLess.getOperator());
265 ASSERT_EQ(storm::expressions::OperatorType::Plus, translatedLess.getOperand(0).getOperator());
266 auto const* zeroLiteral = dynamic_cast<storm::expressions::IntegerLiteralExpression const*>(&translatedLess.getOperand(1).getBaseExpression());
267 ASSERT_TRUE(zeroLiteral != nullptr);
268 ASSERT_EQ(0, zeroLiteral->getValue());
269
270 // cvc5 admits sums and products with more than two operands, so all of them have to be taken into account.
271 cvc5::Term cvc5ThreeWaySum = termManager.mkTerm(cvc5::Kind::ADD, {cvc5X, cvc5Y, adapter.translateExpression(manager->integer(3))});
272 storm::expressions::Expression translatedThreeWaySum = adapter.translateExpression(cvc5ThreeWaySum);
273 ASSERT_EQ(storm::expressions::OperatorType::Plus, translatedThreeWaySum.getOperator());
274 EXPECT_EQ((translatedSum + manager->integer(3)).toString(), translatedThreeWaySum.toString());
275
276 cvc5::Term cvc5ThreeWayProduct = termManager.mkTerm(cvc5::Kind::MULT, {cvc5X, cvc5Y, adapter.translateExpression(manager->integer(3))});
277 storm::expressions::Expression translatedThreeWayProduct = adapter.translateExpression(cvc5ThreeWayProduct);
278 ASSERT_EQ(storm::expressions::OperatorType::Times, translatedThreeWayProduct.getOperator());
279 EXPECT_EQ((translatedSum.getOperand(0) * translatedSum.getOperand(1) * manager->integer(3)).toString(), translatedThreeWayProduct.toString());
280
281 // cvc5 also uses EQUAL for boolean equivalence. Translating such a term back has to yield an iff, because storm's
282 // operator== only accepts numerical operands and throws otherwise. This matters for getUnsatAssumptions(), which
283 // translates every assumption of the returned core back.
284 storm::expressions::Variable aVar = manager->declareBooleanVariable("a");
285 storm::expressions::Variable bVar = manager->declareBooleanVariable("b");
286 cvc5::Term cvc5Equivalent = termManager.mkTerm(cvc5::Kind::EQUAL, {adapter.translateExpression(aVar), adapter.translateExpression(bVar)});
287 storm::expressions::Expression translatedEquivalent = adapter.translateExpression(cvc5Equivalent);
288 ASSERT_TRUE(translatedEquivalent.hasBooleanType());
289 EXPECT_EQ(storm::expressions::iff(aVar.getExpression(), bVar.getExpression()).toString(), translatedEquivalent.toString());
290}
