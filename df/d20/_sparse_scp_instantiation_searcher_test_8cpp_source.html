22// Builds a FeasibilitySynthesisTask the same way storm-pars-cli's createFeasibilitySynthesisTaskFromSettings does:
23// the bound is stripped from the formula and carried separately.
24std::shared_ptr<storm::pars::FeasibilitySynthesisTask const> makeTask(
25 std::shared_ptr<storm::logic::Formula const> const& formulaWithBound,
26 std::optional<storm::storage::ParameterRegion<storm::RationalFunction>> const& region = std::nullopt) {
27 std::shared_ptr<storm::logic::Formula> clone = formulaWithBound->clone();
28 clone->asOperatorFormula().removeBound();
29 storm::pars::FeasibilitySynthesisTask t(clone->asSharedPointer());
30 t.setBound(formulaWithBound->asOperatorFormula().getBound());
31 if (region) {
32 t.setRegion(*region);
33 }
34 return std::make_shared<storm::pars::FeasibilitySynthesisTask const>(std::move(t));
35}

87// Exercises ConstantType = storm::RationalNumber end-to-end (not just that it compiles, which the
