20UnpackedValue AffinePMdpDecomposition::decomposeAffine(storm::RationalFunction const& value) {
21 std::set<storm::RationalFunctionVariable> occurring;
22 storm::utility::parametric::gatherOccurringVariables(value, occurring);
23
24 UnpackedValue unpacked;
25 storm::utility::parametric::Valuation<storm::RationalFunction> zeroValuation;
26 for (auto const& var : occurring) {
27 zeroValuation[var] = storm::utility::zero<storm::RationalFunctionCoefficient>();
28 }
29 unpacked.constant = storm::utility::parametric::evaluate<double>(value, zeroValuation);
30
31 unpacked.parameterCoefficients.reserve(occurring.size());
32 for (auto const& var : occurring) {
33 storm::RationalFunction derivative = value.derivative(var);
34 STORM_LOG_ASSERT(derivative.isConstant(), "Derivative of an affine function w.r.t. one of its own parameters must be constant.");
35 unpacked.parameterCoefficients.emplace_back(var, storm::utility::convertNumber<double>(derivative.constantPart()));
36 }
37
38 parameters.insert(occurring.begin(), occurring.end());
39 return unpacked;
40}

42AffinePMdpDecomposition::AffinePMdpDecomposition(storm::models::sparse::Model<storm::RationalFunction> const& model,
43 storm::models::sparse::StandardRewardModel<storm::RationalFunction> const* rewardModel) {
44 auto const& matrix = model.getTransitionMatrix();
45 auto const& rowGroupIndices = matrix.getRowGroupIndices();
46 auto stateOfRow = [&](uint64_t row) { return std::upper_bound(rowGroupIndices.begin(), rowGroupIndices.end(), row) - rowGroupIndices.begin() - 1; };
47
48 rows.reserve(matrix.getRowCount());
49 for (uint64_t row = 0; row < matrix.getRowCount(); ++row) {
50 std::vector<UnpackedTransition> unpackedRow;
51 unpackedRow.reserve(matrix.getRow(row).getNumberOfEntries());
52
53 for (auto const& entry : matrix.getRow(row)) {
54 STORM_LOG_THROW(storm::utility::parametric::isLinear(entry.getValue()), storm::exceptions::NotSupportedException,
55 NotAffine << "the transition from state " << stateOfRow(row) << " (row " << row << ") to state " << entry.getColumn() << " is '"
56 << entry.getValue() << "', which is not affine.");
57 UnpackedValue decomposed = decomposeAffine(entry.getValue());
58 UnpackedTransition unpacked;
59 unpacked.successor = entry.getColumn();
60 unpacked.constant = decomposed.constant;
61 unpacked.parameterCoefficients = std::move(decomposed.parameterCoefficients);
62 unpackedRow.push_back(std::move(unpacked));
63 }
64 rows.push_back(std::move(unpackedRow));
65 }
66
67 if (rewardModel != nullptr) {
68 std::vector<storm::RationalFunction> rewardVector = rewardModel->getTotalRewardVector(matrix);
69 rewardRows.reserve(rewardVector.size());
70 for (uint64_t row = 0; row < rewardVector.size(); ++row) {
71 STORM_LOG_THROW(
72 storm::utility::parametric::isLinear(rewardVector[row]), storm::exceptions::NotSupportedException,
73 NotAffine << "the reward of state " << stateOfRow(row) << " (row " << row << ") is '" << rewardVector[row] << "', which is not affine.");
74 rewardRows.push_back(decomposeAffine(rewardVector[row]));
75 }
76 }
77}
