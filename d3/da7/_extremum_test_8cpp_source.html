19TEST(ExtremumTest, emptyExtremumIsTheInfinityItStandsFor) {
20 // The extremum over an empty set is +infinity if we minimize and -infinity if we maximize. That is a value like
21 // any other, so reading it needs no special case.
22 storm::utility::Minimum<storm::RationalNumber> minimum;
23 EXPECT_TRUE(minimum.empty());
24 EXPECT_TRUE(storm::utility::isInfinity(minimum.getExtendedValue()));

26 storm::utility::Maximum<storm::RationalNumber> maximum;

28 EXPECT_TRUE(storm::utility::isNegativeInfinity(maximum.getExtendedValue()));

34}

36TEST(ExtremumTest, holdsAnInfiniteValue) {
37 // A minimum that has been given +infinity and one that has been given nothing are the same thing: the minimum over
38 // {+infinity} is +infinity, and so is the minimum over the empty set. What matters is that the value is right.
39 storm::utility::Minimum<storm::RationalNumber> minimum;
40 EXPECT_FALSE(minimum &= ExtendedRationalNumber::infinity());
41 EXPECT_TRUE(storm::utility::isInfinity(minimum.getExtendedValue()));
42
43 // The infinity that is not the base value is stored like any other value.
44 storm::utility::Minimum<storm::RationalNumber> other;
45 EXPECT_TRUE(other &= ExtendedRationalNumber::negativeInfinity());
46 EXPECT_FALSE(other.empty());
47 EXPECT_TRUE(storm::utility::isNegativeInfinity(other.getExtendedValue()));
48
49 // Maximizing, +infinity is an improvement over anything and is kept.
50 storm::utility::Maximum<storm::RationalNumber> maximum;
51 EXPECT_TRUE(maximum &= rational(2));
52 EXPECT_TRUE(maximum &= ExtendedRationalNumber::infinity());
53 EXPECT_FALSE(maximum.empty());
54 EXPECT_TRUE(storm::utility::isInfinity(maximum.getExtendedValue()));
55 // Nothing finite improves upon it any more.
56 EXPECT_FALSE(maximum &= rational(1000000));
57}

59TEST(ExtremumTest, ordinaryUseIsUnchanged) {
60 storm::utility::Minimum<storm::RationalNumber> minimum;
61 EXPECT_TRUE(minimum &= rational(5));
