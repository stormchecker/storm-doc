29void validateRewardModelSelection(PropertyInformation const& propertyInformation, std::vector<std::string> const& rewardModelNames) {
30 STORM_LOG_THROW(!rewardModelNames.empty(), storm::exceptions::IllegalArgumentException,
31 "Reward-aware belief exploration requires at least one selected reward model.");
32 for (auto const& rewardBound : propertyInformation.rewardBounds) {
33 // TODO: we can lift this if the handling of empty reward models is made consistent in storm-main
34 STORM_LOG_THROW(!rewardBound.rewardModelName.empty(), storm::exceptions::NotSupportedException,
35 "For POMDPs, reward-bounded formulae must explicitly name a reward model for each bound.");
36 STORM_LOG_THROW(
37 !(propertyInformation.dir == storm::OptimizationDirection::Minimize && rewardBound.lowerBound.has_value() && rewardBound.upperBound.has_value()),
38 storm::exceptions::NotSupportedException,
39 "Reward-bounded probability minimisation does not support dimensions with both lower and upper reward bounds.");
40 }
41 STORM_LOG_THROW(rewardModelNames.size() >= propertyInformation.rewardBounds.size(), storm::exceptions::IllegalArgumentException,
42 "The selected reward models must include all reward-bounded dimensions first.");
43 for (std::size_t i = 0; i < propertyInformation.rewardBounds.size(); ++i) {
44 STORM_LOG_THROW(
45 rewardModelNames[i] == propertyInformation.rewardBounds[i].rewardModelName, storm::exceptions::IllegalArgumentException,
46 "Selected reward model " << i << " must match the corresponding reward bound '" << propertyInformation.rewardBounds[i].rewardModelName << "'.");
47 }
48}

57template<typename PomdpModelType, typename BeliefType, typename BeliefMdpValueType, typename InfoType>
58typename BeliefExploration<BeliefMdpValueType, PomdpModelType, BeliefType>::TerminationCallback getTerminationCallback(
59 storm::pomdp::beliefs::BeliefBasedModelCheckerOptions<BeliefMdpValueType> const& options, InfoType& info, storm::utility::Stopwatch& swExplore) {
60 switch (options.getTerminationCriterion()) {
61 case MAX_EXPLORATION_SIZE:
62 return [&info, maxSize = options.maxExplorationSize.value()]() { return info.discoveredBeliefs.getNumberOfBeliefIds() > maxSize; };
63 case MAX_EXPLORATION_TIME:
64 return [&swExplore, maxDuration = options.maxExplorationTime.value()]() { return swExplore.getTimeInSeconds() > static_cast<double>(maxDuration); };
65 case MAX_EXPLORATION_SIZE_AND_TIME:
66 return [&info, &swExplore, maxSize = options.maxExplorationSize.value(), maxDuration = options.maxExplorationTime.value()]() {
67 return info.discoveredBeliefs.getNumberOfBeliefIds() > maxSize || swExplore.getTimeInSeconds() > static_cast<double>(maxDuration);
68 };
69 case NONE:
70 // Unlimited unfolding (useful for known finite belief MDPs)
71 return []() { return false; };
72 default:
73 STORM_LOG_ERROR("Unknown termination criterion for belief exploration.");
74 return []() { return false; };
75 }
76}

83template<typename PomdpModelType, typename BeliefType, typename BeliefMdpValueType>
84typename BeliefExploration<BeliefMdpValueType, PomdpModelType, BeliefType>::TerminalBeliefCallback getTerminalBeliefCallback(
85 PropertyInformation const& propertyInformation, storm::pomdp::beliefs::BeliefBasedModelCheckerOptions<BeliefMdpValueType> const& options,
86 storm::pomdp::storage::PreprocessingPomdpValueBounds<typename PomdpModelType::ValueType> const& valueBounds) {
87 using PomdpValueType = typename PomdpModelType::ValueType;
88 if (propertyInformation.kind == PropertyInformation::Kind::ExpectedTotalReachabilityReward) {
89 if (options.cutZeroGap) {
90 // Terminate only if the preprocessing bounds have exactly zero gap.
91 return [&propertyInformation, &valueBounds](BeliefType const& belief) -> std::optional<BeliefMdpValueType> {
92 if (propertyInformation.targetObservations.contains(belief.observation())) {
93 return storm::utility::zero<BeliefMdpValueType>();
94 } else {
95 auto smallestUpper = storm::utility::infinity<PomdpValueType>();
96 for (auto const& valueList : valueBounds.upper) {
97 smallestUpper = std::min(smallestUpper, belief.template getWeightedSum<PomdpValueType>(valueList));
98 }
99 PomdpValueType largestLower = -storm::utility::infinity<PomdpValueType>();
100 for (auto const& valueList : valueBounds.lower) {
101 largestLower = storm::utility::max(largestLower, belief.template getWeightedSum<PomdpValueType>(valueList));
102 }
103 if (storm::utility::abs<PomdpValueType>(smallestUpper - largestLower) <= storm::utility::zero<PomdpValueType>()) {
104 if constexpr (std::is_same_v<PomdpValueType, BeliefMdpValueType>) {
105 return propertyInformation.dir == solver::OptimizationDirection::Maximize ? largestLower : smallestUpper;
106 } else {
107 return storm::utility::convertNumber<BeliefMdpValueType>(
108 propertyInformation.dir == solver::OptimizationDirection::Maximize ? largestLower : smallestUpper);
109 }
110 }
111 return std::nullopt;
112 }
113 };
114 } else {
115 return [&propertyInformation](BeliefType const& belief) -> std::optional<BeliefMdpValueType> {
116 if (propertyInformation.targetObservations.contains(belief.observation())) {
117 return storm::utility::zero<BeliefMdpValueType>();
118 } else {
119 return std::nullopt;
120 }
121 };
122 }
123 } else if (propertyInformation.kind == PropertyInformation::Kind::RewardBoundedReachabilityProbability) {
124 return [](BeliefType const& belief) -> std::optional<BeliefMdpValueType> {
125 // Reward-aware exploration deliberately has no terminal beliefs. Reaching a target observation does not decide
126 // a reward-bounded property: lower bounds may require further reward accumulation, and the reward vector is
127 // encoded only on the transition. The belief MDP builder consequently receives ordinary explored/frontier edges.
128 return std::nullopt;
129 };
130 } else if (options.cutZeroGap) {
131 // Terminate only if the preprocessing bounds have exactly zero gap.
132 return [&propertyInformation, &valueBounds](BeliefType const& belief) -> std::optional<BeliefMdpValueType> {
133 if (propertyInformation.targetObservations.contains(belief.observation())) {
134 return storm::utility::one<BeliefMdpValueType>();
135 } else {
136 auto smallestUpper = storm::utility::infinity<PomdpValueType>();
137 for (auto const& valueList : valueBounds.upper) {
138 smallestUpper = std::min(smallestUpper, belief.template getWeightedSum<PomdpValueType>(valueList));
139 }
140 PomdpValueType largestLower = -storm::utility::infinity<PomdpValueType>();
141 for (auto const& valueList : valueBounds.lower) {
142 largestLower = storm::utility::max(largestLower, belief.template getWeightedSum<PomdpValueType>(valueList));
143 }
144 if (storm::utility::abs<PomdpValueType>(smallestUpper - largestLower) <= storm::utility::zero<PomdpValueType>()) {
145 if constexpr (std::is_same_v<PomdpValueType, BeliefMdpValueType>) {
146 return propertyInformation.dir == solver::OptimizationDirection::Maximize ? largestLower : smallestUpper;
147 } else {
148 return storm::utility::convertNumber<BeliefMdpValueType>(
149 propertyInformation.dir == solver::OptimizationDirection::Maximize ? largestLower : smallestUpper);
150 }
151 }
152 return std::nullopt;
153 }
154 };
155 } else {
156 return [&propertyInformation](BeliefType const& belief) -> std::optional<BeliefMdpValueType> {
157 if (propertyInformation.targetObservations.contains(belief.observation())) {
158 return storm::utility::one<BeliefMdpValueType>();
159 } else {
160 return std::nullopt;
161 }
162 };
163 }
164}

171template<typename PomdpModelType, typename BeliefType, typename BeliefMdpValueType, typename InfoType>
172std::pair<std::shared_ptr<models::sparse::Mdp<BeliefMdpValueType>>, std::unordered_map<uint64_t, BeliefId>> buildBeliefMdpFromInfo(
173 PropertyInformation const& propertyInformation, storm::pomdp::storage::PreprocessingPomdpValueBounds<typename PomdpModelType::ValueType> const& valueBounds,
174 InfoType const& info) {
175 using PomdpValueType = typename PomdpModelType::ValueType;
176 std::function<std::unordered_map<std::string, BeliefMdpValueType>(BeliefType const&)> computeCutOffValueMap =
177 [&valueBounds, &propertyInformation](BeliefType const& belief) {
178 // Add addtional cut-off sources here
179 uint64_t const nrCutoffPolicies =
180 propertyInformation.dir == storm::OptimizationDirection::Minimize ? valueBounds.upper.size() : valueBounds.lower.size();
181 std::unordered_map<std::string, BeliefMdpValueType> result;
182 for (uint64_t i = 0; i < nrCutoffPolicies; ++i) {
183 auto val = belief.template getWeightedSum<PomdpValueType>(
184 propertyInformation.dir == storm::OptimizationDirection::Minimize ? valueBounds.upper.at(i) : valueBounds.lower.at(i));
185 if constexpr (std::is_same_v<PomdpValueType, BeliefMdpValueType>) {
186 result["__sched_" + std::to_string(i)] = val;
187 } else {
188 result["__sched_" + std::to_string(i)] = storm::utility::convertNumber<BeliefMdpValueType>(val);
189 }
190 }
191 return result;
192 };
193 return buildBeliefMdp(info, propertyInformation, computeCutOffValueMap);
194}

196template<typename PomdpModelType, typename BeliefType, typename BeliefMdpValueType, typename PolicyValueType, typename AbstractionType,
197 typename InfoType = StandardExplorationInformation<BeliefMdpValueType, BeliefType>>
198BeliefBasedModelCheckerResult<BeliefMdpValueType, PolicyValueType> checkUnfoldOrDiscretize(
199 storm::Environment const& env, PomdpModelType const& pomdp, PropertyInformation const& propertyInformation,
200 storm::pomdp::beliefs::BeliefBasedModelCheckerOptions<BeliefMdpValueType> const& options,
201 storage::BeliefExplorationBounds<typename PomdpModelType::ValueType> const& valueBounds, storm::OptionalRef<AbstractionType> abstraction = {}) {
202 BeliefBasedModelCheckerStatistics statistics;
203 std::optional<storm::pomdp::policy::ObservationBasedFiniteStateController<PolicyValueType>> generatedPolicy;
204 STORM_LOG_ASSERT(propertyInformation.kind == PropertyInformation::Kind::ReachabilityProbability ||
205 propertyInformation.kind == PropertyInformation::Kind::ExpectedTotalReachabilityReward,
206 "Unexpected kind of property.");
207
208 if (options.generatePolicy) {
209 STORM_LOG_THROW(valueBounds.preprocessingBounds.has_value(), storm::exceptions::IllegalArgumentException,
210 "Policy generation requires preprocessing bounds with continuation schedulers.");
211 auto const& bounds = *valueBounds.preprocessingBounds;
212 auto const& cutOffBounds = storm::solver::minimize(propertyInformation.dir) ? bounds.upper : bounds.lower;
213 auto const& cutOffSchedulers = storm::solver::minimize(propertyInformation.dir) ? bounds.upperSchedulers : bounds.lowerSchedulers;
214 STORM_LOG_THROW(cutOffSchedulers.size() == cutOffBounds.size(), storm::exceptions::IllegalArgumentException,
215 "Policy generation requires one continuation scheduler per " << (storm::solver::minimize(propertyInformation.dir) ? "upper" : "lower")
216 << " cut-off bound, but received " << cutOffSchedulers.size()
217 << " schedulers for " << cutOffBounds.size() << " bounds.");
218 }
219
220 STORM_LOG_INFO("Exploring the belief space.");
221
222 // First, explore the beliefs and its successors
223 using BeliefExplorationType = BeliefExploration<BeliefMdpValueType, PomdpModelType, BeliefType>;
224 storm::utility::Stopwatch swExplore(true);
225 BeliefExplorationType exploration(pomdp);
226
227 auto info = exploration.template initializeExploration<InfoType>(pomdp.getNrObservations(), options.explorationQueueOrder);
228 info.generateChoiceLabeling = options.buildChoiceLabeling || options.generatePolicy;
229
230 // Determine terminationCallback based on options
231 typename BeliefExplorationType::TerminationCallback terminationCallback =
232 getTerminationCallback<PomdpModelType, BeliefType, BeliefMdpValueType>(options, info, swExplore);
233
234 // Determine terminalBeliefCallback based on options
235 typename BeliefExplorationType::TerminalBeliefCallback terminalBeliefCallback =
236 getTerminalBeliefCallback<PomdpModelType, BeliefType, BeliefMdpValueType>(propertyInformation, options, *valueBounds.preprocessingBounds);
237
238 if constexpr (std::is_same_v<InfoType, ClippingExplorationInformation<BeliefMdpValueType, BeliefType>>) {
239 STORM_LOG_ASSERT(options.useClipping, "Clipping exploration information requires clipping to be enabled.");
240 if (propertyInformation.kind == PropertyInformation::Kind::ExpectedTotalReachabilityReward) {
241 exploration.resumeClippingExploration(info, terminalBeliefCallback, terminationCallback, propertyInformation.rewardModelName.value(), abstraction);
242 } else {
243 exploration.resumeClippingExploration(info, terminalBeliefCallback, terminationCallback, storm::NullRef, abstraction);
244 }
245 STORM_LOG_TRACE("Starting clipping phase.");
246 STORM_LOG_ASSERT(options.clippingResolutions.has_value(), "Clipping requested, but no resolution vector given.");
247 std::vector<uint64_t> resolutions(options.clippingResolutions.value());
248 if (propertyInformation.kind == PropertyInformation::Kind::ExpectedTotalReachabilityReward) {
249 STORM_LOG_ASSERT(valueBounds.extremeBounds.has_value(),
250 "Clipping for expected total reachability reward requires extreme value bounds to be given.");
251 ClippingBeliefAbstraction<BeliefType> clippingAbstraction(
252 env, std::move(resolutions), std::move(valueBounds.extremeBounds->template copyValues<typename BeliefType::ValueType>()));
253 exploration.resumeClippingExploration(
254 info, terminalBeliefCallback, []() { return false; }, propertyInformation.rewardModelName.value(), storm::OptionalRef(clippingAbstraction));
255 } else {
256 ClippingBeliefAbstraction<BeliefType> clippingAbstraction(env, std::move(resolutions));
257 exploration.resumeClippingExploration(
258 info, terminalBeliefCallback, []() { return false; }, storm::NullRef, storm::OptionalRef(clippingAbstraction));
259 }
260 STORM_LOG_TRACE("Finished clipping phase.");
261 } else {
262 if (propertyInformation.kind == PropertyInformation::Kind::ExpectedTotalReachabilityReward) {
263 exploration.resumeExploration(info, terminalBeliefCallback, terminationCallback, propertyInformation.rewardModelName.value(), abstraction);
264 } else {
265 exploration.resumeExploration(info, terminalBeliefCallback, terminationCallback, storm::NullRef, abstraction);
266 }
267 }
268 swExplore.stop();
269 bool earlyExplorationStop = info.queue.hasNext();
270 statistics.completedExploration = !earlyExplorationStop;
271 statistics.discoveredBeliefs = info.discoveredBeliefs.getNumberOfBeliefIds();
272 statistics.exploredBeliefs = info.exploredBeliefs.size();
273 statistics.explorationTimeMilliseconds = swExplore.getTimeInMilliseconds();
274 if (earlyExplorationStop) {
275 STORM_LOG_INFO("Exploration stopped before all beliefs were explored. " << info.discoveredBeliefs.getNumberOfBeliefIds() << " beliefs discovered. "
276 << info.exploredBeliefs.size() << " beliefs explored.");
277 }
278
279 // Second, build the Belief MDP from the exploration information
280 STORM_LOG_INFO("Constructing the belief MDP.");
281 storm::utility::Stopwatch swBuild(true);
282 auto [beliefMdp, stateToBeliefMap] =
283 buildBeliefMdpFromInfo<PomdpModelType, BeliefType, BeliefMdpValueType, InfoType>(propertyInformation, *valueBounds.preprocessingBounds, info);
284 swBuild.stop();
285 {
286 std::stringstream stream;
287 beliefMdp->printModelInformationToStream(stream);
288 STORM_LOG_INFO("Constructed belief MDP:\n" << stream.str());
289 }
290 statistics.beliefMdpStates = beliefMdp->getNumberOfStates();
291 statistics.beliefMdpChoices = beliefMdp->getNumberOfChoices();
292 statistics.beliefMdpTransitions = beliefMdp->getNumberOfTransitions();
293 statistics.beliefMdpBuildTimeMilliseconds = swBuild.getTimeInMilliseconds();
294
295 // Finally, perform model checking on the belief MDP.
296 storm::utility::Stopwatch swCheck(true);
297 auto formula = createFormulaForBeliefMdp(propertyInformation);
298 storm::modelchecker::CheckTask<storm::logic::Formula, BeliefMdpValueType> task(*formula, !options.generatePolicy);
299 task.setProduceSchedulers(options.generatePolicy);
300 std::unique_ptr<storm::modelchecker::CheckResult> res(storm::api::verifyWithSparseEngine<BeliefMdpValueType>(env, beliefMdp, task));
301 swCheck.stop();
302 statistics.beliefMdpAnalysisTimeMilliseconds = swCheck.getTimeInMilliseconds();
303 STORM_LOG_INFO("Time for exploring beliefs: " << swExplore << ".");
304 STORM_LOG_INFO("Time for building the belief MDP: " << swBuild << ".");
305 STORM_LOG_INFO("Time for analyzing the belief MDP: " << swCheck << ".");
306 STORM_LOG_ASSERT(res, "Model checking of belief MDP did not return any result.");
307 STORM_LOG_ASSERT(res->isExplicitQuantitativeCheckResult(), "Model checking of belief MDP did not return result of expected type.");
308 STORM_LOG_ASSERT(beliefMdp->getInitialStates().getNumberOfSetBits() == 1, "Unexpected number of initial states for belief Mdp.");
309 auto const initState = beliefMdp->getInitialStates().getNextSetIndex(0);
310 if (options.generatePolicy) {
311 auto const& quantitativeResult = res->asExplicitQuantitativeCheckResult<BeliefMdpValueType>();
312 STORM_LOG_THROW(beliefMdp->hasChoiceLabeling(), storm::exceptions::UnexpectedException, "Policy extraction requires choice labels in the belief MDP.");
313 STORM_LOG_THROW(quantitativeResult.hasScheduler(), storm::exceptions::UnexpectedException,
314 "Model checking of the belief MDP did not produce a scheduler for policy extraction.");
315 std::unordered_map<uint64_t, uint32_t> beliefStateToObservationMap;
316 for (auto const& [state, beliefId] : stateToBeliefMap) {
317 beliefStateToObservationMap.emplace(state, info.discoveredBeliefs.getBeliefFromId(beliefId).observation());
318 }
319 std::optional<std::vector<storm::storage::Scheduler<typename PomdpModelType::ValueType>>> approximationSchedulers =
320 propertyInformation.dir == storm::OptimizationDirection::Minimize ? valueBounds.preprocessingBounds->upperSchedulers
321 : valueBounds.preprocessingBounds->lowerSchedulers;
322 policy::PolicyExtractor<PomdpModelType, typename BeliefType::ValueType, BeliefMdpValueType, PolicyValueType> extractor(
323 pomdp, *beliefMdp, beliefStateToObservationMap, quantitativeResult.getScheduler(), approximationSchedulers);
324 generatedPolicy = extractor.exportPolicyAsFiniteStateController();
325 }
326 return {res->asExplicitQuantitativeCheckResult<BeliefMdpValueType>()[initState], !earlyExplorationStop, std::move(statistics), std::move(generatedPolicy)};
327}

329template<typename PomdpModelType, typename BeliefType, typename BeliefMdpValueType, typename PolicyValueType>
330BeliefBasedModelCheckerResult<BeliefMdpValueType, PolicyValueType> checkRewardAwareUnfoldOrDiscretize(
331 storm::Environment const& env, PomdpModelType const& pomdp, PropertyInformation const& propertyInformation,
332 storm::pomdp::beliefs::BeliefBasedModelCheckerOptions<BeliefMdpValueType> const& options,
333 storage::BeliefExplorationBounds<typename PomdpModelType::ValueType> const& valueBounds,
334 RewardBoundedBeliefSplitter<BeliefMdpValueType, PomdpModelType, BeliefType>& rewardSplitter,
335 storm::OptionalRef<FreudenthalTriangulationBeliefAbstraction<BeliefType>> abstraction = {}) {
336 BeliefBasedModelCheckerStatistics statistics;
337 STORM_LOG_ASSERT(propertyInformation.kind == PropertyInformation::Kind::RewardBoundedReachabilityProbability, "Unexpected kind of property.");
338 STORM_LOG_THROW(!options.useClipping, storm::exceptions::NotSupportedException, "Grid clipping is not supported for reward-bounded POMDP properties.");
339 STORM_LOG_ASSERT(rewardSplitter.getNumberOfSetRewardModels() != 0, "rewardSplitter must have a reward model set for reward-aware belief MDP construction.");
340
341 STORM_LOG_INFO("Exploring the belief space.");
342
343 // First, explore the beliefs and its successors
344 using BeliefExplorationType = BeliefExploration<BeliefMdpValueType, PomdpModelType, BeliefType>;
345 storm::utility::Stopwatch swExplore(true);
346 BeliefExplorationType exploration(pomdp);
347 using InfoType = RewardAwareExplorationInformation<BeliefMdpValueType, BeliefType>;
348 auto info = exploration.template initializeExploration<InfoType>(pomdp.getNrObservations(), options.explorationQueueOrder);
349 info.generateChoiceLabeling = options.buildChoiceLabeling;
350
351 // Determine terminationCallback based on options
352 typename BeliefExplorationType::TerminationCallback terminationCallback =
353 getTerminationCallback<PomdpModelType, BeliefType, BeliefMdpValueType>(options, info, swExplore);
354
355 // Determine terminalBeliefCallback based on options
356 typename BeliefExplorationType::TerminalBeliefCallback terminalBeliefCallback =
357 getTerminalBeliefCallback<PomdpModelType, BeliefType, BeliefMdpValueType>(propertyInformation, options, *valueBounds.preprocessingBounds);
358
359 exploration.resumeRewardAwareExploration(info, terminalBeliefCallback, terminationCallback, rewardSplitter, abstraction);
360 swExplore.stop();
361 bool const earlyExplorationStop = info.queue.hasNext();
362 statistics.completedExploration = !earlyExplorationStop;
363 statistics.discoveredBeliefs = info.discoveredBeliefs.getNumberOfBeliefIds();
364 statistics.exploredBeliefs = info.exploredBeliefs.size();
365 statistics.explorationTimeMilliseconds = swExplore.getTimeInMilliseconds();
366 if (earlyExplorationStop) {
367 STORM_LOG_INFO("Exploration stopped before all beliefs were explored. " << info.discoveredBeliefs.getNumberOfBeliefIds() << " beliefs discovered. "
368 << info.exploredBeliefs.size() << " beliefs explored.");
369 }
370
371 // Second, build the Belief MDP from the exploration information
372 STORM_LOG_INFO("Constructing the belief MDP.");
373 storm::utility::Stopwatch swBuild(true);
374 auto [beliefMdp, stateToBeliefMap] =
375 buildBeliefMdpFromInfo<PomdpModelType, BeliefType, BeliefMdpValueType, InfoType>(propertyInformation, *valueBounds.preprocessingBounds, info);
376 swBuild.stop();
377 {
378 std::stringstream stream;
379 beliefMdp->printModelInformationToStream(stream);
380 STORM_LOG_INFO("Constructed belief MDP:\n" << stream.str());
381 }
382 statistics.beliefMdpStates = beliefMdp->getNumberOfStates();
383 statistics.beliefMdpChoices = beliefMdp->getNumberOfChoices();
384 statistics.beliefMdpTransitions = beliefMdp->getNumberOfTransitions();
385 statistics.beliefMdpBuildTimeMilliseconds = swBuild.getTimeInMilliseconds();
386
387 // Finally, perform model checking on the belief MDP.
388 auto formula = createFormulaForBeliefMdp(propertyInformation);
389 STORM_LOG_INFO("Analyzing property '" << *formula << "' on the belief MDP.");
390 storm::utility::Stopwatch swCheck(true);
391 std::shared_ptr<storm::models::sparse::Mdp<BeliefMdpValueType>> processedMdp = beliefMdp;
392 if (propertyInformation.kind == PropertyInformation::Kind::RewardBoundedReachabilityProbability) {
393 std::vector<std::string> rewardModelNames;
394 for (auto const& bnd : propertyInformation.rewardBounds) {
395 rewardModelNames.push_back(bnd.rewardModelName);
396 }
397 processedMdp = storm::transformer::transformTransitionToActionRewards<BeliefMdpValueType>(processedMdp, rewardModelNames)
398 .model->template as<storm::models::sparse::Mdp<BeliefMdpValueType>>();
399 double increase = static_cast<double>(processedMdp->getNumberOfStates()) / static_cast<double>(beliefMdp->getNumberOfStates());
400 STORM_LOG_INFO("Transformation of transition rewards resulted in a model with " << processedMdp->getNumberOfStates() << " states. " << increase
401 << " times more states than the original belief MDP.");
402
403 // Cut away states that can not reach the target
404 auto targetStates = processedMdp->getStateLabeling().getStates("target");
405 storm::storage::BitVector allStates(targetStates.size(), true);
406 storm::storage::BitVector probGreaterZeroStates;
407 if (storm::solver::maximize(propertyInformation.dir)) {
408 probGreaterZeroStates = storm::utility::graph::performProbGreater0E(processedMdp->getBackwardTransitions(), allStates, targetStates);
409 } else {
410 probGreaterZeroStates =
411 storm::utility::graph::performProbGreater0A(processedMdp->getTransitionMatrix(), processedMdp->getTransitionMatrix().getRowGroupIndices(),
412 processedMdp->getBackwardTransitions(), allStates, targetStates);
413 }
414 auto mergingResult = storm::transformer::GoalStateMerger<BeliefMdpValueType>(*processedMdp)
415 .mergeTargetAndSinkStates(probGreaterZeroStates, ~probGreaterZeroStates, ~allStates, rewardModelNames);
416 processedMdp = mergingResult.model->template as<storm::models::sparse::Mdp<BeliefMdpValueType>>();
417 STORM_LOG_INFO("Merging of sink states resulted in a model with " << processedMdp->getNumberOfStates() << " states.");
418 }
419 if (processedMdp != beliefMdp) {
420 statistics.processedMdpStates = processedMdp->getNumberOfStates();
421 statistics.processedMdpChoices = processedMdp->getNumberOfChoices();
422 statistics.processedMdpTransitions = processedMdp->getNumberOfTransitions();
423 }
424 storm::modelchecker::CheckTask<storm::logic::Formula, BeliefMdpValueType> task(*formula, true);
425 std::unique_ptr<storm::modelchecker::CheckResult> res(storm::api::verifyWithSparseEngine<BeliefMdpValueType>(env, processedMdp, task));
426 swCheck.stop();
427 statistics.beliefMdpAnalysisTimeMilliseconds = swCheck.getTimeInMilliseconds();
428 STORM_LOG_INFO("Time for exploring beliefs: " << swExplore << ".");
429 STORM_LOG_INFO("Time for building the belief MDP: " << swBuild << ".");
