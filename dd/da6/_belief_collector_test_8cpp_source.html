15template<typename ValueType>
16Belief<ValueType> makeBelief(BeliefStateType state = 0, BeliefObservationType observation = 0, double firstProbability = 0.5) {
17 BeliefBuilder<Belief<ValueType>> builder;
18 builder.setObservation(observation);
19 builder.addValue(state, storm::utility::convertNumber<ValueType>(firstProbability));
20 builder.addValue(state + 1, storm::utility::convertNumber<ValueType>(1.0 - firstProbability));

25ValueType const* firstValueAddress(Belief<ValueType> const& belief) {
