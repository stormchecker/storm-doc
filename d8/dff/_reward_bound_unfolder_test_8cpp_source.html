42std::shared_ptr<storm::logic::Formula const> makeRewardBoundedFormula(
43 std::optional<int64_t> lowerBound, std::optional<int64_t> upperBound,
44 storm::solver::OptimizationDirection direction = storm::solver::OptimizationDirection::Maximize) {
45 static auto const expressionManager = std::make_shared<storm::expressions::ExpressionManager>();
46 std::optional<storm::logic::TimeBound> lowerTimeBound;
47 std::optional<storm::logic::TimeBound> upperTimeBound;
48 if (lowerBound) {
49 lowerTimeBound.emplace(false, expressionManager->integer(*lowerBound));
50 }
51 if (upperBound) {
52 upperTimeBound.emplace(false, expressionManager->integer(*upperBound));
53 }
54
55 auto pathFormula = std::make_shared<storm::logic::BoundedUntilFormula>(std::make_shared<storm::logic::BooleanLiteralFormula>(true),
56 std::make_shared<storm::logic::AtomicLabelFormula>("goal"), lowerTimeBound,
57 upperTimeBound, storm::logic::TimeBoundReference(boost::optional<std::string>{"r"}));
58 return std::make_shared<storm::logic::ProbabilityOperatorFormula>(
59 std::move(pathFormula), storm::logic::OperatorInformation(boost::optional<storm::solver::OptimizationDirection>{direction}));
60}
