17// The LP solvers that the SCP tests are run against. Each entry is only included when the
18// corresponding solver was compiled in, so the set adapts to the configuration. Z3 is deliberately
19// excluded: its LP backend is an SMT encoding that is not intended as a general-purpose LP solver
20// (and is considerably slower).
21inline std::vector<storm::solver::LpSolverType> lpSolversUnderTest() {
22 std::vector<storm::solver::LpSolverType> solvers;
23#if defined STORM_HAVE_GLPK
24 solvers.push_back(storm::solver::LpSolverType::Glpk);
25#endif
26#if defined STORM_HAVE_GUROBI
27 solvers.push_back(storm::solver::LpSolverType::Gurobi);
28#endif
29#if defined STORM_HAVE_HIGHS
30 solvers.push_back(storm::solver::LpSolverType::Highs);
31#endif
32#if defined STORM_HAVE_SOPLEX
33 solvers.push_back(storm::solver::LpSolverType::Soplex);
34#endif
35 return solvers;
36}
