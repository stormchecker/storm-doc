18inline storm::expressions::Expression pigeonholeFormula(storm::expressions::ExpressionManager& manager, uint64_t holes) {
19 uint64_t const pigeons = holes + 1;
20 std::vector<std::vector<storm::expressions::Variable>> inHole(pigeons);
21 for (uint64_t p = 0; p < pigeons; ++p) {
22 for (uint64_t h = 0; h < holes; ++h) {
23 inHole[p].push_back(manager.declareBooleanVariable("p" + std::to_string(p) + "h" + std::to_string(h)));
24 }
25 }
26 storm::expressions::Expression formula = manager.boolean(true);
27 // Each pigeon sits in some hole.
28 for (uint64_t p = 0; p < pigeons; ++p) {
29 storm::expressions::Expression somewhere = manager.boolean(false);
30 for (uint64_t h = 0; h < holes; ++h) {
31 somewhere = somewhere || inHole[p][h];
32 }
33 formula = formula && somewhere;
34 }
35 // No two pigeons share a hole.
36 for (uint64_t h = 0; h < holes; ++h) {
37 for (uint64_t p1 = 0; p1 < pigeons; ++p1) {
38 for (uint64_t p2 = p1 + 1; p2 < pigeons; ++p2) {
39 formula = formula && !(inHole[p1][h] && inHole[p2][h]);
40 }
41 }
42 }
43 return formula;
44}

49inline void checkSmtSolverTimeout(storm::solver::SmtSolver& solver, storm::expressions::ExpressionManager& manager) {
