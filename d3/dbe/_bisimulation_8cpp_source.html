23template<typename ValueType>
24ReturnType<ValueType> performBisimulationMinimization(storm::models::sparse::Model<ValueType> const& model,
25 std::vector<std::shared_ptr<storm::logic::Formula const>> const& formulas, Options const& options) {
26 // Step 0: Sanity checks and set-up
27 if constexpr (storm::IsIntervalType<ValueType>) {
28 STORM_LOG_THROW(false, storm::exceptions::NotSupportedException, "Bisimulation is not supported for Interval models.");
29 }
30 STORM_LOG_THROW(options.tolerance >= storm::utility::zero<storm::RationalNumber>(), storm::exceptions::InvalidArgumentException,
31 "Tolerance for bisimulation minimization must be non-negative, but was " << options.tolerance << ".");
32 bool const isWeak = options.bisimulationType == BisimulationType::Weak;
33 if (isWeak) {
34 STORM_LOG_THROW(!model.isNondeterministicModel(), storm::exceptions::NotSupportedException,
35 "Weak bisimulation is only supported for deterministic models, but the given model is of type " << model.getType() << ".");
36 STORM_LOG_WARN_COND(!options.preferSignatureRefinement,
37 "Using splitter-based refinement because weak bisimulation is not supported for signature-based refinement.");
38 }
39 // Weak bisimulation for signature-based refinement is not supported.
40 // Deterministic models default to splitter-based refinement, which is usually faster
41 bool const useSignatureRefinement = !isWeak && (model.isNondeterministicModel() || options.preferSignatureRefinement);
42 STORM_LOG_INFO("Using " << (useSignatureRefinement ? "signature" : "splitter") << "-based refinement for bisimulation minimization.");
43 storm::utility::Stopwatch sw(true);
44 STORM_LOG_STATISTICS("-------- Bisimulation Minimization --------");
45
46 // Step 1: Obtain an initial partition based on what needs to be preserved (labels, rewards, ...)
47 storm::bisimulation::Initialization<ValueType> initialization(model, options, formulas);
48 auto const preservationInformation = initialization.getPreservationInformation();
49 auto const choiceClasses = initialization.getChoiceClasses();
50 auto partition = initialization.getInitialStatePartition(choiceClasses);
51 STORM_LOG_STATISTICS(sw << " seconds for initial partition (" << partition.getNumberOfBlocks() << " blocks).");
52 sw.restart();
53
54 // Step 2: Apply refinement using the initial partition and choiceClasses. Initialize QuotientData.
55 std::optional<storm::bisimulation::QuotientData<ValueType>> quotientData;
56 // commonly called right after refinement.
57 auto initializeQuotientData = [&partition, &sw,
58 &quotientData](storm::OptionalRef<storm::storage::BitVector const> preferredRepresentatives = storm::NullRef) {
59 STORM_LOG_STATISTICS(sw << " seconds for refinement (" << partition.getNumberOfBlocks() << " blocks).");
60 sw.restart();
61 quotientData.emplace(partition, preferredRepresentatives);
62 };
63 if constexpr (storm::IsIntervalType<ValueType>) {
64 // Unreachable, as Step 0 already threw for interval models. The branch remains so that the refinement below is not instantiated for them.
65 STORM_LOG_ASSERT(false, "Unexpected value type.");
66 } else if (isWeak) {
67 // Weak bisimulation additionally needs to know the divergent, step sensitive and silent states. Computing them further refines the partition.
68 // Both that computation and the refinement need the transposed transition matrix, so we build it once and share it.
69 auto const backwardTransitions = model.getBackwardTransitions();
70 auto weakData = initialization.getWeakBisimulationData(partition, backwardTransitions, preservationInformation);
71 STORM_LOG_STATISTICS(sw << " seconds for weak bisimulation initialization (" << partition.getNumberOfBlocks() << " blocks).");
72 sw.restart();
73 if (model.isDiscreteTimeModel()) {
74 storm::bisimulation::performSplitterBasedRefinement<ValueType, SplitterRefinementMode::WeakDiscreteTime>(
75 model, backwardTransitions, partition, storm::utility::convertNumber<ValueType>(options.tolerance), weakData);
76 } else {
77 storm::bisimulation::performSplitterBasedRefinement<ValueType, SplitterRefinementMode::WeakContinuousTime>(
78 model, backwardTransitions, partition, storm::utility::convertNumber<ValueType>(options.tolerance), weakData);
79 }
80 // For weak bisimulation, the quotient transitions have to be derived from a state that can actually leave its block.
81 auto const nonSilentStates = ~weakData.silentStates;
82 initializeQuotientData(nonSilentStates);
83 quotientData->weakData.emplace(std::move(weakData));
84 if (options.createQuotientChoiceMapping) {
85 // For deterministic models, quotient state and choice mappings are identical.
86 STORM_LOG_ASSERT(!model.isNondeterministicModel(), "Expected a deterministic model.");
87 quotientData->toQuotientChoice = quotientData->toQuotientState;
88 }
89 } else if (useSignatureRefinement && storm::utility::isZero(options.tolerance)) {
90 storm::bisimulation::Signatures<ValueType, SignatureMode::Exact> signatures(model, choiceClasses, partition);
91 storm::bisimulation::performSignatureBasedRefinement(model, partition, signatures);
92 initializeQuotientData();
93 signatures.extendQuotientData(quotientData.value(), options.createQuotientChoiceMapping);
94 } else if (useSignatureRefinement && !storm::utility::isZero(options.tolerance)) {
95 if constexpr (std::is_same_v<ValueType, storm::RationalFunction>) {
96 STORM_LOG_THROW(false, storm::exceptions::NotSupportedException,
97 "Bisimulation with positive tolerance " << options.tolerance << " (approximately "
98 << storm::utility::convertNumber<double>(options.tolerance)
99 << ") is not supported for parametric models.");
100 } else {
101 storm::bisimulation::Signatures<ValueType, SignatureMode::Approximative> signatures(model, choiceClasses, partition,
102 storm::utility::convertNumber<ValueType>(options.tolerance));
103 storm::bisimulation::performSignatureBasedRefinement(model, partition, signatures);
104 initializeQuotientData();
105 signatures.extendQuotientData(quotientData.value(), options.createQuotientChoiceMapping);
106 }
107 } else {
108 // For deterministic models we do splitter based refinement.
109 STORM_LOG_ASSERT(!model.isNondeterministicModel(), "Splitter-based refinement is only applicable to deterministic models.");
110 storm::bisimulation::performSplitterBasedRefinement<ValueType>(model, model.getBackwardTransitions(), partition,
111 storm::utility::convertNumber<ValueType>(options.tolerance));
112 initializeQuotientData();
113 if (options.createQuotientChoiceMapping) {
114 // For deterministic models, quotient state and choice mappings are identical.
115 quotientData->toQuotientChoice = quotientData->toQuotientState;
116 }
117 }
118
119 // Step 3: Extract the quotient
120 auto quotientModel = storm::bisimulation::Quotient<ValueType>::buildFromPartition(model, options, preservationInformation, quotientData.value());
121 STORM_LOG_STATISTICS(sw << " seconds for quotient extraction.");
122 STORM_LOG_STATISTICS("-------------------------------------------");
123
124 return {.quotient = std::move(quotientModel),
125 .toQuotientStateMapping = std::move(quotientData->toQuotientState),
126 .toQuotientChoiceMapping = std::move(quotientData->toQuotientChoice)};
127}
