14storm::storage::SparseMatrix<double> createRandomMatrix(std::mt19937_64& rng, uint64_t groups, uint64_t columns, bool grouped, double density) {
15 std::uniform_real_distribution<double> coin(0.0, 1.0);
16 std::uniform_int_distribution<uint64_t> groupSize(1, 3);
17 storm::storage::SparseMatrixBuilder<double> builder(0, columns, 0, true, grouped);
18 uint64_t row = 0;
19 for (uint64_t g = 0; g < groups; ++g) {
20 uint64_t const size = grouped ? groupSize(rng) : 1;
21 if (grouped) {
22 builder.newRowGroup(row);
23 }
24 for (uint64_t i = 0; i < size; ++i, ++row) {
25 for (uint64_t c = 0; c < columns; ++c) {
26 if (coin(rng) < density) {
27 builder.addNextValue(row, c, coin(rng) + 0.5);
28 }
29 }
30 }
31 }
32 return builder.build(row, columns, grouped ? groups : 0);
33}

35storm::storage::BitVector randomBitVector(std::mt19937_64& rng, uint64_t size, double p) {
36 std::uniform_real_distribution<double> coin(0.0, 1.0);
37 storm::storage::BitVector result(size);
38 for (uint64_t i = 0; i < size; ++i) {

41 if (result.empty()) {

48storm::storage::SparseMatrix<double> referenceSubmatrix(storm::storage::SparseMatrix<double> const& matrix, bool useGroups,
49 storm::storage::BitVector const& rowConstraint, storm::storage::BitVector const& columnConstraint,
50 bool insertDiagonalEntries, storm::storage::BitVector const& makeZeroColumns) {
51 using index_type = uint64_t;
52 std::vector<index_type> rowGroupIndices;
53 if (useGroups) {
54 rowGroupIndices = matrix.getRowGroupIndices();
55 } else {
56 for (index_type i = 0; i <= matrix.getRowCount(); ++i) {
57 rowGroupIndices.push_back(i);
58 }
59 }
60 index_type const submatrixColumnCount = columnConstraint.getNumberOfSetBits();
61 std::vector<index_type> columnBitsSetBeforeIndex = columnConstraint.getNumberOfSetBitsBeforeIndices();
62 std::vector<index_type> rowBitsSetBeforeIndex = rowConstraint.getNumberOfSetBitsBeforeIndices();
63 auto keep = [&](index_type col) { return columnConstraint.get(col) && (makeZeroColumns.size() == 0 || !makeZeroColumns.get(col)); };
64
65 index_type subEntries = 0, subRows = 0, rowGroupCount = 0;
66 for (uint64_t index : rowConstraint) {
67 subRows += rowGroupIndices[index + 1] - rowGroupIndices[index];
68 for (index_type i = rowGroupIndices[index]; i < rowGroupIndices[index + 1]; ++i) {
69 bool foundDiagonalElement = false;
70 for (auto it = matrix.begin(i); it != matrix.end(i); ++it) {
71 if (keep(it->getColumn())) {
72 ++subEntries;
73 if (columnBitsSetBeforeIndex[it->getColumn()] == rowBitsSetBeforeIndex[index]) {
74 foundDiagonalElement = true;
75 }
76 }
77 }
78 if (insertDiagonalEntries && !foundDiagonalElement && rowGroupCount < submatrixColumnCount) {
79 ++subEntries;
80 }
81 }
82 ++rowGroupCount;
83 }
84 bool const groupedResult = !matrix.hasTrivialRowGrouping();
85 storm::storage::SparseMatrixBuilder<double> builder(subRows, submatrixColumnCount, subEntries, true, groupedResult);
86 rowGroupCount = 0;
87 index_type rowCount = 0;
88 for (uint64_t index : rowConstraint) {
89 if (groupedResult) {
90 builder.newRowGroup(rowCount);
91 }
92 for (index_type i = rowGroupIndices[index]; i < rowGroupIndices[index + 1]; ++i) {
93 bool insertedDiagonalElement = false;
94 for (auto it = matrix.begin(i); it != matrix.end(i); ++it) {
95 if (keep(it->getColumn())) {
96 if (columnBitsSetBeforeIndex[it->getColumn()] == rowBitsSetBeforeIndex[index]) {
97 insertedDiagonalElement = true;
98 } else if (insertDiagonalEntries && !insertedDiagonalElement && columnBitsSetBeforeIndex[it->getColumn()] > rowBitsSetBeforeIndex[index]) {
99 builder.addNextValue(rowCount, rowGroupCount, 0.0);
100 insertedDiagonalElement = true;
101 }
102 builder.addNextValue(rowCount, columnBitsSetBeforeIndex[it->getColumn()], it->getValue());
103 }
104 }
105 if (insertDiagonalEntries && !insertedDiagonalElement && rowGroupCount < submatrixColumnCount) {
106 builder.addNextValue(rowCount, rowGroupCount, 0.0);
107 }
108 ++rowCount;
109 }
110 ++rowGroupCount;
111 }
112 auto result = builder.build();
113 if (groupedResult && !useGroups) {
114 // Row groups of the result: one group for each original group that has at least one selected row.
115 std::vector<index_type> newGroups{0};
116 auto selectedRowIt = rowConstraint.begin();
117 for (index_type group = 0; group < matrix.getRowGroupCount(); ++group) {
118 index_type newRowCount = 0;
119 while (selectedRowIt != rowConstraint.end() && *selectedRowIt < matrix.getRowGroupIndices()[group + 1]) {
120 ++selectedRowIt;
121 ++newRowCount;
122 }
123 if (newRowCount > 0) {
124 newGroups.push_back(newGroups.back() + newRowCount);
125 }
126 }
127 result.setRowGroupIndices(newGroups);
128 }
129 return result;
130}

153TEST(SubmatrixBuilderTest, AgreesWithGetSubmatrix) {
154 std::mt19937_64 rng(42);
155 for (bool grouped : {false, true}) {
156 for (int iteration = 0; iteration < 200; ++iteration) {
157 uint64_t const n = 1 + rng() % 12;
158 auto matrix = createRandomMatrix(rng, n, n, grouped, 0.4);
159 storm::storage::SubmatrixBuilder<double> builder(matrix);
160 // Reuse the builder to also check that the internal state is reset properly.
161 for (int j = 0; j < 5; ++j) {
162 auto const groupConstraint = randomBitVector(rng, matrix.getRowGroupCount(), 0.6);
163 auto const rowConstraint = randomBitVector(rng, matrix.getRowCount(), 0.6);
164 auto const columnConstraint = randomBitVector(rng, matrix.getColumnCount(), 0.6);
165 auto const zeroColumns = randomBitVector(rng, matrix.getColumnCount(), 0.3);

170 builder.getByRowGroupConstraint(groupConstraint, columnConstraint, diagonal, zeroColumns));
171 expectEqual(referenceSubmatrix(matrix, false, rowConstraint, columnConstraint, diagonal, {}),
172 builder.getByRowConstraint(rowConstraint, columnConstraint, diagonal));
