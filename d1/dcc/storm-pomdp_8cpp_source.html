54template<typename ValueType>
55bool performPreprocessing(std::shared_ptr<storm::models::sparse::Pomdp<ValueType>>& pomdp, storm::pomdp::analysis::FormulaInformation& formulaInfo,
56 storm::logic::Formula const& formula) {
57 auto const& pomdpSettings = storm::settings::getModule<storm::settings::modules::POMDPSettings>();
58 bool preprocessingPerformed = false;
59 if (pomdpSettings.isSelfloopReductionSet()) {
60 storm::transformer::GlobalPOMDPSelfLoopEliminator<ValueType> selfLoopEliminator(*pomdp);
61 if (selfLoopEliminator.preservesFormula(formula)) {
62 STORM_PRINT_AND_LOG("Eliminating self-loop choices ...");
63 uint64_t oldChoiceCount = pomdp->getNumberOfChoices();
64 pomdp = selfLoopEliminator.transform();
65 STORM_PRINT_AND_LOG(oldChoiceCount - pomdp->getNumberOfChoices() << " choices eliminated through self-loop elimination.\n");
66 preprocessingPerformed = true;
67 } else {
68 STORM_PRINT_AND_LOG("Not eliminating self-loop choices as it does not preserve the formula.\n");
69 }
70 }
71 if (pomdpSettings.isQualitativeReductionSet() && formulaInfo.isNonNestedReachabilityProbability() && !formulaInfo.isBounded()) {
72 storm::analysis::QualitativeAnalysisOnGraphs<ValueType> qualitativeAnalysis(*pomdp);
73 STORM_PRINT_AND_LOG("Computing states with probability 0 ...");
74 storm::storage::BitVector prob0States = qualitativeAnalysis.analyseProb0(formula.asProbabilityOperatorFormula());
75 STORM_PRINT_AND_LOG(" done. " << prob0States.getNumberOfSetBits() << " states found.\n");
76 STORM_PRINT_AND_LOG("Computing states with probability 1 ...");
77 storm::storage::BitVector prob1States = qualitativeAnalysis.analyseProb1(formula.asProbabilityOperatorFormula());
78 STORM_PRINT_AND_LOG(" done. " << prob1States.getNumberOfSetBits() << " states found.\n");
79 storm::pomdp::transformer::KnownProbabilityTransformer<ValueType> kpt = storm::pomdp::transformer::KnownProbabilityTransformer<ValueType>();
80 pomdp = kpt.transform(*pomdp, prob0States, prob1States);
81 // Update formulaInfo to changes from Preprocessing
82 formulaInfo.updateTargetStates(*pomdp, std::move(prob1States));
83 formulaInfo.updateSinkStates(*pomdp, std::move(prob0States));
84 preprocessingPerformed = true;
85 }
86 return preprocessingPerformed;
87}

176template<typename ValueType>
177void performQualitativeAnalysis(storm::Environment const& env, std::shared_ptr<storm::models::sparse::Pomdp<ValueType>> const& origpomdp,
178 storm::pomdp::analysis::FormulaInformation const& formulaInfo, storm::logic::Formula const& formula) {
179 auto const& qualSettings = storm::settings::getModule<storm::settings::modules::QualitativePOMDPAnalysisSettings>();
180 auto const& coreSettings = storm::settings::getModule<storm::settings::modules::CoreSettings>();
181 std::stringstream sstr;
182 origpomdp->printModelInformationToStream(sstr);
183 STORM_LOG_INFO(sstr.str());
184 STORM_LOG_THROW(formulaInfo.isNonNestedReachabilityProbability() && !formulaInfo.isBounded(), storm::exceptions::NotSupportedException,
185 "Qualitative memoryless scheduler search is not implemented for this property type.");
186 STORM_LOG_TRACE("Run qualitative preprocessing...");
187 storm::models::sparse::Pomdp<ValueType> pomdp(*origpomdp);
188 storm::analysis::QualitativeAnalysisOnGraphs<ValueType> qualitativeAnalysis(pomdp);
189 // After preprocessing, this might be done cheaper.
190 storm::storage::BitVector surelyNotAlmostSurelyReachTarget = qualitativeAnalysis.analyseProbSmaller1(formula.asProbabilityOperatorFormula());
191 pomdp.getTransitionMatrix().makeRowGroupsAbsorbing(surelyNotAlmostSurelyReachTarget);
192 storm::storage::BitVector targetStates = qualitativeAnalysis.analyseProb1(formula.asProbabilityOperatorFormula());
193 bool computedSomething = false;
194 if (qualSettings.isMemlessSearchSet()) {
195 computedSomething = true;
196 std::shared_ptr<storm::utility::solver::SmtSolverFactory> smtSolverFactory = std::make_shared<storm::utility::solver::SmtSolverFactory>();
197 uint64_t lookahead = qualSettings.getLookahead();
198 if (lookahead == 0) {
199 lookahead = pomdp.getNumberOfStates();
200 }
201 if (qualSettings.getMemlessSearchMethod() == "one-shot") {
202 storm::pomdp::OneShotPolicySearch<ValueType> memlessSearch(env, pomdp, targetStates, surelyNotAlmostSurelyReachTarget, smtSolverFactory);
203 if (qualSettings.isWinningRegionSet()) {
204 STORM_LOG_ERROR("Computing winning regions is not supported by the one-shot method.");
205 } else {
206 bool result = memlessSearch.analyzeForInitialStates(lookahead);
207 if (result) {
208 STORM_PRINT_AND_LOG("From initial state, one can almost-surely reach the target.\n");
209 } else {
210 STORM_PRINT_AND_LOG("From initial state, one may not almost-surely reach the target .\n");
211 }
212 }
213 } else if (qualSettings.getMemlessSearchMethod() == "iterative") {
214 storm::pomdp::MemlessSearchOptions options = fillMemlessSearchOptionsFromSettings();
215 storm::pomdp::IterativePolicySearch<ValueType> search(env, pomdp, targetStates, surelyNotAlmostSurelyReachTarget, smtSolverFactory, options);
216 if (qualSettings.isWinningRegionSet()) {
217 search.computeWinningRegion(lookahead);
218 } else {
219 bool result = search.analyzeForInitialStates(lookahead);
220 if (result) {
221 STORM_PRINT_AND_LOG("From initial state, one can almost-surely reach the target.");
222 } else {
223 // TODO consider adding check for end components to improve this message.
224 STORM_PRINT_AND_LOG("From initial state, one may not almost-surely reach the target.");
225 }
226 }
227
228 if (qualSettings.isPrintWinningRegionSet()) {
229 search.getLastWinningRegion().print();
230 std::cout << '\n';
231 }
232 if (qualSettings.isExportWinningRegionSet()) {
233 std::size_t hash = pomdp.hash();
234 search.getLastWinningRegion().storeToFile(qualSettings.exportWinningRegionPath(), "model hash: " + std::to_string(hash));
235 }
236
237 search.finalizeStatistics();
238 if (pomdp.getInitialStates().getNumberOfSetBits() == 1) {
239 uint64_t initialState = pomdp.getInitialStates().getNextSetIndex(0);
240 uint64_t initialObservation = pomdp.getObservation(initialState);
241 // TODO this is inefficient.
242 uint64_t offset = 0;
243 for (uint64_t state = 0; state < pomdp.getNumberOfStates(); ++state) {
244 if (state == initialState) {
245 break;
246 }
247 if (pomdp.getObservation(state) == initialObservation) {
248 ++offset;
249 }
250 }
251
252 if (search.getLastWinningRegion().isWinning(initialObservation, offset)) {
253 STORM_PRINT_AND_LOG("Initial state is safe!\n");
254 } else {
255 STORM_PRINT_AND_LOG("Initial state may not be safe.\n");
256 }
257 } else {
258 STORM_LOG_WARN("Output for multiple initial states is incomplete");
259 }
260
261 if (coreSettings.isShowStatisticsSet()) {
262 STORM_PRINT_AND_LOG("#STATS Number of belief support states: " << search.getLastWinningRegion().beliefSupportStates() << '\n');
263 if (qualSettings.computeExpensiveStats()) {
264 auto wbss = search.getLastWinningRegion().computeNrWinningBeliefs();
265 STORM_PRINT_AND_LOG("#STATS Number of winning belief support states: [" << wbss.first << "," << wbss.second << "]");
266 }
267 search.getStatistics().print();
268 }
269
270 } else {
271 STORM_LOG_ERROR("This method is not implemented.");
272 }
273 }
274 if (qualSettings.isComputeOnBeliefSupportSet()) {
275 computedSomething = true;
276 storm::pomdp::qualitative::JaniBeliefSupportMdpGenerator<ValueType> janicreator(pomdp);
277 janicreator.generate(targetStates, surelyNotAlmostSurelyReachTarget);
278 bool initialOnly = !qualSettings.isWinningRegionSet();
279 janicreator.verifySymbolic(env, initialOnly);
280 STORM_PRINT_AND_LOG("Initial state is safe: " << janicreator.isInitialWinning() << "\n");
281 }
282 STORM_LOG_THROW(computedSomething, storm::exceptions::InvalidSettingsException, "Nothing to be done, did you forget to set a method?");
283}

287 storm::pomdp::analysis::FormulaInformation const& formulaInfo, storm::logic::Formula const& formula) {
288 auto const& pomdpSettings = storm::settings::getModule<storm::settings::modules::POMDPSettings>();
289 auto const& belExplSettings = storm::settings::getModule<storm::settings::modules::BeliefExplorationSettings>();
290 STORM_LOG_THROW(!formulaInfo.isBounded() || !belExplSettings.isUseClippingSet(), storm::exceptions::NotSupportedException,
291 "Grid clipping is not supported for reward-bounded POMDP properties.");
292
293 bool const policyExportRequested = pomdpSettings.isExportPolicySet();
294 STORM_LOG_THROW(!policyExportRequested || !belExplSettings.isCutZeroGapSet(), storm::exceptions::InvalidSettingsException,

296 "allow policy export.");
297 std::optional<PolicyExportFormat> policyExportFormat;
298 if (policyExportRequested) {
299 policyExportFormat = getPolicyExportFormat(pomdpSettings.getExportPolicyFilename());
300 }
301
302 storm::pomdp::beliefs::BeliefBasedModelCheckerOptions<BeliefMDPType> revisedOptions;
303 // We hard-code this to FIFO for now to mimic the legacy behaviour
304 revisedOptions.explorationQueueOrder = beliefs::ExplorationQueueOrder::FIFO;
305 if (belExplSettings.getExplorationTimeLimit() != 0) {
306 revisedOptions.maxExplorationTime = belExplSettings.getExplorationTimeLimit();
307 }
308 revisedOptions.cutZeroGap = belExplSettings.isCutZeroGapSet();
309
310 std::shared_ptr<storm::models::sparse::Pomdp<ValueType>> preprocessedPomdpPtr = pomdp;
311
312 std::optional<std::string> rewardModelName;
313 std::set<uint32_t> targetObservations;
314 if (formulaInfo.isNonNestedReachabilityProbability() || formulaInfo.isNonNestedExpectedRewardFormula()) {
315 if (formulaInfo.getTargetStates().observationClosed) {
316 targetObservations = formulaInfo.getTargetStates().observations;
317 } else {
318 storm::transformer::MakeStateSetObservationClosed<ValueType> obsCloser(pomdp);

320 }
321 if (formulaInfo.isNonNestedReachabilityProbability()) {
322 if (!formulaInfo.getSinkStates().empty()) {
323 storm::storage::sparse::ModelComponents<ValueType> components;
324 components.stateLabeling = preprocessedPomdpPtr->getStateLabeling();
325 components.rewardModels = preprocessedPomdpPtr->getRewardModels();
326 auto matrix = preprocessedPomdpPtr->getTransitionMatrix();
327 matrix.makeRowGroupsAbsorbing(formulaInfo.getSinkStates().states, true);
328 STORM_LOG_ASSERT(matrix.isProbabilistic(storm::utility::zero<ValueType>()), "Resulting transition matrix is not a probability matrix.");
329 STORM_LOG_ASSERT(matrix.hasOnlyPositiveEntries(), "Resulting transition matrix has non-positive entries.");
330 components.transitionMatrix = matrix;
331 components.observabilityClasses = preprocessedPomdpPtr->getObservations();
332 if (preprocessedPomdpPtr->hasChoiceLabeling()) {
333 components.choiceLabeling = preprocessedPomdpPtr->getChoiceLabeling();
334 }
335 if (preprocessedPomdpPtr->hasObservationValuations()) {
336 components.observationValuations = preprocessedPomdpPtr->getObservationValuations();
337 }

339 auto reachableFromSinkStates =
340 storm::utility::graph::getReachableStates(preprocessedPomdpPtr->getTransitionMatrix(), formulaInfo.getSinkStates().states,
341 formulaInfo.getSinkStates().states, ~formulaInfo.getSinkStates().states);
342 reachableFromSinkStates &= ~formulaInfo.getSinkStates().states;
343 STORM_LOG_THROW(reachableFromSinkStates.empty(), storm::exceptions::NotSupportedException,
344 "There are sink states that can reach non-sink states. This is currently not supported");
345 }
346 } else {
347 // Expected reward formula!
348 rewardModelName = formulaInfo.getRewardModelName();
349 }
350 } else {
351 STORM_LOG_THROW(false, storm::exceptions::NotSupportedException, "Unsupported formula '" << formula << "'.");
352 }
353 std::optional<storm::storage::BitVector> optionalTargetStates;
354 optionalTargetStates = formulaInfo.getTargetStates().states;

356 STORM_LOG_INFO("Detected that the belief MDP is finite.");
357 }

360
361 if (!formulaInfo.isBounded()) {
362 // Precompute initial bounds used for cut-offs and clipping
363 if (belExplSettings.isInexactPreprocessingSet()) {
364 STORM_LOG_WARN("Using inexact preprocessing for belief exploration can lead to inaccurate results.");
365 std::shared_ptr<storm::models::sparse::Pomdp<double>> preprocessedPomdpDouble;
366 if constexpr (std::is_same_v<ValueType, storm::RationalNumber>) {
367 preprocessedPomdpDouble = storm::transformer::sparseRationalModelToDouble(

369 ->template as<storm::models::sparse::Pomdp<double>>();
370 } else {
371 STORM_LOG_WARN("Inexact preprocessing option has no effect for floating-point models.");
372 preprocessedPomdpDouble = preprocessedPomdpPtr;

375 modelchecker::PreprocessingPomdpValueBoundsModelChecker<storm::models::sparse::Pomdp<double>>(*preprocessedPomdpDouble);
376 beliefExplorationBounds.preprocessingBounds = inExactPreProcessingMC.getValueBounds(env, formula).template toValueType<ValueType>();

378 beliefExplorationBounds.extremeBounds = inExactPreProcessingMC.getExtremeValueBound(env, formula).template toValueType<ValueType>();
379 }
380 } else {
381 auto preProcessingMC = modelchecker::PreprocessingPomdpValueBoundsModelChecker<storm::models::sparse::Pomdp<ValueType>>(*preprocessedPomdpPtr);
382 beliefExplorationBounds.preprocessingBounds = preProcessingMC.getValueBounds(env, formula);
383 if (belExplSettings.isUseClippingSet() && rewardModelName) {
384 beliefExplorationBounds.extremeBounds = preProcessingMC.getExtremeValueBound(env, formula);
385 }
386 }
387 } else {
388 // We only consider bounded probability formulae, so we can use 0-1 bounds
389 storm::pomdp::storage::PreprocessingPomdpValueBounds<ValueType> zeroOneValueBound;
390 zeroOneValueBound.lower.push_back(std::vector<ValueType>(preprocessedPomdpPtr->getNumberOfStates(), storm::utility::zero<ValueType>()));
391 zeroOneValueBound.upper.push_back(std::vector<ValueType>(preprocessedPomdpPtr->getNumberOfStates(), storm::utility::one<ValueType>()));
392
393 beliefExplorationBounds.preprocessingBounds = zeroOneValueBound;
394 }
395
396 uint64_t initialPomdpState = preprocessedPomdpPtr->getInitialStates().getNextSetIndex(0);
397 using ExtendedBeliefMDPType = storm::utility::ExtendedValueType<BeliefMDPType>;
398 std::optional<ExtendedBeliefMDPType> lowerBound =
399 storm::utility::fromSentinel(beliefExplorationBounds.preprocessingBounds->template getHighestLowerBound<BeliefMDPType>(initialPomdpState));
400 std::optional<ExtendedBeliefMDPType> upperBound =
401 storm::utility::fromSentinel(beliefExplorationBounds.preprocessingBounds->template getSmallestUpperBound<BeliefMDPType>(initialPomdpState));
402 STORM_LOG_INFO("Initial value bounds are [" << *lowerBound << ", " << *upperBound << "]");
403
404 auto updateLowerBound = [&lowerBound](ExtendedBeliefMDPType const& value) {
405 if (!lowerBound || value > *lowerBound) {
406 lowerBound = value;
407 }
408 };
409 auto updateUpperBound = [&upperBound](ExtendedBeliefMDPType const& value) {
410 if (!upperBound || value < *upperBound) {
411 upperBound = value;
412 }
413 };
414
415 storm::pomdp::beliefs::PropertyInformation propertyInfo;
416 if (rewardModelName) {
417 propertyInfo.kind = storm::pomdp::beliefs::PropertyInformation::Kind::ExpectedTotalReachabilityReward;
418 propertyInfo.rewardModelName = rewardModelName;
419 } else if (formulaInfo.isBounded()) {
420 propertyInfo.kind = storm::pomdp::beliefs::PropertyInformation::Kind::RewardBoundedReachabilityProbability;
421 // Collect reward bounds from bounded formula
422 auto boundedFormula = formula.asProbabilityOperatorFormula().getSubformula().asBoundedUntilFormula();
423 for (uint64_t i = 0; i < boundedFormula.getDimension(); ++i) {
424 const auto& tbRef = boundedFormula.getTimeBoundReference(i);
425 if (tbRef.isRewardBound()) {
426 propertyInfo.rewardBounds.push_back({.rewardModelName = tbRef.getOptionalRewardModelName().get_value_or(""),
427 .lowerBound = boundedFormula.getLowerBoundAsOptionalTimeBound(i),
428 .upperBound = boundedFormula.getUpperBoundAsOptionalTimeBound(i)});
429 }
430 }
431 } else {
432 propertyInfo.kind = storm::pomdp::beliefs::PropertyInformation::Kind::ReachabilityProbability;
433 }
434 propertyInfo.dir = formulaInfo.getOptimizationDirection();
435 propertyInfo.targetObservations = targetObservations;
436 STORM_LOG_THROW(!policyExportRequested || propertyInfo.kind != beliefs::PropertyInformation::Kind::RewardBoundedReachabilityProbability,
437 storm::exceptions::InvalidSettingsException, "Policy export is not supported for reward-aware belief exploration.");
438
439 storm::pomdp::beliefs::BeliefBasedModelChecker<storm::models::sparse::Pomdp<ValueType>, BeliefType, BeliefMDPType> checker(*preprocessedPomdpPtr);
440 ExtendedBeliefMDPType overResultValue;
441 ExtendedBeliefMDPType underResultValue;
442 bool isOverApproximation{false};
443 bool isUnderApproximation{false};
444 bool completedExploration{false};
445 if (pomdpSettings.isBeliefExplorationDiscretizeSet()) {
446 isOverApproximation = true;
447 STORM_PRINT_AND_LOG("Computing an over-approximation via belief MDP discretization...\n");
448 STORM_LOG_WARN_COND(belExplSettings.getSizeThresholdInit() == 0,
449 "The size threshold is ignored for discretization, which requires complete exploration.");
450 STORM_LOG_WARN_COND(belExplSettings.getExplorationTimeLimit() == 0,
451 "The exploration time threshold is ignored for discretization, which requires complete exploration.");
452 revisedOptions.maxExplorationSize.reset();
453 revisedOptions.maxExplorationTime.reset();
454
455 if (propertyInfo.kind == beliefs::PropertyInformation::Kind::RewardBoundedReachabilityProbability) {
456 std::vector<std::string> relevantRewardModelNames;
457 for (auto const& rewardBound : propertyInfo.rewardBounds) {
458 relevantRewardModelNames.push_back(rewardBound.rewardModelName);
459 }
460 auto checkResult =
461 checker.checkRewardAwareDiscretize(env, propertyInfo, revisedOptions, belExplSettings.getResolutionInit(),
462 belExplSettings.isDynamicTriangulationModeSet(), beliefExplorationBounds, relevantRewardModelNames);
463 overResultValue = checkResult.value;
464 printBeliefExplorationStatistics(checkResult.statistics);
465 } else {
466 auto checkResult = checker.checkDiscretize(env, propertyInfo, revisedOptions, belExplSettings.getResolutionInit(),
467 belExplSettings.isDynamicTriangulationModeSet(), beliefExplorationBounds);
468 overResultValue = checkResult.value;
469 printBeliefExplorationStatistics(checkResult.statistics);
470 }
471 }
472
473 if (pomdpSettings.isBeliefExplorationUnfoldSet()) {
474 STORM_PRINT_AND_LOG("Computing an under-approximation via belief MDP unfolding...\n");
475 revisedOptions.generatePolicy = policyExportRequested;
476 if (belExplSettings.getExplorationTimeLimit() != 0) {
477 revisedOptions.maxExplorationTime = belExplSettings.getExplorationTimeLimit();
478 }
479 if (belExplSettings.getSizeThresholdInit() == 0) {
480 revisedOptions.maxExplorationSize = preprocessedPomdpPtr->getNumberOfStates() * preprocessedPomdpPtr->getMaxNrStatesWithSameObservation();
481 STORM_PRINT_AND_LOG("Heuristically selected an under-approximation MDP size threshold of " << revisedOptions.maxExplorationSize.value() << ".\n");
482 } else {
483 revisedOptions.maxExplorationSize = belExplSettings.getSizeThresholdInit();
484 }
485 if (belExplSettings.isUseClippingSet()) {
486 revisedOptions.useClipping = true;
487 revisedOptions.clippingResolutions = std::vector<uint64_t>(preprocessedPomdpPtr->getNrObservations(), belExplSettings.getClippingGridResolution());
488 }
489 isUnderApproximation = true;
490 auto checkResult = [&]() -> typename storm::pomdp::beliefs::BeliefBasedModelCheckerResult<BeliefMDPType> {
491 if (propertyInfo.kind == beliefs::PropertyInformation::Kind::RewardBoundedReachabilityProbability) {
492 std::vector<std::string> relevantRewardModelNames;
493 for (auto const& rewardBound : propertyInfo.rewardBounds) {
494 relevantRewardModelNames.push_back(rewardBound.rewardModelName);
495 }
496 return checker.checkRewardAwareUnfold(env, propertyInfo, revisedOptions, beliefExplorationBounds, relevantRewardModelNames);
497 }
498 return checker.checkUnfold(env, propertyInfo, revisedOptions, beliefExplorationBounds);
499 }();
500 underResultValue = checkResult.value;
501 completedExploration = checkResult.completedExploration;
502 printBeliefExplorationStatistics(checkResult.statistics);
503 if (policyExportRequested) {
504 STORM_LOG_THROW(checkResult.policy.has_value(), storm::exceptions::UnexpectedException,
505 "Policy generation was requested, but belief-space unfolding did not produce a policy.");
506 auto const& filename = pomdpSettings.getExportPolicyFilename();
507 if (*policyExportFormat == PolicyExportFormat::Dot) {
508 storm::pomdp::api::exportPolicyAsDot(*checkResult.policy, filename);
509 } else {
510 storm::pomdp::api::exportPolicyAsJson(*checkResult.policy, filename);
511 }
512 STORM_PRINT_AND_LOG("Exported policy to '" << filename << "'.\n");
513 if (belExplSettings.isUseClippingSet()) {
514 STORM_LOG_WARN(
515 "The reported result includes the clipping approximation and may differ from the value achieved by the exported policy "
516 "on the original POMDP.");
517 }
518 }
519 isOverApproximation = (completedExploration && !belExplSettings.isUseClippingSet()) || isOverApproximation;
520 }
521
522 if (completedExploration && !belExplSettings.isUseClippingSet()) {
523 updateLowerBound(underResultValue);
524 updateUpperBound(underResultValue);
525 } else {
526 if (isOverApproximation) {
527 if (storm::solver::maximize(propertyInfo.dir)) {
528 updateUpperBound(overResultValue);
529 if (!isUnderApproximation) {
530 lowerBound.reset();
531 }
532 } else {
533 updateLowerBound(overResultValue);
534 if (!isUnderApproximation) {
535 upperBound.reset();
536 }
537 }
538 }
539 if (isUnderApproximation) {
540 if (storm::solver::maximize(propertyInfo.dir)) {
541 updateLowerBound(underResultValue);
542 if (!isOverApproximation) {
543 upperBound.reset();
544 }
545 } else {
546 updateUpperBound(underResultValue);
547 if (!isOverApproximation) {
548 lowerBound.reset();
549 }
550 }
551 }
552 }
553
554 if (storm::utility::resources::isTerminate()) {
555 STORM_PRINT_AND_LOG("\nResult till abort: ");
556 } else {
557 STORM_PRINT_AND_LOG("\nResult: ");
558 }
559 printResult(lowerBound, upperBound);
560 STORM_PRINT_AND_LOG('\n');
561 return true;

609bool performTransformation(std::shared_ptr<storm::models::sparse::Pomdp<ValueType>>& pomdp, storm::logic::Formula const& formula) {
610 auto const& pomdpSettings = storm::settings::getModule<storm::settings::modules::POMDPSettings>();
611 auto const& ioSettings = storm::settings::getModule<storm::settings::modules::IOSettings>();
612 auto const& transformSettings = storm::settings::getModule<storm::settings::modules::ToParametricSettings>();
613 bool transformationPerformed = false;
614 bool memoryUnfolded = false;
615 if (pomdpSettings.getMemoryBound() > 1) {
616 STORM_PRINT_AND_LOG("Computing the unfolding for memory bound " << pomdpSettings.getMemoryBound() << " and memory pattern '"
617 << storm::storage::toString(pomdpSettings.getMemoryPattern()) << "' ...");
618 storm::storage::PomdpMemory memory = storm::storage::PomdpMemoryBuilder().build(pomdpSettings.getMemoryPattern(), pomdpSettings.getMemoryBound());
619 std::cout << memory.toString() << '\n';
620 storm::transformer::PomdpMemoryUnfolder<ValueType> memoryUnfolder(*pomdp, memory);
621 pomdp = memoryUnfolder.transform();
622 STORM_PRINT_AND_LOG(" done.\n");
623 pomdp->printModelInformationToStream(std::cout);
624 transformationPerformed = true;
625 memoryUnfolded = true;
626 }
627
628 // From now on the POMDP is considered memoryless
629
630 if (transformSettings.isMecReductionSet()) {
631 STORM_PRINT_AND_LOG("Eliminating mec choices ...");
632 // Note: Elimination of mec choices only preserves memoryless schedulers.
633 uint64_t oldChoiceCount = pomdp->getNumberOfChoices();
634 storm::transformer::GlobalPomdpMecChoiceEliminator<ValueType> mecChoiceEliminator(*pomdp);
635 pomdp = mecChoiceEliminator.transform(formula);
636 STORM_PRINT_AND_LOG(" done.\n");
637 STORM_PRINT_AND_LOG(oldChoiceCount - pomdp->getNumberOfChoices() << " choices eliminated through MEC choice elimination.\n");
638 pomdp->printModelInformationToStream(std::cout);
639 transformationPerformed = true;
640 }
641
642 if (transformSettings.isTransformBinarySet() || transformSettings.isTransformSimpleSet()) {
643 if (transformSettings.isTransformSimpleSet()) {
644 STORM_PRINT_AND_LOG("Transforming the POMDP to a simple POMDP.");
645 pomdp = storm::transformer::BinaryPomdpTransformer<ValueType>().transform(*pomdp, true).transformedPomdp;
646 } else {
647 STORM_PRINT_AND_LOG("Transforming the POMDP to a binary POMDP.");
648 pomdp = storm::transformer::BinaryPomdpTransformer<ValueType>().transform(*pomdp, false).transformedPomdp;
649 }
650 pomdp->printModelInformationToStream(std::cout);
651 STORM_PRINT_AND_LOG(" done.\n");
652 transformationPerformed = true;
653 }
654
655 if (pomdpSettings.isExportToParametricSet()) {
656 STORM_PRINT_AND_LOG("Transforming memoryless POMDP to pMC...");
657 storm::transformer::ApplyFiniteSchedulerToPomdp<ValueType> toPMCTransformer(*pomdp);
658 std::string transformMode = transformSettings.getFscApplicationTypeString();
659 auto pmc = toPMCTransformer.transform(storm::transformer::parsePomdpFscApplicationMode(transformMode));
660 STORM_PRINT_AND_LOG(" done.\n");
661 if (transformSettings.allowPostSimplifications()) {
662 STORM_PRINT_AND_LOG("Simplifying pMC...");
663 pmc = storm::api::performBisimulationMinimization<storm::RationalFunction>(pmc->template as<storm::models::sparse::Dtmc<storm::RationalFunction>>(),
664 {formula.asSharedPointer()})
665 ->template as<storm::models::sparse::Dtmc<storm::RationalFunction>>();
666 STORM_PRINT_AND_LOG(" done.\n");
667 pmc->printModelInformationToStream(std::cout);
668 }
669 STORM_PRINT_AND_LOG("Exporting pMC...");
670 storm::pars::analysis::ConstraintCollector constraints(*pmc);
671 auto const& parameterSet = constraints.getVariables();
672 std::vector<storm::RationalFunctionVariable> parameters(parameterSet.begin(), parameterSet.end());
673 std::vector<std::string> parameterNames;
674 for (auto const& parameter : parameters) {
675 parameterNames.push_back(parameter.name());
676 }
677 storm::api::exportSparseModelAsDrn(pmc, pomdpSettings.getExportToParametricFilename(), parameterNames,
678 !ioSettings.isExplicitExportPlaceholdersDisabled());
679 STORM_PRINT_AND_LOG(" done.\n");
680 transformationPerformed = true;
681 }
682 if (transformationPerformed && !memoryUnfolded) {
683 STORM_PRINT_AND_LOG("Implicitly assumed restriction to memoryless schedulers for at least one transformation.\n");
684 }
685 return transformationPerformed;
686}

704template<typename ValueType>
705void processFormula(storm::Environment const& env, std::shared_ptr<storm::models::sparse::Pomdp<ValueType>>&& pomdp,
706 std::shared_ptr<storm::logic::Formula const> formula) {
707 if (formula->asOperatorFormula().getSubformula().isBoundedUntilFormula()) {
708 // TODO: allow for more general reward-bounded until properties than reachability
709
710 auto const& boundedUntilFormula = formula->asOperatorFormula().getSubformula().asBoundedUntilFormula();
711 for (uint64_t i = 0; i < boundedUntilFormula.getDimension(); ++i) {
712 STORM_LOG_THROW(boundedUntilFormula.getLeftSubformula(i).isTrueFormula(), storm::exceptions::NotSupportedException,
713 "Reward-bounded until properties for POMDPs currently require 'true' as the left-hand side, i.e., they must be reward-bounded "
714 "reachability properties.");
715 }
716 auto const& pomdpSettings = storm::settings::getModule<storm::settings::modules::POMDPSettings>();
717 // Process bounded until formulas
718 // If level widths are given, unfold the levels and make sure that the level rewards (and only those) are made observable
719 // If explicit unfolding is requested, unfold all reward bounds (if levels were unfolded before, this means a second round of unfolding)
720 // If reward observability is set and no level widths are given, make all rewards occurring in the formula observable. This also happens if no
721 // unfolding was requested.
722 storm::utility::Stopwatch boundedUntilProcessingWatch(true);
723 auto const levelWidths = pomdpSettings.getLevelWidthForBoundedReachability();
724 if (!levelWidths.empty()) {
725 STORM_PRINT_AND_LOG("Perform unfolding for observation levels.\n");
726 // Unfold the levels (includes dimensions with level-width 0)
727 typename transformer::RewardBoundUnfolder<ValueType>::UnfoldingOptions options;
728 options.levelWidths = levelWidths;
729 auto const unfoldingResult = transformer::RewardBoundUnfolder<ValueType>::transform(*pomdp, *formula, options);
730 pomdp = unfoldingResult.model->template as<storm::models::sparse::Pomdp<ValueType>>();
731 formula = unfoldingResult.formula;
732 // make sure the levels are observable
733 std::set<std::string> levelRewardModels;
734 formula->gatherReferencedRewardModels(levelRewardModels);
735 pomdp = storm::pomdp::transformer::ToStateBasedObservationTransformer<ValueType>::transformRewardAware(*pomdp, levelRewardModels);
736 }
737 std::set<std::string> rewardModelsToObserve;
738 if (pomdpSettings.isRewardObservableSet() && levelWidths.empty()) {
739 formula->gatherReferencedRewardModels(rewardModelsToObserve); // keep rewards to make them observable later
740 }
741 if (pomdpSettings.isBoundedToUnboundedReachabilityTransformationSet() && formula->asOperatorFormula().getSubformula().isBoundedUntilFormula()) {
742 STORM_PRINT_AND_LOG("Perform explicit unfolding of reward bounds.\n");
743 typename transformer::RewardBoundUnfolder<ValueType>::UnfoldingOptions options;
744 options.preservedRewardModels = rewardModelsToObserve;
745 auto const unfoldingResult = transformer::RewardBoundUnfolder<ValueType>::transform(*pomdp, *formula, options);
746 pomdp = unfoldingResult.model->template as<storm::models::sparse::Pomdp<ValueType>>();
747 formula = unfoldingResult.formula;
748 }
749 if (pomdpSettings.isRewardObservableSet() && levelWidths.empty()) {
750 STORM_PRINT_AND_LOG("Extend observation function to become reward aware.\n");
751 pomdp = storm::pomdp::transformer::ToStateBasedObservationTransformer<ValueType>::transformRewardAware(*pomdp, rewardModelsToObserve);
752 }
753 STORM_LOG_THROW(!levelWidths.empty() || pomdpSettings.isRewardObservableSet() || pomdpSettings.isBoundedToUnboundedReachabilityTransformationSet(),
754 storm::exceptions::InvalidSettingsException,
755 "No handling of bounded until formulas specified. Consider setting --unfold-reward-bound and/or --reward-aware.");
756 STORM_PRINT_AND_LOG("bounded reachability processing done. POMDP Information:\n");
757 pomdp->printModelInformationToStream(std::cout);
758 STORM_PRINT_AND_LOG("Transformed formula: " << *formula << "\n");
759 boundedUntilProcessingWatch.stop();
760 STORM_PRINT_AND_LOG("Time for pre-processing: " << boundedUntilProcessingWatch << ".\n");
761 }
762 auto formulaInfo = storm::pomdp::analysis::getFormulaInformation(*pomdp, *formula);
763 STORM_LOG_THROW(!formulaInfo.isUnsupported(), storm::exceptions::InvalidPropertyException,
764 "The formula '" << *formula << "' is not supported by storm-pomdp.");
765
766 storm::utility::Stopwatch sw(true);
767 // Note that formulaInfo contains state-based information which potentially needs to be updated during preprocessing
768 if (performPreprocessing(pomdp, formulaInfo, *formula)) {
769 sw.stop();
770 STORM_PRINT_AND_LOG("Time for graph-based POMDP (pre-)processing: " << sw << ".\n");
771 pomdp->printModelInformationToStream(std::cout);
772 }
773
774 sw.restart();
775 if (performTransformation(pomdp, *formula)) {
776 sw.stop();
777 STORM_PRINT_AND_LOG("Time for POMDP transformation(s): " << sw << ".\n");
778 }
779
780 sw.restart();
781 if (performAnalysis(env, pomdp, formulaInfo, *formula)) {
782 sw.stop();
783 STORM_PRINT_AND_LOG("Time for POMDP analysis: " << sw << ".\n");
784 }
785}
