90template<storm::OptimizationDirection Dir, bool Relative, storm::OptimizationDirection RobustDir, bool TrackDirection>
91SolverStatus ValueIterationHelper<ValueType, TrivialRowGrouping, SolutionType>::VI(std::vector<SolutionType>& operand, std::vector<ValueType> const& offsets,
92 uint64_t& numIterations, SolutionType const& precision,
93 std::function<SolverStatus(SolverStatus const&)> const& iterationCallback,
94 MultiplicationStyle mult,
95 storm::OptionalRef<SolutionBounds<SolutionType>> solutionBounds,
96 SolverGuarantee const& guarantee) const {
97 VIOperatorBackend<SolutionType, Dir, Relative, TrackDirection> backend{precision};
98 std::vector<SolutionType>* operand1{&operand};
99 std::vector<SolutionType>* operand2{&operand};
100 if (mult == MultiplicationStyle::Regular) {
101 operand2 = &viOperator->allocateAuxiliaryVector(operand.size());
102 }
103 bool resultInAuxVector{false};
104 SolverStatus status{SolverStatus::InProgress};
105 while (status == SolverStatus::InProgress) {
106 ++numIterations;
107 bool applyResult = viOperator->template applyRobust<RobustDir>(*operand1, *operand2, offsets, backend);
108 if (applyResult) {
109 status = SolverStatus::Converged;
110 } else if (iterationCallback) {
111 status = iterationCallback(status);
112 }
113 if (mult == MultiplicationStyle::Regular) {
114 std::swap(operand1, operand2);
115 resultInAuxVector = !resultInAuxVector;
116 }
117 }
118 if (mult == MultiplicationStyle::Regular) {
119 if (resultInAuxVector) {
120 STORM_LOG_ASSERT(&operand == operand2, "Unexpected operand address.");
121 std::swap(*operand1, *operand2);
122 }
123 viOperator->freeAuxiliaryVector();
124 }
125 if (!solutionBounds.has_value()) {
126 return status;
127 }
128 if constexpr (TrackDirection) {
129 if (mult == MultiplicationStyle::GaussSeidel) {
130 // An iteration that decreased nothing gives x <= T(x), so x lies below the unique fixpoint; dually for
131 // one that increased nothing.
132 if (backend.nonDecreasing()) {
133 solutionBounds->lower = operand;
134 }
135 if (backend.nonIncreasing()) {
136 solutionBounds->upper = operand;
137 }
138 }
139 } else if (guarantee == SolverGuarantee::LessOrEqual) {
140 // The iteration maintains a guarantee that held for the initial operand, so the result carries it too.
141 solutionBounds->lower = operand;
142 } else if (guarantee == SolverGuarantee::GreaterOrEqual) {
143 solutionBounds->upper = operand;
144 }
145 return status;
146}

148template<typename ValueType, bool TrivialRowGrouping, typename SolutionType>
149template<storm::OptimizationDirection Dir, bool Relative>
150SolverStatus ValueIterationHelper<ValueType, TrivialRowGrouping, SolutionType>::VI(std::vector<SolutionType>& operand, std::vector<ValueType> const& offsets,
151 uint64_t& numIterations, SolutionType const& precision,
152 const std::function<SolverStatus(const SolverStatus&)>& iterationCallback,
153 MultiplicationStyle mult,
154 UncertaintyResolutionMode const& uncertaintyResolutionMode,
155 storm::OptionalRef<SolutionBounds<SolutionType>> solutionBounds,
156 SolverGuarantee const& guarantee) const {
157 bool robustUncertainty = false;
158 if constexpr (storm::IsIntervalType<ValueType>) {
159 robustUncertainty = isUncertaintyResolvedRobust(uncertaintyResolutionMode, Dir);
160 }
161
162 // Reading the direction of every sweep costs in the innermost loop, so only do it where it is the sole route
163 // to a bound: where one is wanted and the operand carries no guarantee that the iteration already maintains.
164 bool const trackDirection = solutionBounds.has_value() && guarantee == SolverGuarantee::None;
165 if (robustUncertainty) {
166 if (trackDirection) {
167 return VI<Dir, Relative, invert(Dir), true>(operand, offsets, numIterations, precision, iterationCallback, mult, solutionBounds, guarantee);
168 }
169 return VI<Dir, Relative, invert(Dir), false>(operand, offsets, numIterations, precision, iterationCallback, mult, solutionBounds, guarantee);
170 } else {
171 if (trackDirection) {
172 return VI<Dir, Relative, Dir, true>(operand, offsets, numIterations, precision, iterationCallback, mult, solutionBounds, guarantee);
173 }
174 return VI<Dir, Relative, Dir, false>(operand, offsets, numIterations, precision, iterationCallback, mult, solutionBounds, guarantee);
175 }
176}

179SolverStatus ValueIterationHelper<ValueType, TrivialRowGrouping, SolutionType>::VI(
180 std::vector<SolutionType>& operand, std::vector<ValueType> const& offsets, uint64_t& numIterations, bool relative, SolutionType const& precision,
181 std::optional<storm::OptimizationDirection> const& dir, std::function<SolverStatus(SolverStatus const&)> const& iterationCallback, MultiplicationStyle mult,
182 UncertaintyResolutionMode const& uncertaintyResolutionMode, storm::OptionalRef<SolutionBounds<SolutionType>> solutionBounds,
183 SolverGuarantee const& guarantee) const {
184 if constexpr (storm::IsIntervalType<ValueType>) {
185 STORM_LOG_THROW(uncertaintyResolutionMode != UncertaintyResolutionMode::Unset, storm::exceptions::IllegalFunctionCallException,
186 "Uncertainty resolution mode must be set for uncertain (interval) models.");
187 STORM_LOG_THROW(dir.has_value() || (uncertaintyResolutionMode != UncertaintyResolutionMode::Robust &&
188 uncertaintyResolutionMode != UncertaintyResolutionMode::Cooperative),
189 storm::exceptions::IllegalFunctionCallException,
190 "Robust or cooperative nature resolution modes cannot be used if optimization direction is not set.");
191 }
192
193 STORM_LOG_ASSERT(TrivialRowGrouping || dir.has_value(), "No optimization direction given!");
194 if (!dir.has_value() || maximize(*dir)) {
195 if (relative) {
196 return VI<storm::OptimizationDirection::Maximize, true>(operand, offsets, numIterations, precision, iterationCallback, mult,
197 uncertaintyResolutionMode, solutionBounds, guarantee);
198 } else {
199 return VI<storm::OptimizationDirection::Maximize, false>(operand, offsets, numIterations, precision, iterationCallback, mult,
200 uncertaintyResolutionMode, solutionBounds, guarantee);
201 }
202 } else {
203 if (relative) {
204 return VI<storm::OptimizationDirection::Minimize, true>(operand, offsets, numIterations, precision, iterationCallback, mult,
205 uncertaintyResolutionMode, solutionBounds, guarantee);
206 } else {
207 return VI<storm::OptimizationDirection::Minimize, false>(operand, offsets, numIterations, precision, iterationCallback, mult,
208 uncertaintyResolutionMode, solutionBounds, guarantee);
209 }
210 }
