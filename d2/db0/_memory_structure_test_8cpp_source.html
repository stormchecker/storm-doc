11std::shared_ptr<storm::models::sparse::Mdp<double>> buildTinyMdp() {
12 // 3-state chain: 0 -> 1 -> 2 (self-loop)
13 // State 0 is initial, state 2 is a "goal" state.
14 uint64_t numStates = 3;
15 storm::storage::SparseMatrixBuilder<double> builder(numStates, numStates, numStates, true, true, numStates);
16 builder.newRowGroup(0);
17 builder.addNextValue(0, 1, 1.0);
18 builder.newRowGroup(1);
19 builder.addNextValue(1, 2, 1.0);
20 builder.newRowGroup(2);
21 builder.addNextValue(2, 2, 1.0);
22 auto matrix = builder.build();
23
24 storm::models::sparse::StateLabeling labeling(numStates);
25 labeling.addLabel("init");
26 labeling.addLabel("goal");
27 labeling.addLabelToState("init", 0);
28 labeling.addLabelToState("goal", 2);
29
30 return std::make_shared<storm::models::sparse::Mdp<double>>(std::move(matrix), std::move(labeling));
31}
