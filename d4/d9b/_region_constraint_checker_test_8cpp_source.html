51Region buildRegion(std::vector<std::pair<std::string, std::pair<double, double>>> const& bounds) {
52 Region::Valuation lowerBoundaries;
53 Region::Valuation upperBoundaries;
54 for (auto const& [variable, boundsOfVariable] : bounds) {
55 lowerBoundaries[variableByName(variable)] = coefficient(boundsOfVariable.first);
56 upperBoundaries[variableByName(variable)] = coefficient(boundsOfVariable.second);
57 }
58 return Region(lowerBoundaries, upperBoundaries);
59}

239TEST(RegionConstraintCheckerTest, RealModelConstraintsAreDecidedExactly) {
240#ifndef STORM_HAVE_SMT_SOLVER
241 GTEST_SKIP() << "No SMT solver available.";
242#endif
243 storm::clearRFVariablePool();
244
245 storm::prism::Program program = storm::api::parseProgram(STORM_TEST_RESOURCES_DIR "/pdtmc/parametric_die_2.pm");
246 program.checkValidity();
247 auto formulas = storm::api::extractFormulasFromProperties(storm::api::parsePropertiesForPrismProgram("P=? [F s=7]", program));
248 EXPECT_EQ(1ul, formulas.size());
249 storm::generator::NextStateGeneratorOptions options(*formulas.front());
250 auto model =
251 storm::builder::ExplicitModelBuilder<storm::RationalFunction>(program, options).build()->as<storm::models::sparse::Dtmc<storm::RationalFunction>>();
252
253 storm::pars::analysis::ConstraintCollector collector(*model);
254 EXPECT_FALSE(collector.getWellformedConstraints().empty());
255 EXPECT_FALSE(collector.getGraphPreservingConstraints().empty());
256 Checker checker(collector);
257
258 // The collected well-formedness constraints are p <= 1, q <= 1, p >= 0 and q >= 0, so the closed unit
259 // square satisfies all of them. All constraints are linear, hence they are decided rather than undecided.
260 auto unitSquare = buildRegion({{"p", {0.0, 1.0}}, {"q", {0.0, 1.0}}});
261 auto wellformed = checker.checkWellformedness(unitSquare);
262 EXPECT_EQ(storm::pars::analysis::ConstraintVerdict::Sat, wellformed.verdict);
263 EXPECT_TRUE(wellformed.undecidedConstraints.empty());
264 EXPECT_TRUE(wellformed.holdsThroughout());
265
266 // The graph-preserving constraints are p != 0, p != 1, q != 0 and q != 1. Because the region is closed and
267 // therefore contains its boundary, no rectangular region containing 0 or 1 can satisfy all of them. The
268 // checker reports the violation and provides a witness, rather than pretending the model is graph
269 // preserving.
270 auto graphPreserving = checker.checkGraphPreservation(unitSquare);
271 EXPECT_EQ(storm::pars::analysis::ConstraintVerdict::Unsat, graphPreserving.verdict);
272 EXPECT_FALSE(graphPreserving.holdsThroughout());
273 EXPECT_TRUE(graphPreserving.violatedConstraint.has_value());
274 EXPECT_TRUE(graphPreserving.witness.has_value());
275
276 // On a strictly interior region, both properties hold.
277 auto interior = buildRegion({{"p", {0.1, 0.9}}, {"q", {0.1, 0.9}}});
278 EXPECT_TRUE(checker.isWellDefined(interior));
279 EXPECT_TRUE(checker.isGraphPreserving(interior));
280
281 // A region outside [0, 1] satisfies p != 0, p != 1, q != 0 and q != 1, but is not well-formed. Hence it is not graph-preserving.
282 auto outside = buildRegion({{"p", {2.0, 3.0}}, {"q", {2.0, 3.0}}});
283 EXPECT_FALSE(checker.isWellDefined(outside));
284 EXPECT_TRUE(Checker::checkConstraints(collector.getGraphPreservingConstraints(), outside, "test").holdsThroughout());
285 auto outsideResult = checker.checkGraphPreservation(outside);
286 EXPECT_EQ(storm::pars::analysis::ConstraintVerdict::Unsat, outsideResult.verdict);
287 EXPECT_FALSE(checker.isGraphPreserving(outside));
288 EXPECT_TRUE(outsideResult.violatedConstraint.has_value());
289 EXPECT_TRUE(outsideResult.witness.has_value());
290}
