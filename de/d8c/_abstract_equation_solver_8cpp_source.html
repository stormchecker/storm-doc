357template<typename SolutionType>
358void AbstractEquationSolver<SolutionType>::reportStatus(SolverStatus status, boost::optional<uint64_t> const& iterations) const {
359 if (iterations) {
360 switch (status) {
361 case SolverStatus::Converged:
362 STORM_LOG_TRACE("Iterative solver converged after " << iterations.get() << " iterations.");
363 break;
364 case SolverStatus::TerminatedEarly:
365 STORM_LOG_TRACE("Iterative solver terminated early after " << iterations.get() << " iterations.");
366 break;
367 case SolverStatus::MaximalIterationsExceeded:
368 STORM_LOG_WARN("Iterative solver did not converge after " << iterations.get() << " iterations.");
369 break;
370 case SolverStatus::Aborted:
371 STORM_LOG_WARN("Iterative solver was aborted after " << iterations.get() << " iterations.");
372 break;
373 default:
374 STORM_LOG_THROW(false, storm::exceptions::InvalidStateException, "Iterative solver terminated unexpectedly.");
375 }
376 } else {
377 switch (status) {
378 case SolverStatus::Converged:
379 STORM_LOG_TRACE("Solver converged.");
380 break;
381 case SolverStatus::TerminatedEarly:
382 STORM_LOG_TRACE("Solver terminated early.");
383 break;
384 case SolverStatus::MaximalIterationsExceeded:
385 STORM_LOG_ASSERT(false, "Non-iterative solver should not exceed maximal number of iterations.");
386 STORM_LOG_WARN("Solver did not converge.");
387 break;
388 case SolverStatus::Aborted:
389 STORM_LOG_WARN("Solver was aborted.");
390 break;
391 default:
392 STORM_LOG_THROW(false, storm::exceptions::InvalidStateException, "Solver terminated unexpectedly.");
393 }
394 }

397template<typename SolutionType>
398SolverStatus AbstractEquationSolver<SolutionType>::updateStatus(SolverStatus status, bool earlyTermination, uint64_t iterations,
399 uint64_t maximalNumberOfIterations) const {
400 if (status != SolverStatus::Converged) {
401 if (earlyTermination) {
402 status = SolverStatus::TerminatedEarly;
403 } else if (iterations >= maximalNumberOfIterations) {
404 status = SolverStatus::MaximalIterationsExceeded;
405 } else if (storm::utility::resources::isTerminate()) {
406 status = SolverStatus::Aborted;
407 }
408 }
409 return status;
410}
