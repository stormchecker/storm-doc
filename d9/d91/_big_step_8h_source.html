133 template<typename ConstantType>
134 ConstantType evaluate(ConstantType input) const {
135 ConstantType sumOfTerms = utility::zero<ConstantType>();
136 for (auto const& [info, constant] : *this) {
137 ConstantType outerMult = utility::one<ConstantType>();
138 for (uint64_t i = 0; i < info.size(); i++) {
139 auto polynomial = this->polynomialCache->at(parameter).second[i];
140 // Evaluate the inner polynomial by its coefficients
141 auto coefficients = polynomial.coefficients();
142 ConstantType innerSum = utility::zero<ConstantType>();
143 for (uint64_t exponent = 0; exponent < coefficients.size(); exponent++) {
144 if (exponent != 0) {
145 innerSum += storm::utility::pow(input, exponent) * storm::utility::convertNumber<ConstantType>(coefficients[exponent]);
146 } else {
147 innerSum += utility::convertNumber<ConstantType>(coefficients[exponent]);
148 }
149 }
150 // Inner polynomial ^ exponent
151 outerMult *= storm::utility::pow(innerSum, info[i]);
152 }
153 sumOfTerms += outerMult * utility::convertNumber<ConstantType>(constant);
154 }
155 return sumOfTerms;
156 }
