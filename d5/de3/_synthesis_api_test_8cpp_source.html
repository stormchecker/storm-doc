33 " [] s=0 -> " +
34 probability + " : (s'=1) + (1-(" + probability + ")) : (s'=2);\n" +
35 (mdp ? " [] s=0 -> (s'=1);\n [] s=0 -> (s'=2);\n" : "") +
36 " [] s=1 -> (s'=1);\n [] s=2 -> (s'=2);\nendmodule\nlabel \"target\" = s=1;\n" + rewards,
37 "SynthesisApiTest");

39 storm::builder::BuilderOptions options({formula("P=? [F \"target\"]")}, program);
