54template<typename ModelType, typename ValueType = typename ModelType::ValueType>
55std::shared_ptr<ModelType> buildModel(storm::storage::SparseMatrix<ValueType> matrix, std::map<std::string, std::vector<uint64_t>> const& labels,
56 std::vector<ValueType> const& stateRewards = {}, std::vector<ValueType> const& stateActionRewards = {}) {
57 uint64_t const numStates = matrix.getColumnCount();
58 storm::models::sparse::StateLabeling labeling(numStates);
59 labeling.addLabel("init");
60 labeling.addLabelToState("init", 0);
61 for (auto const& [label, states] : labels) {
62 labeling.addLabel(label);
63 for (uint64_t const state : states) {
64 labeling.addLabelToState(label, state);
65 }
66 }
67 storm::storage::sparse::ModelComponents<ValueType> components(std::move(matrix), std::move(labeling));
68 if constexpr (std::is_same_v<ModelType, storm::models::sparse::Ctmc<ValueType>>) {
69 components.rateTransitions = true;
70 }
71 if (!stateRewards.empty() || !stateActionRewards.empty()) {
72 components.rewardModels.emplace("rew", storm::models::sparse::StandardRewardModel<ValueType>(
73 stateRewards.empty() ? std::nullopt : std::optional<std::vector<ValueType>>(stateRewards),
74 stateActionRewards.empty() ? std::nullopt : std::optional<std::vector<ValueType>>(stateActionRewards)));
75 }
76 return std::make_shared<ModelType>(std::move(components));
77}

83template<typename ValueType = double>
84std::shared_ptr<storm::models::sparse::MarkovAutomaton<ValueType>> buildMarkovAutomaton(storm::storage::SparseMatrix<ValueType> matrix,
85 storm::storage::BitVector markovianStates,
86 std::map<std::string, std::vector<uint64_t>> const& labels) {
87 uint64_t const numStates = matrix.getColumnCount();
88 storm::models::sparse::StateLabeling labeling(numStates);
89 labeling.addLabel("init");
90 labeling.addLabelToState("init", 0);
91 for (auto const& [label, states] : labels) {
92 labeling.addLabel(label);
93 for (uint64_t const state : states) {
94 labeling.addLabelToState(label, state);
95 }
96 }
97 storm::storage::sparse::ModelComponents<ValueType> components(std::move(matrix), std::move(labeling));
98 components.markovianStates = std::move(markovianStates);
99 components.rateTransitions = true; // The rows of the Markovian states hold rates, which the model turns into probabilities plus exit rates.
100 return std::make_shared<storm::models::sparse::MarkovAutomaton<ValueType>>(std::move(components));
101}

142template<typename ValueType>
143PrismInput<ValueType> buildFromPrism(std::string const& prismFile, std::string const& formulaString, BuildOptions const& buildOptions = {}) {
144 // Preprocessing substitutes the constants. This is also where the declared type of a constant is taken into account, e.g. `const double x = 1/500;` only
145 // becomes 0.002 instead of the integer division 0 after preprocessing.
146 auto const program = storm::parser::PrismParser::parse(prismFile, true).preprocess();
147 PrismInput<ValueType> result;
148 result.formulas = storm::api::extractFormulasFromProperties(storm::api::parsePropertiesForPrismProgram(formulaString, program));
149 auto builderOptions = buildOptions.allLabels ? storm::builder::BuilderOptions(false, true) : storm::builder::BuilderOptions(result.formulas, program);
150 if (buildOptions.choiceLabels) {
151 builderOptions.setBuildChoiceLabels();
152 }
153 if (buildOptions.choiceOrigins) {
154 builderOptions.setBuildChoiceOrigins();
155 }
156 if (buildOptions.stateValuations) {
157 builderOptions.setBuildStateValuations();
158 }
159 result.model = storm::api::buildSparseModel<ValueType>(program, builderOptions);
160 return result;
161}
