71 std::string result;
72 result.reserve(label.size());
73 for (char character : label) {
74 switch (character) {

81 case '\n':

232void ObservationBasedFiniteStateController<ValueType>::addActionOutputUpdate(uint64_t const originId, uint64_t const observationId,
233 std::unique_ptr<FSCOutputUpdate> update) {
234 auto& outputsForNode = transitions[originId];
235 if (auto const existingUpdate = outputsForNode.find(observationId);
236 existingUpdate != outputsForNode.end() && existingUpdate->second->randomisedActionOutput()) {
237 --numberOfRandomisedOutputs;
238 }
239 if (update->randomisedActionOutput()) {
240 ++numberOfRandomisedOutputs;
241 }
242 outputsForNode[observationId] = std::move(update);
243 isDeterministicPolicy = numberOfRandomisedOutputs == 0;
244}

372 }

383void ObservationBasedFiniteStateController<ValueType>::writeJsonToStream(std::ostream& outStream) const {
384 storm::json<ValueType> output;
385 output["initial-node"] = initialNodeId;
386 output["deterministic"] = isDeterministicPolicy;
387 output["nodes"] = storm::json<ValueType>::array();
388
389 std::vector<uint64_t> nodeIds;
390 nodeIds.push_back(initialNodeId);
391 for (auto const& [originId, actionOutputMap] : transitions) {
392 nodeIds.push_back(originId);
393 for (auto const& [observationId, outputUpdatePtr] : actionOutputMap) {
394 nodeIds.push_back(outputUpdatePtr->nextMemoryNode);
395 }
396 }
397 std::ranges::sort(nodeIds);
398 nodeIds.erase(std::unique(nodeIds.begin(), nodeIds.end()), nodeIds.end());
399
400 for (uint64_t nodeId : nodeIds) {
401 storm::json<ValueType> node;
402 node["id"] = nodeId;
403 node["transitions"] = storm::json<ValueType>::array();
404 if (transitions.contains(nodeId)) {
405 std::vector<uint64_t> observationIds;
406 observationIds.reserve(transitions.at(nodeId).size());
407 for (auto const& [observationId, outputUpdatePtr] : transitions.at(nodeId)) {
408 observationIds.push_back(observationId);
409 }
410 std::ranges::sort(observationIds);
411
412 for (uint64_t observationId : observationIds) {
413 auto const& update = *transitions.at(nodeId).at(observationId);
414 storm::json<ValueType> transition;
415 transition["observation"]["id"] = observationId;
416 if (idToObservationSignature && idToObservationSignature->contains(observationId)) {
417 auto& signatureJson = transition["observation"]["signature"];
418 signatureJson = storm::json<ValueType>::array();
419 for (auto const& [observable, value] : idToObservationSignature->at(observationId).values) {
420 storm::json<ValueType> observableJson;
421 observableJson["name"] = observable;
422 std::visit(
423 [&observableJson](auto const& typedValue) {
424 using TypedValue = std::decay_t<decltype(typedValue)>;
425 if constexpr (std::is_same_v<TypedValue, std::monostate>) {
426 observableJson["type"] = "unset";
427 observableJson["value"] = nullptr;
428 } else if constexpr (std::is_same_v<TypedValue, storm::RationalNumber>) {
429 observableJson["type"] = "rational";
430 observableJson["value"] = storm::utility::to_string(typedValue);
431 } else if constexpr (std::is_same_v<TypedValue, bool>) {
432 observableJson["type"] = "boolean";
433 observableJson["value"] = typedValue;
434 } else if constexpr (std::is_same_v<TypedValue, uint64_t>) {
435 observableJson["type"] = "unsigned-integer";
436 observableJson["value"] = typedValue;
437 } else if constexpr (std::is_same_v<TypedValue, int64_t>) {
438 observableJson["type"] = "integer";
439 observableJson["value"] = typedValue;
440 } else if constexpr (std::is_same_v<TypedValue, double>) {
441 observableJson["type"] = "double";

443 } else {

447 }

449 value);

451 }
452 }
453 transition["successor"] = update.nextMemoryNode;
454 transition["actions"] = storm::json<ValueType>::array();

460 actions.emplace_back(actionId, probability);

462 } else {
463 auto const& deterministicUpdate = dynamic_cast<DeterministicActionUpdate const&>(update);
464 actions.emplace_back(deterministicUpdate.action, storm::utility::one<ValueType>());
465 }
466 std::ranges::sort(actions, {}, &std::pair<uint64_t, ValueType>::first);
467
468 for (auto const& [actionId, probability] : actions) {
469 storm::json<ValueType> action;
470 action["id"] = actionId;
471 if (idToActionName && idToActionName->contains(observationId) && idToActionName->at(observationId).contains(actionId)) {
472 action["name"] = idToActionName->at(observationId).at(actionId);
473 }
474 action["probability"] = probability;
475 transition["actions"].push_back(std::move(action));
476 }
477 node["transitions"].push_back(std::move(transition));
478 }
479 }
480 output["nodes"].push_back(std::move(node));
481 }
482 outStream << storm::dumpJson(output) << '\n';
