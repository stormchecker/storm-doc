230 template<typename Order, typename Condition>
231 requires std::invocable<Order, ElementIndex, ElementIndex> && std::invocable<Condition, ElementIndex, ElementIndex>
232 bool splitBlockByOrder(Block const& block, Order const& less, Condition const& splitCondition) {
233 if (block.size() <= 1) {
234 return false; // nothing to do
235 }
236
237 // Sort the contents of the block first
238 STORM_LOG_ASSERT(!isProperSuperBlock(block), "Tried to split a block that consists of multiple sub-blocks.");
239 auto const blockStart = getBlockIndex(block);
240
241 auto const blockEnd = blockStart + block.size();
242 std::sort(blockContents.begin() + blockStart, blockContents.begin() + blockEnd, less);
243
244 // update the inverse after sorting
245 for (ElementIndex i = blockStart; i < blockEnd; ++i) {
246 blockContentsInverse[blockContents[i]] = i;
247 }
248
249 // Catch the special case where there is no split
250 if (!splitCondition(blockContents[blockStart], blockContents[blockEnd - 1])) {
251 return false; // nothing to do
252 }
253
254 // helper function to find the end index of a current block
255 auto getEndOfBlock = [this, splitCondition, blockEnd](BlockIndex const currIndex) {
256 for (auto i = currIndex + 1; i < blockEnd; ++i) {
257 if (splitCondition(blockContents[currIndex], blockContents[i])) {
258 return i;
259 }
260 }
261 return blockEnd;
262 };
263
264 // Now create new blocks whenever the order changes
265 for (auto newBlockIndex = blockStart; newBlockIndex < blockEnd;) {
266 auto const newBlockEnd = getEndOfBlock(newBlockIndex);
267 registerNewBlock(newBlockIndex, newBlockEnd);
268 newBlockIndex = newBlockEnd;
269 ++numBlocks;
270 }
271 --numBlocks; // the old superblock is no longer a block as counted by this partition
272 STORM_LOG_ASSERT(isProperSuperBlock(block), "Partition in inconsistent state: Block was not split into multiple sub-blocks.");
273 STORM_LOG_ASSERT(checkBlockValidity(block), "Partition in inconsistent state: Block is not valid.");
274 return true; // there must have been a split because the case without a split is already caught above
275 }

286 template<typename Condition>
287 requires std::invocable<Condition, ElementIndex, ElementIndex>
288 bool splitBlockByClustering(Block const& block, Condition const& condition) {
289 STORM_LOG_ASSERT(!isProperSuperBlock(block), "Tried to split a block that consists of multiple sub-blocks.");
290 if (block.size() <= 1) {
291 return false; // nothing to do
292 }
293
294 auto const blockStart = getBlockIndex(block);
295 auto const blockEnd = blockStart + block.size();
296
297 // Swaps the elements in [start, blockEnd) so that those compatible (!condition) with the anchor at start directly follow it.
298 // @return mid such that all elements in [start, mid) are compatible with the anchor (located at start) and those at [mid, blockEnd) are not.
299 auto const partitionBlock = [this, &condition, &blockEnd](BlockIndex const start) {
300 // The anchor itself stays in place, so we only partition the elements after it.
301 auto const anchor = blockContents[start];
302 return partitionBlockContents(start + 1, blockEnd, [&condition, &anchor](ElementIndex const element) { return condition(anchor, element); });
303 };
304
305 for (uint64_t start = blockStart; start < blockEnd;) {
306 auto const currEnd = partitionBlock(start);
307 // Catch the special case where there is no split
308 if (start == blockStart && currEnd == blockEnd) {
309 return false; // nothing to do
310 }
311 registerNewBlock(start, currEnd);
312 start = currEnd;
313 ++numBlocks;
314 }
315 --numBlocks; // the old superblock is no longer a block as counted by this partition
316 STORM_LOG_ASSERT(isProperSuperBlock(block), "Partition in inconsistent state: Block was not split into multiple sub-blocks.");
317 STORM_LOG_ASSERT(checkBlockValidity(block), "Partition in inconsistent state: Block is not valid.");
318 return true; // there must have been a split because the case without a split is already caught above
319 }

327 template<typename SplittingPredicate>
328 requires std::predicate<SplittingPredicate, ElementIndex>
329 std::pair<Block, Block> splitBlockByPredicate(Block const& block, SplittingPredicate const& f) {
330 STORM_LOG_ASSERT(!isProperSuperBlock(block), "Tried to split a block that consists of multiple sub-blocks.");
331 STORM_LOG_ASSERT(!block.empty(), "Tried to split an empty block");
332
333 // swap the block contents so that all elements e with f(e) == false come first
334 auto const blockStart = getBlockIndex(block);
335 auto const blockEnd = blockStart + block.size();
336 auto const l = partitionBlockContents(blockStart, blockEnd, f);
337
338 // Handle cases where there is no split
339 if (l == blockStart) {
340 return {Block{}, block}; // all elements are true
341 } else if (l == blockEnd) {
342 return {block, Block{}}; // all elements are false
343 }
344
345 // Perform a split
346 Block noBlock = registerNewBlock(blockStart, l);
347 Block yesBlock = registerNewBlock(l, blockEnd);
348 ++numBlocks; // We have split a single block into two blocks, so the total number increments by 1
349 return {noBlock, yesBlock};
350 }

359 template<typename SplitRange>
360 requires std::ranges::input_range<std::remove_reference_t<SplitRange>> &&
361 std::same_as<std::ranges::range_value_t<std::remove_reference_t<SplitRange>>, ElementIndex>
362 std::pair<Block, Block> splitBlockByRange(Block const& block, SplitRange&& r) {
363 STORM_LOG_ASSERT(!isProperSuperBlock(block), "Tried to split a block that consists of multiple sub-blocks.");
364 STORM_LOG_ASSERT(!block.empty(), "Tried to split an empty block");
365
366 auto const blockStart = getBlockIndex(block);
367 auto const blockEnd = blockStart + block.size();
368
369 // We split the block into a "no"-part and a "yes"-part
370
371 auto yesBlockStart = blockEnd;
372 for (ElementIndex const element : r) {
373 auto const src = blockContentsInverse[element];
374 if (src < blockStart || src >= yesBlockStart) {
375 continue; // element is either not in the block or already occurred before (duplicate in r)
376 }
377 // swap the element to the end of the block, into the 'yes' part
378 --yesBlockStart;
379 std::swap(blockContents[src], blockContents[yesBlockStart]);
380 blockContentsInverse[blockContents[src]] = src;
381 blockContentsInverse[blockContents[yesBlockStart]] = yesBlockStart;
382 }
383
384 // Handle cases where there is no split
385 if (yesBlockStart == blockStart) {
386 return {Block{}, block}; // all elements are in the range
387 } else if (yesBlockStart == blockEnd) {
388 return {block, Block{}}; // no elements are in the range
389 }
390
391 // Perform a split
392 Block noBlock = registerNewBlock(blockStart, yesBlockStart);
393 Block yesBlock = registerNewBlock(yesBlockStart, blockEnd);
394 ++numBlocks; // We have split a single block into two blocks, so the total number increments by 1
395 return {noBlock, yesBlock};
396 }

409 template<typename SplittingPredicate>
410 requires std::predicate<SplittingPredicate, ElementIndex>
411 BlockIndex partitionBlockContents(BlockIndex const start, BlockIndex const end, SplittingPredicate const& f) {
412 STORM_LOG_ASSERT(start <= end, "Tried to partition an invalid range.");
413 if (start == end) {
414 return start; // empty range, nothing to swap
415 }
416 auto l = start;
417 auto r = end - 1;
418 // Loop invariant: all elements at position < l are false, all elements at position > r are true
419 while (l <= r) {
420 while (l <= r && !f(std::as_const(blockContents[l]))) {
421 ++l;
422 }
423 if (l > r) {
424 break; // no more elements to swap
425 }
426 // At this point we know that f(blockContents[l]) == true
427 while (l < r && f(std::as_const(blockContents[r]))) {
428 --r;
429 }
430 if (l == r) {
431 // We have f(blockContents[r]) == f(blockContents[l]) == true
432 --r;
433 break; // l > r holds now
434 }
435 std::swap(blockContents[l], blockContents[r]);
436 blockContentsInverse[blockContents[l]] = l;
437 blockContentsInverse[blockContents[r]] = r;
438 ++l;
439 --r;
440 }
441 STORM_LOG_ASSERT(l == r + 1, "Unexpected indices");
442 return l;
443 }
