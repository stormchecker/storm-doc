27template<typename ValueType>
28storm::utility::ExtendedValueType<ValueType> checkInitialState(storm::Environment const& env, std::string const& file, std::string const& formulaString) {
29 auto program = storm::api::parseProgram(file);
30 auto formulas = storm::api::extractFormulasFromProperties(storm::api::parsePropertiesForPrismProgram(formulaString, program));
31 auto model = storm::api::buildSparseModel<ValueType>(program, formulas);
32 auto task = storm::api::createTask<ValueType>(formulas.front(), true);
33 auto result = storm::api::verifyWithSparseEngine<ValueType>(env, model, task);
34 return result->template asExplicitQuantitativeCheckResult<ValueType>()[*model->getInitialStates().begin()];
35}

53TEST(EnvironmentFactoryTest, SoundEnvironmentSelectsSoundMethods) {

56
57 EXPECT_TRUE(solver.isForceSoundness());

59
60 // Defaults which are already sound remain untouched
61 EXPECT_EQ(storm::solver::EquationSolverType::Topological, solver.getLinearEquationSolverType());
62 EXPECT_EQ(storm::solver::MinMaxMethod::Topological, solver.minMax().getMethod());
63
64 EXPECT_EQ(storm::solver::EquationSolverType::Native, solver.topological().getUnderlyingEquationSolverType());
65 EXPECT_EQ(storm::solver::MinMaxMethod::OptimisticValueIteration, solver.topological().getUnderlyingMinMaxMethod());
66 EXPECT_EQ(storm::solver::NativeLinearEquationSolverMethod::OptimisticValueIteration, solver.native().getMethod());
67 EXPECT_EQ(storm::solver::EigenLinearEquationSolverMethod::SparseLU, solver.eigen().getMethod());
68 EXPECT_EQ(storm::solver::GameMethod::PolicyIteration, solver.game().getMethod());
69 EXPECT_EQ(storm::solver::LraMethod::ValueIteration, solver.lra().getDetLraMethod());
70 EXPECT_EQ(storm::solver::LraMethod::ValueIteration, solver.lra().getNondetLraMethod());
71 EXPECT_EQ(storm::SteadyStateDistributionAlgorithm::ExpectedVisitingTimes, env.modelchecker().getSteadyStateDistributionAlgorithm());
72
73 // The choices count as defaults, such that e.g. exact mode can still override them
74 EXPECT_TRUE(solver.isLinearEquationSolverTypeSetFromDefaultValue());
75 EXPECT_TRUE(solver.topological().isUnderlyingEquationSolverTypeSetFromDefault());
76 EXPECT_TRUE(solver.topological().isUnderlyingMinMaxMethodSetFromDefault());
77 EXPECT_TRUE(solver.native().isMethodSetFromDefault());

80 EXPECT_TRUE(solver.game().isMethodSetFromDefault());
