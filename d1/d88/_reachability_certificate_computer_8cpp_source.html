124 void initializeValueVectors(Algorithm const& alg) {
125 // Initialize value vector and set values for terminal states.
126 uint64_t const numberOfStates = totalRewardSpecification.terminalStates.size();
127 globalLowerValues.assign(numberOfStates, storm::utility::zero<ValueType>());
128 for (auto const& [value, stateSet] : totalRewardSpecification.terminalStateValues) {
129 if (value.isFinite()) {
130 storm::utility::vector::setVectorValues(globalLowerValues, stateSet, value.getFiniteValue());
131 } else {
132 STORM_LOG_ASSERT(value.isPositiveInfinity(), "Unexpected terminal state value.");
133 // Infinity is not representable in ValueType (in particular not for rational numbers).
134 // The values of these states are never used in any computation, because choices leading to such states are either cut away or do not exist.
135 // The actual infinite values are set by the caller.
136 }
137 }
138 // We assume that 0 is a valid lower bound for all states.
139 STORM_LOG_THROW(!totalRewardSpecification.actionRewards.has_value() ||
140 std::all_of(totalRewardSpecification.actionRewards->begin(), totalRewardSpecification.actionRewards->end(),
141 [](auto const& v) { return v >= storm::utility::zero<ValueType>(); }),
142 storm::exceptions::NotSupportedException, "Certificates for negative rewards are not supported.");
143
144 // Initialize second vector for upper bounds (if needed by the selected algorithm)
145 if (alg.type == AlgorithmType::FpII || alg.type == AlgorithmType::FpSmoothII || alg.type == AlgorithmType::FpRoundII) {
146 globalUpperValues = globalLowerValues;
147 } else {
148 STORM_LOG_ASSERT(alg.type == AlgorithmType::ExPI, "Unsupported algorithm.");
149 globalUpperValues.clear(); // Do not use upper values for this.
150 }
151 }

299 template<typename VT, bool SingleValue>
300 SubsystemData<VT, SingleValue> eliminateECs(SubsystemData<VT, SingleValue>&& original) const {
301 storm::storage::BitVector allStates(original.transitions.getRowGroupCount(), true);
302 auto possibleECChoices = ~original.exitChoices;
303 if (totalRewardSpecification.actionRewards) {
304 // Do not consider choices with rewards
305 auto const candidateChoices = possibleECChoices;
306 for (auto const choice : candidateChoices) {
307 if constexpr (SingleValue) {
308 if (!storm::utility::isZero(original.offsets[choice])) {
309 possibleECChoices.set(choice, false);
310 }
311 } else {
312 STORM_LOG_ASSERT(storm::utility::isZero(original.offsets.first[choice]) == storm::utility::isZero(original.offsets.second[choice]),
313 "Unexpected offsets in non-exiting choice.");
314 if (!storm::utility::isZero(original.offsets.first[choice])) {
315 possibleECChoices.set(choice, false);
316 }
317 }
318 }
319 }
320 storm::storage::MaximalEndComponentDecomposition<VT> ecs(original.transitions, original.transitions.transpose(true), allStates, possibleECChoices);
321 if (ecs.empty()) {
322 // No ECs to eliminate
323 return std::move(original);
324 }
325 auto reductionRes = storm::transformer::EndComponentEliminator<VT>::transform(original.transitions, ecs, allStates, ~allStates, false);
326 SubsystemData<VT, SingleValue> reduced{
327 std::move(reductionRes.matrix), std::move(reductionRes.sinkRows), {}, {}, {}, std::move(reductionRes.oldToNewStateMapping)};
328 if constexpr (SingleValue) {
329 reduced.offsets.reserve(reduced.exitChoices.size());
330 } else {
331 reduced.offsets.first.reserve(reduced.exitChoices.size());
332 reduced.offsets.second.reserve(reduced.exitChoices.size());
333 }
334 if (original.exitProbabilities.has_value()) {
335 reduced.exitProbabilities.emplace();
336 reduced.exitProbabilities->reserve(reduced.exitChoices.size());
337 }
338 for (uint64_t newRow = 0; newRow < reduced.exitChoices.size(); ++newRow) {
339 auto const oldRow = reductionRes.newToOldRowMapping[newRow];
340 if (original.exitProbabilities.has_value()) {
341 // Set exit probability to '1' for all sinkRows, i.e., those that correspond to staying in the EC forever
342 reduced.exitProbabilities->push_back(reduced.exitChoices.get(newRow) ? storm::utility::one<VT>() : original.exitProbabilities->at(oldRow));
343 }
344 if (original.exitChoices.get(oldRow)) {
345 reduced.exitChoices.set(newRow, true);
346 }
347 if constexpr (SingleValue) {
348 reduced.offsets.push_back(original.offsets[oldRow]);
349 } else {
350 reduced.offsets.first.push_back(original.offsets.first[oldRow]);
351 reduced.offsets.second.push_back(original.offsets.second[oldRow]);
352 }
353 }
354 STORM_LOG_ASSERT(std::all_of(reduced.originalToReducedStateMapping->begin(), reduced.originalToReducedStateMapping->end(),
355 [&reduced](auto const& i) { return i < reduced.transitions.getRowGroupCount(); }),
356 "No representative for some state found in the reduced subsystem.");
357 return reduced;
358 }

405 void computeForSingleStateSubsystem(uint64_t state) {
406 storm::utility::Extremum<Dir, ValueType> lowerValue, upperValue;
407 bool const computeUpperValues = !globalUpperValues.empty();
408 // If we minimize, choices that lead to states with infinite value can be ignored
409 // (If we maximize, there are no such choices since the optimal value would be infinite as well.)
410 auto const& infinityStates = totalRewardSpecification.terminalStateValues.front().second;
411 bool const ignoreChoicesToInfinity = storm::solver::minimize(Dir) && totalRewardSpecification.terminalStateValues.front().first.isPositiveInfinity();
412 for (auto const rowIndex : transitionProbabilityMatrix.getRowGroupIndices(state)) {
413 if (ignoreChoicesToInfinity) {
414 auto const row = transitionProbabilityMatrix.getRow(rowIndex);
415 if (std::any_of(row.begin(), row.end(), [&infinityStates](auto const& entry) { return infinityStates.get(entry.getColumn()); })) {
416 continue;
417 }
418 }
419 auto const reward = totalRewardSpecification.actionRewards.has_value() ? totalRewardSpecification.actionRewards.value()[rowIndex]
420 : storm::utility::zero<ValueType>();
421 auto lowerRowValue = reward;
422 auto upperRowValue = reward;
423 std::optional<ValueType> diagonalEntry;
424 for (auto const& entry : transitionProbabilityMatrix.getRow(rowIndex)) {
425 if (state == entry.getColumn()) {
426 STORM_LOG_ASSERT(!diagonalEntry, "Already have diag entry.");
427 diagonalEntry = entry.getValue();
428 } else {
429 lowerRowValue += entry.getValue() * globalLowerValues[entry.getColumn()];
430 if (computeUpperValues) {
431 upperRowValue += entry.getValue() * globalUpperValues[entry.getColumn()];
432 }
433 }
434 }
435 if (diagonalEntry.has_value()) {
436 if (diagonalEntry.value() == storm::utility::one<ValueType>()) {
437 // This choice is a self-loop and can therefore never be optimal
438 continue;
439 }
440 lowerRowValue /= storm::utility::one<ValueType>() - diagonalEntry.value();
441 if (computeUpperValues) {
442 upperRowValue /= storm::utility::one<ValueType>() - diagonalEntry.value();
443 }
444 }
445 lowerValue &= lowerRowValue;
446 if (computeUpperValues) {
447 upperValue &= upperRowValue;
448 }
449 }
450 STORM_LOG_ASSERT(!lowerValue.empty() && !(computeUpperValues && upperValue.empty()),
451 "Failed to compute lower and/or upper value for state " << state << ".");
452 globalLowerValues[state] = *lowerValue;
453 if (computeUpperValues) {
454 globalUpperValues[state] = *upperValue;
455 }
456 }

490 void computeForSubsystemExPI(storm::storage::BitVector const& subsystemStates) {
491 using VT = std::conditional_t<storm::NumberTraits<ValueType>::IsExact, ValueType, storm::RationalNumber>; // VT is exact
492
493 auto subsystemData = initializeSubsystemData<VT, true>(subsystemStates, AlgorithmType::ExPI);
494
495 storm::Environment env;
496 env.solver().minMax().setMethod(storm::solver::MinMaxMethod::ViToPi);
497 env.solver().setLinearEquationSolverType(storm::solver::EquationSolverType::Eigen);
498 env.solver().eigen().setMethod(solver::EigenLinearEquationSolverMethod::SparseLU);
499 env.solver().setForceExact(true);
500
501 if constexpr (Nondeterministic) {
502 auto solverFactory = storm::solver::GeneralMinMaxLinearEquationSolverFactory<VT>();
503 auto req = solverFactory.getRequirements(env, true, false, Dir, false, false);
504 std::vector<uint64_t> initSched;
505 if (req.validInitialScheduler()) {
506 storm::storage::SparseMatrix<VT> const& submatrix = subsystemData.transitions;
507 auto exitStates = submatrix.getRowGroupFilter(subsystemData.exitChoices, false);
508 storm::storage::Scheduler<VT> validScheduler(exitStates.size());
509 storm::utility::graph::computeSchedulerProb1E(storm::storage::BitVector(exitStates.size(), true), submatrix, submatrix.transpose(true),
510 ~exitStates, exitStates, validScheduler);
511
512 // Extract the relevant parts of the scheduler for the solver.
513 initSched.reserve(exitStates.size());
514 for (uint64_t i = 0; i < exitStates.size(); ++i) {
515 if (exitStates.get(i)) {
516 initSched.push_back(validScheduler.getChoice(i).getDeterministicChoice());
517 } else {
518 auto const groupStart = submatrix.getRowGroupIndices()[i];
519 auto globalChoice = subsystemData.exitChoices.getNextSetIndex(groupStart);
520 STORM_LOG_ASSERT(globalChoice < submatrix.getRowGroupIndices()[i + 1], "No valid choice for exit state " << i << ".");
521 initSched.push_back(globalChoice - groupStart);
522 }
523 }
524 req.clearValidInitialScheduler();
525 }
526 STORM_LOG_THROW(!req.hasEnabledCriticalRequirement(), storm::exceptions::UnmetRequirementException,
527 "Unmet requirements for MinMax linear equation solver: " << req.getEnabledRequirementsAsString() << ".");
528 auto solver = solverFactory.create(env, std::move(subsystemData.transitions));
529 solver->setHasUniqueSolution(true); // TODO: Has no end components?
530 if (!initSched.empty()) {
531 solver->setInitialScheduler(std::move(initSched));
532 }
533 solver->setRequirementsChecked(true);
534 solver->solveEquations(env, Dir, subsystemData.operands, subsystemData.offsets);
535 } else {
536 auto solverFactory = storm::solver::GeneralLinearEquationSolverFactory<VT>();
537 if (solverFactory.getEquationProblemFormat(env) == storm::solver::LinearEquationSolverProblemFormat::EquationSystem) {
538 subsystemData.transitions = storm::storage::SparseMatrix<VT>(subsystemData.transitions, true); // insert diag entries
539 subsystemData.transitions.convertToEquationSystem(); // Note: Invalidates uses of this matrix as 'transitions'
540 }
541 auto req = solverFactory.getRequirements(env);

543 "Unmet requirements for linear equation solver: " << req.getEnabledRequirementsAsString() << ".");
544 auto solver = solverFactory.create(env, std::move(subsystemData.transitions));
545 solver->solveEquations(env, subsystemData.operands, subsystemData.offsets);
546 }
547 setGlobalValuesFromSubsystem<VT>(subsystemStates, subsystemData, false);

565 void computeTopological(Algorithm const& alg, bool relative, ValueType const& precision, storm::storage::BitVector const& nonTerminalStates) {
566 STORM_LOG_TRACE("Creating SCC decomposition.");
567 storm::utility::Stopwatch sccSw(true);
568 storm::storage::StronglyConnectedComponentDecomposition<ValueType> const sccDecomposition(
569 transitionProbabilityMatrix,
570 storm::storage::StronglyConnectedComponentDecompositionOptions().forceTopologicalSort().computeSccDepths().subsystem(nonTerminalStates));
571 uint64_t const longestSccChainSize = sccDecomposition.getMaxSccDepth() + 1;
572 sccSw.stop();
573 STORM_LOG_INFO("SCC decomposition computed in "
574 << sccSw << ". Found " << sccDecomposition.size() << " SCC(s) containing a total of " << nonTerminalStates.getNumberOfSetBits()
575 << " states. Average SCC size is "
576 << static_cast<double>(transitionProbabilityMatrix.getRowGroupCount()) / static_cast<double>(sccDecomposition.size())
577 << ". Longest SCC chain size is " << longestSccChainSize << ".");
578 if (sccDecomposition.size() == 1) {
579 // Handle the case where there is just one large scc
580 computeForSubsystem(alg, relative, precision, nonTerminalStates);
581 return;
582 }
583 // At this point, solve each SCC individually
584 storm::storage::BitVector sccAsBitvector(transitionProbabilityMatrix.getRowGroupCount(), false); // allocate memory outside of loop
585 for (uint64_t sccIndex = 0; sccIndex < sccDecomposition.size(); ++sccIndex) {
586 auto const& scc = sccDecomposition.getBlock(sccIndex);
587 if (scc.size() == 1) {
588 // Directly solve single-state SCCs
589 computeForSingleStateSubsystem(*scc.begin());
590 continue;
591 }
592 sccAsBitvector.set(scc.begin(), scc.end());
593 ValueType const sccPrecision =
594 precision / storm::utility::convertNumber<ValueType, uint64_t>(longestSccChainSize - sccDecomposition.getSccDepth(sccIndex));
595 computeForSubsystem(alg, relative, sccPrecision, sccAsBitvector);
596 sccAsBitvector.clear();
597 }
598 }

607template<storm::OptimizationDirection Dir, typename ValueType, typename VectorValueType>
608storm::storage::BitVector computeInductiveChoices(storm::storage::SparseMatrix<ValueType> const& transitionProbabilityMatrix,
609 std::vector<VectorValueType> const& valueVector, storm::storage::BitVector const& relevantStates,
610 storm::OptionalRef<std::vector<ValueType> const> rewardVector = {}) {
611 storm::storage::BitVector inductiveChoices(transitionProbabilityMatrix.getRowCount(), false);
612 bool warnIfEmpty = true;
613 for (auto state : relevantStates) {
614 bool stateHasInductiveChoice = false;
615 for (auto choice : transitionProbabilityMatrix.getRowGroupIndices(state)) {
616 VectorValueType const choiceValue = detail::multiplyRowWithVectorAddNumber(
617 transitionProbabilityMatrix, choice, valueVector, rewardVector.has_value() ? rewardVector->at(choice) : storm::utility::zero<ValueType>());
618 if (storm::solver::maximize(Dir) ? valueVector[state] <= choiceValue : choiceValue <= valueVector[state]) {
619 inductiveChoices.set(choice);
620 stateHasInductiveChoice = true;
621 }
622 }
623 if (!stateHasInductiveChoice) {
624 STORM_LOG_WARN_COND(!warnIfEmpty, "State " << state << " has no inductive choice. The certificate might not be valid");
625 warnIfEmpty = false;
626 }
627 }
628 return inductiveChoices;
629}

631template<typename ValueType, storm::OptimizationDirection Dir>
632std::vector<RankingType> computeDistanceRanking(storm::storage::SparseMatrix<ValueType> const& transitionProbabilityMatrix,
633 storm::storage::SparseMatrix<ValueType> const& backwardTransitions,
634 storm::storage::BitVector const& targetStates,
635 storm::OptionalRef<storm::storage::BitVector const> constraintStates,
636 std::optional<storm::storage::BitVector> const& choiceConstraint = std::nullopt) {
637 auto const& rowGroupIndices = transitionProbabilityMatrix.getRowGroupIndices();
638
639 std::vector<RankingType> ranks(targetStates.size(), InfRank);
640 storm::utility::vector::setVectorValues<RankingType>(ranks, targetStates, 0);
641 std::vector<uint64_t> stateQueue(targetStates.begin(), targetStates.end());
642 while (!stateQueue.empty()) {
643 auto const state = stateQueue.back();
644 stateQueue.pop_back();
645 for (auto const& preEntry : backwardTransitions.getRow(state)) {
646 auto const preState = preEntry.getColumn();
647 if (ranks[preState] != InfRank) {
648 continue;
649 }
650 if (constraintStates.has_value() && !constraintStates->get(preState)) {
651 continue;
652 }
653 storm::utility::Extremum<Dir, RankingType> newRankMinusOne;
654 // Iterate over all (selected) choices of the pre-state
655 uint64_t currentChoice = rowGroupIndices[preState];
656 if (choiceConstraint.has_value()) {
657 currentChoice = choiceConstraint->getNextSetIndex(currentChoice);
658 }
659 auto nextChoice = [&choiceConstraint](auto c) { return choiceConstraint.has_value() ? choiceConstraint->getNextSetIndex(c + 1) : c + 1; };
660 for (auto const endChoice = rowGroupIndices[preState + 1]; currentChoice < endChoice; currentChoice = nextChoice(currentChoice)) {
661 // Compute current rank for this choice
662 storm::utility::Minimum<RankingType> choiceRank;
663 for (auto const& entry : transitionProbabilityMatrix.getRow(currentChoice)) {
664 if (storm::utility::isZero(entry.getValue())) {
665 continue;
666 }
667 choiceRank &= ranks[entry.getColumn()];
668 }
669 STORM_LOG_ASSERT(!choiceRank.empty(),
670 "Ranking operator failed to compute rank for state " << state << " since the row " << currentChoice << " is empty.");
671 newRankMinusOne &= *choiceRank;
672 }
673 if (!newRankMinusOne.empty() && *newRankMinusOne != InfRank) {
674 ranks[preState] = *newRankMinusOne + 1;
675 stateQueue.push_back(preState);
676 }
677 }
678 }
679 return ranks;
680}

682template<typename ValueType, storm::OptimizationDirection Dir>
683std::vector<RankingType> computeModifiedDistanceRanking(storm::storage::SparseMatrix<ValueType> const& transitionProbabilityMatrix,
684 storm::storage::SparseMatrix<ValueType> const& backwardTransitions,
685 storm::storage::BitVector const& probLess1States) {
686 std::vector<RankingType> ranks(probLess1States.size(), InfRank);
687 storm::utility::vector::setVectorValues<RankingType>(ranks, probLess1States, 0);
688
689 // Apply the modified ranking operator until no more states are updated
690 while (true) {
691 bool updated = false;
692 for (auto state : probLess1States) {
693 storm::utility::Extremum<Dir, RankingType> newStateRank;
694 for (auto const currentChoice : transitionProbabilityMatrix.getRowGroupIndices(state)) {
695 storm::utility::Minimum<RankingType> minSuccessorRank;
696 bool allSuccessorsEqual = true;
697 for (auto const& entry : transitionProbabilityMatrix.getRow(currentChoice)) {
698 if (storm::utility::isZero(entry.getValue())) {
699 continue;
700 }
701 if (!minSuccessorRank.empty() && ranks[entry.getColumn()] != *minSuccessorRank) {
702 allSuccessorsEqual = false;
703 }
704 minSuccessorRank &= ranks[entry.getColumn()];
705 }
706 STORM_LOG_ASSERT(!minSuccessorRank.empty(),
707 "Ranking operator failed to compute rank for state " << state << " since the row " << currentChoice << " is empty.");
708 auto choiceRank = *minSuccessorRank;
709 if (choiceRank != InfRank && !allSuccessorsEqual) {
710 ++choiceRank;
711 }
712 newStateRank &= choiceRank;
713 }
714 STORM_LOG_ASSERT(!newStateRank.empty(), "Failed to compute rank for state " << state << ".");
715 if (ranks[state] != *newStateRank) {
716 STORM_LOG_ASSERT(ranks[state] <= *newStateRank, "Ranking operator did not generate an increasing sequence for state " << state << ".");
717 ranks[state] = *newStateRank;
718 updated = true;
719 }
720 }
721 if (!updated) {
722 return ranks;
723 }
724 }
