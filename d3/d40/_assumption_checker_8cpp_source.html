22template<typename ValueType, typename ConstantType>
23void AssumptionChecker<ValueType, ConstantType>::initializeCheckingOnSamples(Environment const& env, std::shared_ptr<logic::Formula const> formula,
24 std::shared_ptr<models::sparse::Dtmc<ValueType>> model,
25 storage::ParameterRegion<ValueType> region, uint_fast64_t numberOfSamples) {
26 // Create sample points
27 auto instantiator = utility::ModelInstantiator<models::sparse::Dtmc<ValueType>, models::sparse::Dtmc<ConstantType>>(*model);
28 auto matrix = model->getTransitionMatrix();
29 std::set<VariableType> variables = models::sparse::getProbabilityParameters(*model);
30
31 for (uint_fast64_t i = 0; i < numberOfSamples; ++i) {
32 auto valuation = utility::parametric::Valuation<ValueType>();
33 for (auto var : variables) {
34 auto lb = region.getLowerBoundary(var.name());
35 auto ub = region.getUpperBoundary(var.name());
36 // Creates samples between lb and ub, that is: lb, lb + (ub-lb)/(#samples -1), lb + 2* (ub-lb)/(#samples -1), ..., ub
37 auto val = std::pair<VariableType, CoefficientType>(var, (lb + utility::convertNumber<CoefficientType>(i / (numberOfSamples - 1)) * (ub - lb)));
38 valuation.insert(val);
39 }
40 models::sparse::Dtmc<ConstantType> sampleModel = instantiator.instantiate(valuation);
41 auto checker = modelchecker::SparseDtmcPrctlModelChecker<models::sparse::Dtmc<ConstantType>>(sampleModel);
42 std::unique_ptr<modelchecker::CheckResult> checkResult;
43 if (formula->isProbabilityOperatorFormula() && formula->asProbabilityOperatorFormula().getSubformula().isUntilFormula()) {
44 const modelchecker::CheckTask<logic::UntilFormula, ConstantType> checkTask =
45 modelchecker::CheckTask<logic::UntilFormula, ConstantType>(formula->asProbabilityOperatorFormula().getSubformula().asUntilFormula());
46 checkResult = checker.computeUntilProbabilities(env, checkTask);
47 } else if (formula->isProbabilityOperatorFormula() && formula->asProbabilityOperatorFormula().getSubformula().isEventuallyFormula()) {
48 const modelchecker::CheckTask<logic::EventuallyFormula, ConstantType> checkTask =
49 modelchecker::CheckTask<logic::EventuallyFormula, ConstantType>(formula->asProbabilityOperatorFormula().getSubformula().asEventuallyFormula());
50 checkResult = checker.computeReachabilityProbabilities(env, checkTask);
51 } else {
52 STORM_LOG_THROW_UNCONDITIONALLY(exceptions::NotSupportedException, "Expecting until or eventually formula");
53 }
54 auto const& quantitativeResult = checkResult->asExplicitQuantitativeCheckResult<ConstantType>();
55 std::vector<ConstantType> values = quantitativeResult.getFiniteValueVector();
56 samples.push_back(values);
57 }
58 useSamples = true;
59}

73template<typename ValueType, typename ConstantType>
74AssumptionStatus AssumptionChecker<ValueType, ConstantType>::validateAssumption(Assumption const& assumption, std::shared_ptr<Order> order,
75 storage::ParameterRegion<ValueType> region,
76 std::vector<ConstantType> const minValues,
77 std::vector<ConstantType> const maxValues) const {
78 // First check if based on sample points the assumption can be discharged
79 AssumptionStatus result = AssumptionStatus::UNKNOWN;
80 if (useSamples) {
81 result = checkOnSamples(assumption);
82 }
83 STORM_LOG_ASSERT(result != AssumptionStatus::VALID, "Sample-based checking should never conclude VALID by itself.");
84
85 if (minValues.size() != 0 && maxValues.size() == minValues.size()) {
86 auto const val1 = assumption.state1;
87 auto const val2 = assumption.state2;
88 if (assumption.relation == AssumptionRelation::Greater) {
89 if (minValues[val1] > maxValues[val2]) {
90 return AssumptionStatus::VALID;
91 } else if (minValues[val1] == maxValues[val2] && minValues[val1] == maxValues[val1] && minValues[val2] == maxValues[val2]) {
92 return AssumptionStatus::INVALID;
93 } else if (minValues[val2] > maxValues[val1]) {
94 return AssumptionStatus::INVALID;
95 }
96 } else {
97 if (minValues[val1] == maxValues[val2] && minValues[val1] == maxValues[val1] && minValues[val2] == maxValues[val2]) {
98 return AssumptionStatus::VALID;
99 } else if (minValues[val1] > maxValues[val2]) {
100 return AssumptionStatus::INVALID;
101 } else if (minValues[val2] > maxValues[val1]) {
102 return AssumptionStatus::INVALID;
103 }
104 }
105 }
106
107 if (result == AssumptionStatus::UNKNOWN) {
108 // If result from sample checking was unknown, the assumption might hold
109 result = validateAssumptionSMTSolver(assumption, order, region, minValues, maxValues);
110 }
111 return result;
112}

114template<typename ValueType, typename ConstantType>
115AssumptionStatus AssumptionChecker<ValueType, ConstantType>::checkOnSamples(Assumption const& assumption) const {
116 auto result = AssumptionStatus::UNKNOWN;
117 for (auto const& values : samples) {
118 bool holds = assumption.relation == AssumptionRelation::Greater ? values[assumption.state1] > values[assumption.state2]
119 : values[assumption.state1] == values[assumption.state2];
120 if (!holds) {
121 result = AssumptionStatus::INVALID;
122 break;
123 }
124 }
125 return result;
126}

128template<typename ValueType, typename ConstantType>
129AssumptionStatus AssumptionChecker<ValueType, ConstantType>::validateAssumptionSMTSolver(Assumption const& assumption, std::shared_ptr<Order> order,
130 storage::ParameterRegion<ValueType> region,
131 std::vector<ConstantType> const minValues,
132 std::vector<ConstantType> const maxValues) const {
133 std::shared_ptr<expressions::ExpressionManager> manager(new expressions::ExpressionManager());
134 AssumptionStatus result = AssumptionStatus::UNKNOWN;
135 uint_fast64_t val1 = assumption.state1;
136 uint_fast64_t val2 = assumption.state2;
137 auto row1 = matrix.getRow(val1);
138 auto row2 = matrix.getRow(val2);
139
140 bool orderKnown = true;
141 // if the state with number val1 (val2) occurs in the successors of the state with number val2 (val1) we need to add val1 == expr1 (val2 == expr2) to the
142 // bounds
143 bool addVar1 = false;
144 bool addVar2 = false;
145 // Check if the order between the different successors is known
146 // Also start creating expression for order of states
147 auto exprOrderSucc = manager->boolean(true);
148 std::set<expressions::Variable> stateVariables;
149 std::set<expressions::Variable> topVariables;
150 std::set<expressions::Variable> bottomVariables;
151 for (auto itr1 = row1.begin(); orderKnown && itr1 != row1.end(); ++itr1) {
152 addVar2 |= itr1->getColumn() == val2;
153 auto varname1 = "s" + std::to_string(itr1->getColumn());
154 if (!manager->hasVariable(varname1)) {
155 if (order->isTopState(itr1->getColumn())) {
156 topVariables.insert(manager->declareRationalVariable(varname1));
157 } else if (order->isBottomState(itr1->getColumn())) {
158 bottomVariables.insert(manager->declareRationalVariable(varname1));
159 } else {
160 stateVariables.insert(manager->declareRationalVariable(varname1));
161 }
162 }
163
164 for (auto itr2 = row2.begin(); orderKnown && itr2 != row2.end(); ++itr2) {
165 addVar1 |= itr2->getColumn() == val1;
166 if (itr1->getColumn() != itr2->getColumn()) {
167 auto varname2 = "s" + std::to_string(itr2->getColumn());
168 if (!manager->hasVariable(varname2)) {
169 if (order->isTopState(itr2->getColumn())) {
170 topVariables.insert(manager->declareRationalVariable(varname2));
171 } else if (order->isBottomState(itr2->getColumn())) {
172 bottomVariables.insert(manager->declareRationalVariable(varname2));
173 } else {
174 stateVariables.insert(manager->declareRationalVariable(varname2));
175 }
176 }
177 auto comp = order->compare(itr1->getColumn(), itr2->getColumn());
178 if (minValues.size() > 0 && maxValues.size() == minValues.size() && comp == Order::NodeComparison::UNKNOWN) {
179 // Couldn't add relation between varname1 and varname2 but maybe we can based on min/max values;
180 if (minValues[itr2->getColumn()] > maxValues[itr1->getColumn()]) {
181 if (!order->contains(itr1->getColumn())) {
182 order->add(itr1->getColumn());
183 order->addStateToHandle(itr1->getColumn());
184 }
185 if (!order->contains(itr2->getColumn())) {
186 order->add(itr2->getColumn());
187 order->addStateToHandle(itr2->getColumn());
188 }
189 order->addRelation(itr2->getColumn(), itr1->getColumn());
190 comp = Order::NodeComparison::BELOW;
191 } else if (minValues[itr1->getColumn()] > maxValues[itr2->getColumn()]) {
192 if (!order->contains(itr1->getColumn())) {
193 order->add(itr1->getColumn());
194 order->addStateToHandle(itr1->getColumn());
195 }
196 if (!order->contains(itr2->getColumn())) {
197 order->add(itr2->getColumn());
198 order->addStateToHandle(itr2->getColumn());
199 }
200 order->addRelation(itr1->getColumn(), itr2->getColumn());
201 comp = Order::NodeComparison::ABOVE;
202 }
203 }
204 if (comp == Order::NodeComparison::ABOVE) {
205 exprOrderSucc = exprOrderSucc && !(manager->getVariable(varname1) <= manager->getVariable(varname2));
206 } else if (comp == Order::NodeComparison::BELOW) {
207 exprOrderSucc = exprOrderSucc && !(manager->getVariable(varname1) >= manager->getVariable(varname2));
208 } else if (comp == Order::NodeComparison::SAME) {
209 exprOrderSucc = exprOrderSucc && (manager->getVariable(varname1) >= manager->getVariable(varname2)) &&
210 (manager->getVariable(varname1) <= manager->getVariable(varname2));
211 } else {
212 orderKnown = false;
213 }
214 }
215 }
216 }
217
218 if (orderKnown) {
219 solver::Z3SmtSolver s(*manager);
220 auto valueTypeToExpression = expressions::RationalFunctionToExpression<ValueType>(manager);
221 expressions::Expression expr1 = manager->rational(0);
222 for (auto itr1 = row1.begin(); itr1 != row1.end(); ++itr1) {
223 expr1 = expr1 + (valueTypeToExpression.toExpression(itr1->getValue()) * manager->getVariable("s" + std::to_string(itr1->getColumn())));
224 }
225
226 expressions::Expression expr2 = manager->rational(0);
227 for (auto itr2 = row2.begin(); itr2 != row2.end(); ++itr2) {
228 expr2 = expr2 + (valueTypeToExpression.toExpression(itr2->getValue()) * manager->getVariable("s" + std::to_string(itr2->getColumn())));
229 }
230
231 // Create expression for the assumption based on the relation to successors
232 // It is the negation of actual assumption
233
234 expressions::Expression exprToCheck;
235 if (assumption.relation == AssumptionRelation::Greater) {
236 exprToCheck = expr1 <= expr2;
237 } else {
238 exprToCheck = expr1 != expr2;
239 }
240
241 auto variables = manager->getVariables();
242 // Bounds for the state probabilities and parameters
243 expressions::Expression exprBounds = manager->boolean(true);
244 if (addVar1) {
245 exprBounds = exprBounds && (manager->getVariable("s" + std::to_string(val1)) == expr1);
246 }
247 if (addVar2) {
248 exprBounds = exprBounds && (manager->getVariable("s" + std::to_string(val2)) == expr2);
249 }
250 for (auto var : variables) {
251 if (find(stateVariables.begin(), stateVariables.end(), var) != stateVariables.end()) {
252 // the var is a state
253 if (minValues.size() > 0 && maxValues.size() == minValues.size()) {
254 std::string test = var.getName();
255 auto val = std::stoi(test.substr(1, test.size() - 1));
256 exprBounds = exprBounds && manager->rational(minValues[val]) <= var && var <= manager->rational(maxValues[val]);
257 } else {
258 exprBounds = exprBounds && manager->rational(0) <= var && var <= manager->rational(1);
259 }
260 } else if (find(topVariables.begin(), topVariables.end(), var) != topVariables.end()) {
261 // the var is =)
262 exprBounds = exprBounds && var == manager->rational(1);
263 } else if (find(bottomVariables.begin(), bottomVariables.end(), var) != bottomVariables.end()) {
264 // the var is =(
265 exprBounds = exprBounds && var == manager->rational(0);
266 } else {
267 // the var is a parameter
268 auto lb = utility::convertNumber<RationalNumber>(region.getLowerBoundary(var.getName()));
269 auto ub = utility::convertNumber<RationalNumber>(region.getUpperBoundary(var.getName()));
270 exprBounds = exprBounds && manager->rational(lb) < var && var < manager->rational(ub);
271 }
272 }
273
274 s.add(exprOrderSucc);
275 s.add(exprBounds);
276 s.setTimeout(100);
277 // assert that sorting of successors in the order and the bounds on the expression are at least satisfiable
278 // when this is not the case, the order is invalid
279 // however, it could be that the sat solver didn't finish in time, in that case we just continue.
280 if (s.check() == solver::SmtSolver::CheckResult::Unsat) {
281 return AssumptionStatus::INVALID;
282 }
283
284 s.add(exprToCheck);
285 auto smtRes = s.check();
286 if (smtRes == solver::SmtSolver::CheckResult::Unsat) {
287 // If there is no thing satisfying the negation we are safe.
288 result = AssumptionStatus::VALID;
289 } else if (smtRes == solver::SmtSolver::CheckResult::Sat) {
290 result = AssumptionStatus::INVALID;
291 }
292 }
293 return result;
294}
