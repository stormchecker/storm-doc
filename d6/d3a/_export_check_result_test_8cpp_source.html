29// From state 1 the target is never reached, so the expected reward is infinite there and in state 0, which reaches
30// state 1 with probability 1/2. States 2 and 3 have a finite expected reward of 1 and 0, respectively.
31//
32// Every state is initial: a program that pins the initial state down instead has its initial states enumerated by an
33// SMT solver, which this test would then need a Storm built with one to run at all. Which states are initial makes no
34// difference to what is checked below, since the check result and its export cover all of them either way.
35std::string const modelDescription = R"(
36dtmc
37
38module main
39 s : [0..3];
40 [] s=0 -> 1/2 : (s'=1) + 1/2 : (s'=2);
41 [] s=1 -> 1 : (s'=1);
42 [] s=2 -> 1 : (s'=3);
43 [] s=3 -> 1 : (s'=3);
44endmodule
45
46init
47 true
48endinit
49
50label "target" = s=3;
51
52rewards "steps"
53 s<3 : 1;
54endrewards
55)";

74template<typename ValueType>
75void runExportTest(storm::Environment const& env) {
76 storm::prism::Program program = storm::parser::PrismParser::parseFromString(modelDescription, "testfile");
77 auto formulas = storm::api::extractFormulasFromProperties(storm::api::parsePropertiesForPrismProgram("R{\"steps\"}=? [F \"target\"]", program));
78 ASSERT_EQ(1ull, formulas.size());
79 auto model = storm::api::buildSparseModel<ValueType>(program, formulas)->template as<storm::models::sparse::Dtmc<ValueType>>();
80 ASSERT_EQ(4ull, model->getNumberOfStates());
81
82 storm::modelchecker::SparseDtmcPrctlModelChecker<storm::models::sparse::Dtmc<ValueType>> checker(*model);
83 std::unique_ptr<storm::modelchecker::CheckResult> result = checker.check(env, *formulas[0]);
84 ASSERT_TRUE(result->isExplicitQuantitativeCheckResult());
85 auto const& values = result->template asExplicitQuantitativeCheckResult<ValueType>().getValueVector();
86 ASSERT_EQ(4ull, values.size());
87
88 // Two states cannot reach the target, so their expected reward is infinite. Without those, the export below would
89 // not exercise anything.
90 uint64_t numberOfInfiniteValues = 0;
91 for (auto const& value : values) {
92 if (storm::utility::isInfinity(value)) {
93 ++numberOfInfiniteValues;
94 }
95 }
96 ASSERT_EQ(2ull, numberOfInfiniteValues);
97
98 auto const path = getTemporaryFilePath();
99 ASSERT_NO_THROW(storm::api::exportCheckResultToJson<ValueType>(model, result, path.string()));
100 std::ifstream exportedStream(path);
101 ASSERT_TRUE(exportedStream.good());
102 storm::json<double> exported;
103 ASSERT_NO_THROW(exportedStream >> exported);
104 exportedStream.close();
105 std::filesystem::remove(path);
106
107 ASSERT_TRUE(exported.is_array());
108 ASSERT_EQ(values.size(), exported.size());
109 for (auto const& entry : exported) {
110 ASSERT_TRUE(entry.count("s") == 1 && entry.count("v") == 1);
111 uint64_t const state = entry.at("s").template get<uint64_t>();
112 ASSERT_LT(state, values.size());
113 if (storm::utility::isInfinity(values[state])) {
114 // JSON has no infinity of its own, so an infinite value is exported as this string.
115 ASSERT_TRUE(entry.at("v").is_string()) << "State " << state << " has an infinite value, but was exported as " << entry.at("v").dump() << ".";
116 EXPECT_EQ("inf", entry.at("v").template get<std::string>());
117 } else {
118 ASSERT_TRUE(entry.at("v").is_number()) << "State " << state << " has the finite value " << values[state] << ", but was exported as "
119 << entry.at("v").dump() << ".";
120 EXPECT_EQ(storm::utility::convertNumber<double>(storm::utility::getFinite(values[state])), entry.at("v").template get<double>());
121 }
122 }
123}

130template<typename ValueType>
131void runNegativeInfinityExportTest() {
132 typedef storm::utility::ExtendedValueType<ValueType> ExtendedValueType;
133 std::vector<ExtendedValueType> values{storm::utility::one<ExtendedValueType>(), storm::utility::negativeInfinity<ValueType>(),
134 storm::utility::positiveInfinity<ValueType>()};
135 storm::modelchecker::ExplicitQuantitativeCheckResult<ValueType> result(std::move(values));
136
137 storm::json<ValueType> exported;
138 ASSERT_NO_THROW(exported = result.toJson());
139 ASSERT_TRUE(exported.is_array());
140 ASSERT_EQ(3ull, exported.size());
141
142 EXPECT_TRUE(exported[0].at("v").is_number());
143 // JSON has no infinity of its own, and the two of them must not be exported as the same thing.
144 ASSERT_TRUE(exported[1].at("v").is_string()) << "The negative infinity was exported as " << exported[1].at("v").dump() << ".";
145 EXPECT_EQ("-inf", exported[1].at("v").template get<std::string>());
146 ASSERT_TRUE(exported[2].at("v").is_string()) << "The positive infinity was exported as " << exported[2].at("v").dump() << ".";
147 EXPECT_EQ("inf", exported[2].at("v").template get<std::string>());
148}
