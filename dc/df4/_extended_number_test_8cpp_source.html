170TEST(ExtendedNumberTest, comparesAgainstAPlainValue) {
171 ExtendedRationalNumber const inf = ExtendedRationalNumber::infinity();
172 ExtendedRationalNumber const negInf = ExtendedRationalNumber::negativeInfinity();
173 ExtendedRationalNumber const two(rational(2));
174 storm::RationalNumber const plainTwo = rational(2);
175 storm::RationalNumber const plainThree = rational(3);
176
177 // A finite extended value compares as the value it holds, in both argument orders.
178 EXPECT_TRUE(two == plainTwo);
179 EXPECT_TRUE(plainTwo == two);
180 EXPECT_FALSE(two != plainTwo);
181 EXPECT_TRUE(two < plainThree);
182 EXPECT_TRUE(plainTwo < ExtendedRationalNumber(plainThree));
183 EXPECT_TRUE(two <= plainTwo);
184 EXPECT_TRUE(two >= plainTwo);
185 EXPECT_TRUE(ExtendedRationalNumber(plainThree) > plainTwo);
186 EXPECT_TRUE(plainThree > two);
187
188 // Every plain value lies strictly between the two infinities.
189 EXPECT_TRUE(negInf < plainTwo);
190 EXPECT_TRUE(plainTwo < inf);
191 EXPECT_FALSE(inf < plainTwo);
192 EXPECT_FALSE(plainTwo < negInf);
193 EXPECT_TRUE(inf > plainTwo);
194 EXPECT_TRUE(plainTwo > negInf);
195 EXPECT_TRUE(negInf <= plainTwo);
196 EXPECT_TRUE(inf >= plainTwo);
197
198 // An infinity is equal to no plain value at all.
199 EXPECT_FALSE(inf == plainTwo);
200 EXPECT_FALSE(negInf == plainTwo);
201 EXPECT_TRUE(inf != plainTwo);
202 EXPECT_TRUE(plainTwo != negInf);
203
204 // The mixed comparisons must agree with the ones that go through the conversion, which is what they replace.
205 for (auto const& plain : {rational(-1), rational(0), rational(2), rational(1000)}) {
206 for (auto const& extended : {inf, negInf, two}) {
207 EXPECT_EQ(extended < ExtendedRationalNumber(plain), extended < plain);
208 EXPECT_EQ(ExtendedRationalNumber(plain) < extended, plain < extended);
209 EXPECT_EQ(extended == ExtendedRationalNumber(plain), extended == plain);
210 EXPECT_EQ(extended <= ExtendedRationalNumber(plain), extended <= plain);
211 EXPECT_EQ(extended > ExtendedRationalNumber(plain), extended > plain);
212 EXPECT_EQ(extended >= ExtendedRationalNumber(plain), extended >= plain);
213 }
214 }
