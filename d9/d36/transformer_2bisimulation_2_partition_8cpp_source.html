84bool Partition::isSubBlockOf(Block const& subblock, Block const& superblock) const {
85 STORM_LOG_ASSERT(checkBlockValidity(subblock), "Invalid subblock.");
86 STORM_LOG_ASSERT(checkBlockValidity(superblock), "Invalid superblock.");
87 // Check if subblock is a subspan of superblock
88 return subblock.data() >= superblock.data() && (subblock.data() + subblock.size()) <= (superblock.data() + superblock.size());

119Partition::Block Partition::registerNewBlock(BlockIndex const start, BlockIndex const end) {
120 STORM_LOG_ASSERT(start < end && end <= blockContents.size(), "Invalid block range");
121 Block newBlock(blockContents.data() + start, end - start);
122 // Shrinking an existing block doesn't require iterating over all block contents
123 if (blockSizes[start] == 0) {
124 for (ElementIndex const e : newBlock) {
125 elementToBlockIndex[e] = start;
126 }
127 } else {
128 STORM_LOG_ASSERT(blockSizes[start] >= newBlock.size(), "Shrinking a block should not increase its size");
129 }
130 blockSizes[start] = newBlock.size();
131 return newBlock;
132}

134bool Partition::checkBlockValidity(Block const& block) const {
135 // block must not be empty
136 if (block.empty()) {
137 return false;
138 }
139 // the block refers to this partition
140 if (block.data() < blockContents.data() || block.data() + block.size() > blockContents.data() + blockContents.size()) {
141 return false;
142 }
143
144 BlockIndex const blockIndex = std::distance(blockContents.data(), block.data());
145 BlockIndex const blockEndIndex = blockIndex + block.size();
146
147 // the block has a stored size and is either the last block or ends where another block starts
148 if (!(blockSizes[blockIndex] > 0 && (blockEndIndex == blockContents.size() || blockSizes[blockEndIndex] != 0))) {
149 return false;
150 }
151
152 // The index of the first element must always match this block index (even if the element belongs to a subblock, it must have the same index)
153 if (elementToBlockIndex[block[0]] != blockIndex) {
154 return false;
155 }
156
157 // Check if the inverse mapping is consistent for the first and the last element (must actually be true for all elements but checking that is too costly)
158 if (&blockContents[blockContentsInverse[block.front()]] != &block.front() || &blockContents[blockContentsInverse[block.back()]] != &block.back()) {
159 return false;
160 }
161
162 if (isBlockOfElement(block, block[0])) {
163 // this block has no proper sub-block
164 // The stored sizes should coincide
165 if (blockSizes[blockIndex] != block.size()) {
166 return false;
167 }
168 // all elements should point to this blockIndex
169 // we only check this for the first and last element.
170 return (elementToBlockIndex[block.front()] == blockIndex) && (elementToBlockIndex[block.back()] == blockIndex);
171 } else {
172 // this block has proper sub-blocks.
173 // The stored size should be strictly smaller
174 if (blockSizes[blockIndex] >= block.size()) {
175 return false;
176 }
177 // The block index of all elements should be within the range of this block.
178 return std::all_of(block.begin(), block.end(), [this, blockIndex, blockEndIndex](ElementIndex const& e) {
179 // e's block index should be in the range of this block
180 return elementToBlockIndex[e] >= blockIndex && elementToBlockIndex[e] < blockEndIndex;
181 });
182 }

185bool Partition::isProperSuperBlock(Block const& block) const {
186 STORM_LOG_ASSERT(checkBlockValidity(block), "Tried to check if an invalid block is a proper superblock.");
187 // block is a proper super-block iff it has a proper sub-block iff the block of some element is not equal to block
188 return !isBlockOfElement(block, block[0]);
189}
