195std::pair<boost::optional<storm::RationalNumber>, boost::optional<bool>> SolverEnvironment::getPrecisionOfLinearEquationSolver(
196 storm::solver::EquationSolverType const& solverType) const {
197 std::pair<boost::optional<storm::RationalNumber>, boost::optional<bool>> result;
198 switch (solverType) {
199 case storm::solver::EquationSolverType::Gmmxx:
200 result.first = gmmxx().getPrecision();
201 break;
202 case storm::solver::EquationSolverType::Eigen:
203 result.first = eigen().getPrecision();
204 break;
205 case storm::solver::EquationSolverType::Native:
206 result.first = native().getPrecision();
207 result.second = native().getRelativeTerminationCriterion();
208 break;
209 case storm::solver::EquationSolverType::Elimination:
210 break;
211 case storm::solver::EquationSolverType::Topological:
212 result = getPrecisionOfLinearEquationSolver(topological().getUnderlyingEquationSolverType());
213 break;
214 default:
215 STORM_LOG_THROW(false, storm::exceptions::UnexpectedException, "The selected solver type is unknown.");
216 }
217 return result;
218}

220void SolverEnvironment::setLinearEquationSolverPrecision(boost::optional<storm::RationalNumber> const& newPrecision,
221 boost::optional<bool> const& relativePrecision) {
222 // Assert that each solver type is handled in this method.
223 STORM_LOG_ASSERT(getLinearEquationSolverType() == storm::solver::EquationSolverType::Native ||
224 getLinearEquationSolverType() == storm::solver::EquationSolverType::Gmmxx ||
225 getLinearEquationSolverType() == storm::solver::EquationSolverType::Eigen ||
226 getLinearEquationSolverType() == storm::solver::EquationSolverType::Elimination ||
227 getLinearEquationSolverType() == storm::solver::EquationSolverType::Topological,
228 "The current solver type is not respected in this method.");
229 if (newPrecision) {
230 native().setPrecision(newPrecision.get());
231 gmmxx().setPrecision(newPrecision.get());
232 eigen().setPrecision(newPrecision.get());
233 // Elimination and Topological solver do not have a precision
234 }
235 if (relativePrecision) {
236 native().setRelativeTerminationCriterion(relativePrecision.get());
237 // gmm, eigen, elimination, and topological solvers do not have a precision
238 }
239}
