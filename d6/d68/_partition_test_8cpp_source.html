10std::string blockToString(auto const& block) {
11 std::ostringstream oss;
12 oss << "{";
13 for (bool first{true}; auto const e : block) {
14 if (first) {
15 first = false;
16 } else {
17 oss << ", ";
18 }
19 oss << e;
20 }
21 oss << "}";
22 return oss.str();
23}

25bool equalBlocks(storm::bisimulation::Partition const& partition, std::initializer_list<uint64_t> const& expected,
26 storm::bisimulation::Partition::Block const& actual) {
27 bool const expectedContainsActual =
28 std::all_of(actual.begin(), actual.end(), [&expected](auto const& e) { return std::find(expected.begin(), expected.end(), e) != expected.end(); });
29 EXPECT_TRUE(expectedContainsActual) << "Content mismatch: " << blockToString(expected) << " vs. " << blockToString(actual);
30 bool const actualContainsExpected =

32 EXPECT_TRUE(actualContainsExpected) << "Content mismatch: " << blockToString(expected) << " vs. " << blockToString(actual);
33
34 // Also test the contains method
35 for (auto const& e : expected) {
36 EXPECT_TRUE(partition.contains(actual, e)) << "Element " << e << " not in block " << blockToString(actual);
37 }
38
39 return expectedContainsActual && actualContainsExpected;
40}

165TEST(PartitionTest, SplitByClustering) {
166 // Values are deliberately interleaved so that same-valued elements are not adjacent; splitBlockByClustering must not rely on any pre-existing order.
167 std::vector<int> const value = {0, 5, 0, 5, 0, 9};
168 storm::bisimulation::Partition partition(value.size());
169 auto const condition = [&value](uint64_t const a, uint64_t const b) { return value[a] != value[b]; };
170
171 EXPECT_TRUE(partition.splitBlockByClustering(partition.getUniversalBlock(), condition));
172 EXPECT_EQ(3ul, partition.getNumberOfBlocks());
173 partition.forEachBlock([&partition](auto const& block) {
174 ASSERT_TRUE(partition.checkBlockValidity(block)) << "Block " << blockToString(block) << " is not valid.";
175 if (partition.contains(block, 0)) {
176 EXPECT_TRUE(equalBlocks(partition, {0, 2, 4}, block));
177 } else if (partition.contains(block, 1)) {
178 EXPECT_TRUE(equalBlocks(partition, {1, 3}, block));
179 } else if (partition.contains(block, 5)) {
180 EXPECT_TRUE(equalBlocks(partition, {5}, block));
181 } else {
182 FAIL() << "Block " << blockToString(block) << " does not contain any expected elements.";
183 }
184 });
185
186 // A condition that never holds must not split anything.
187 {
188 storm::bisimulation::Partition trivialPartition(4);
189 EXPECT_FALSE(trivialPartition.splitBlockByClustering(trivialPartition.getUniversalBlock(), [](uint64_t const, uint64_t const) { return false; }));
190 EXPECT_EQ(1ul, trivialPartition.getNumberOfBlocks());
191 }
192
193 // Unlike a transitive-closure-based clustering, splitBlockByClustering only compares every element against the (fixed) anchor of its group, not
194 // against each other: with values 0, 4, 8 and a "must split" threshold of 5, element 1 (value 4) is within reach of anchor 0 (|4-0|=4) and gets
195 // grouped with it, even though it would also be within reach of element 2 (value 8, |8-4|=4) - once claimed by the first anchor, it is not
196 // reconsidered, so element 2 (|8-0|=8, exceeds the threshold) ends up alone rather than transitively merged with 0 via 1.
197 {
198 std::vector<int> const chainValue = {0, 4, 8};
199 storm::bisimulation::Partition chainPartition(chainValue.size());
200 auto const chainCondition = [&chainValue](uint64_t const a, uint64_t const b) { return std::abs(chainValue[a] - chainValue[b]) > 5; };
201 EXPECT_TRUE(chainPartition.splitBlockByClustering(chainPartition.getUniversalBlock(), chainCondition));
202 EXPECT_EQ(2ul, chainPartition.getNumberOfBlocks());
203 EXPECT_TRUE(equalBlocks(chainPartition, {0, 1}, chainPartition.getBlockOfElement(0)));
204 EXPECT_TRUE(equalBlocks(chainPartition, {2}, chainPartition.getBlockOfElement(2)));
205 }
206}
