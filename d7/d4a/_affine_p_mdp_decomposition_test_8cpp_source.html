30TEST(AffinePMdpDecompositionTest, AffineDtmc) {
31 carl::VariablePool::getInstance().clear();
32 auto cache = std::make_shared<storm::RawPolynomialCache>();
33 carl::Variable varP = storm::createRFVariable("p");
34 auto p = makeVarFunction(varP, cache);
35 auto one = storm::RationalFunction(1);
36
37 // 3-state DTMC: 0 -[p]-> 1, 0 -[1-p]-> 2, 1 -[1]-> 1, 2 -[1]-> 2.
38 storm::storage::SparseMatrixBuilder<storm::RationalFunction> builder(3, 3, 4);
39 builder.addNextValue(0, 1, p);
40 builder.addNextValue(0, 2, one - p);
41 builder.addNextValue(1, 1, one);
42 builder.addNextValue(2, 2, one);
43 auto matrix = builder.build();
44
45 storm::models::sparse::Dtmc<storm::RationalFunction> dtmc(matrix, trivialLabeling(3));
46
47 AffinePMdpDecomposition decomposition(dtmc);
48 EXPECT_EQ(3u, decomposition.getNumberOfRows());
49 ASSERT_EQ(1u, decomposition.getParameters().size());
50 EXPECT_EQ(varP, *decomposition.getParameters().begin());
51
52 auto const& row0 = decomposition.getRow(0);
53 ASSERT_EQ(2u, row0.size());
54 EXPECT_EQ(1u, row0[0].successor);
55 EXPECT_DOUBLE_EQ(0.0, row0[0].constant);
56 ASSERT_EQ(1u, row0[0].parameterCoefficients.size());
57 EXPECT_EQ(varP, row0[0].parameterCoefficients[0].first);
58 EXPECT_DOUBLE_EQ(1.0, row0[0].parameterCoefficients[0].second);
59
60 EXPECT_EQ(2u, row0[1].successor);
61 EXPECT_DOUBLE_EQ(1.0, row0[1].constant);
62 ASSERT_EQ(1u, row0[1].parameterCoefficients.size());
63 EXPECT_EQ(varP, row0[1].parameterCoefficients[0].first);
64 EXPECT_DOUBLE_EQ(-1.0, row0[1].parameterCoefficients[0].second);
65
66 auto const& row1 = decomposition.getRow(1);
67 ASSERT_EQ(1u, row1.size());
68 EXPECT_EQ(1u, row1[0].successor);
69 EXPECT_DOUBLE_EQ(1.0, row1[0].constant);
70 EXPECT_TRUE(row1[0].parameterCoefficients.empty());
71}

73TEST(AffinePMdpDecompositionTest, TransitionWithTwoParameters) {
74 carl::VariablePool::getInstance().clear();
75 auto cache = std::make_shared<storm::RawPolynomialCache>();
76 carl::Variable varP = storm::createRFVariable("p");
77 carl::Variable varQ = storm::createRFVariable("q");
78 auto p = makeVarFunction(varP, cache);
79 auto q = makeVarFunction(varQ, cache);
80 auto one = storm::RationalFunction(1);
81
82 // 0 -[p]-> 1, 0 -[q]-> 2, 0 -[1-p-q]-> 3; 1,2,3 self-absorbing.
83 storm::storage::SparseMatrixBuilder<storm::RationalFunction> builder(4, 4, 6);
84 builder.addNextValue(0, 1, p);
85 builder.addNextValue(0, 2, q);
86 builder.addNextValue(0, 3, one - p - q);
87 builder.addNextValue(1, 1, one);
88 builder.addNextValue(2, 2, one);
89 builder.addNextValue(3, 3, one);
90 auto matrix = builder.build();
91
92 storm::models::sparse::Dtmc<storm::RationalFunction> dtmc(matrix, trivialLabeling(4));
93
94 AffinePMdpDecomposition decomposition(dtmc);
95 EXPECT_EQ(2u, decomposition.getParameters().size());
96
97 auto const& row0 = decomposition.getRow(0);
98 ASSERT_EQ(3u, row0.size());
99
100 // Transition to 3 depends on both parameters, each with coefficient -1.
101 auto const& toThree = row0[2];
102 EXPECT_EQ(3u, toThree.successor);
103 EXPECT_DOUBLE_EQ(1.0, toThree.constant);
104 ASSERT_EQ(2u, toThree.parameterCoefficients.size());
105 for (auto const& [var, coeff] : toThree.parameterCoefficients) {
106 EXPECT_TRUE(var == varP || var == varQ);
107 EXPECT_DOUBLE_EQ(-1.0, coeff);
108 }

111TEST(AffinePMdpDecompositionTest, MdpWithTwoChoicesPerState) {
112 carl::VariablePool::getInstance().clear();

114 carl::Variable varP = storm::createRFVariable("p");

119
120 // State 0 has two choices: action 0 goes to 1 w.p. p / to 2 w.p. 1-p; action 1 goes to 1 w.p. q / to 2 w.p. 1-q.

122 storm::storage::SparseMatrixBuilder<storm::RationalFunction> builder(4, 3, 6, true, true, 3);
123 builder.newRowGroup(0);

125 builder.addNextValue(0, 2, one - p);
126 builder.addNextValue(1, 1, q);
127 builder.addNextValue(1, 2, one - q);
128 builder.newRowGroup(2);
129 builder.addNextValue(2, 1, one);
130 builder.newRowGroup(3);
131 builder.addNextValue(3, 2, one);
132 auto matrix = builder.build();
133
134 storm::models::sparse::Mdp<storm::RationalFunction> mdp(matrix, trivialLabeling(3));
135
136 AffinePMdpDecomposition decomposition(mdp);
137 EXPECT_EQ(4u, decomposition.getNumberOfRows());
138 EXPECT_EQ(2u, decomposition.getParameters().size());
139
140 auto const& choice0 = decomposition.getRow(0);
141 ASSERT_EQ(2u, choice0.size());
142 EXPECT_EQ(varP, choice0[0].parameterCoefficients.at(0).first);
143
144 auto const& choice1 = decomposition.getRow(1);
145 ASSERT_EQ(2u, choice1.size());
146 EXPECT_EQ(varQ, choice1[0].parameterCoefficients.at(0).first);
