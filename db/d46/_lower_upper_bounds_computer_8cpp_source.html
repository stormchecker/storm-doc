18template<typename ValueType, bool Lower>
19std::vector<ValueType> computeLowerOrUpperBoundsTotalReward(storm::storage::SparseMatrix<ValueType> const& matrix, std::vector<ValueType> const& offsets,
20 storm::storage::BitVector const& exitRows, std::vector<ValueType> const& exitProbabilities,
21 std::optional<storm::OptimizationDirection> dir) {
22 STORM_LOG_ASSERT(matrix.hasTrivialRowGrouping() || dir.has_value(), "Optimization direction needs to be given if matrix is nondeterministic.");
23 storm::utility::Extremum<Lower ? storm::OptimizationDirection::Minimize : storm::OptimizationDirection::Maximize, ValueType> globalBound;
24 // Try to compute a lower and an upper bound the "easy" way
25 auto const insideRows = ~exitRows;
26 std::conditional_t<Lower, std::greater_equal<ValueType>, std::less_equal<ValueType>> comp;
27 bool const canEasy =
28 std::all_of(insideRows.begin(), insideRows.end(), [&offsets, &comp](uint64_t row) { return comp(offsets[row], storm::utility::zero<ValueType>()); });
29
30 if (canEasy) {
31 for (auto const row : exitRows) {
32 STORM_LOG_ASSERT(!storm::utility::isZero(exitProbabilities.at(row)), "Exit probability must not be zero.");
33 ValueType const rowValue = offsets[row] / exitProbabilities[row];
34 globalBound &= rowValue;
35 }
36 }
37
38 // We might have to invoke the respective reward bound computers.
39 if (!canEasy) {
40 if (dir.has_value() && (Lower == minimize(*dir))) {
41 // Baier's algorithm provides a lower and an upper bound that hold for all strategies. It supports positive and negative rewards.
42 auto const bounds =
43 storm::modelchecker::helper::BaierUpperRewardBoundsComputer<ValueType>(matrix, exitProbabilities).computeTotalRewardBounds(offsets);
44 globalBound &= Lower ? bounds.lower : bounds.upper;
45 } else {
46 // For lower bounds, we actually compute upper bounds for the negated rewards.
47 // For upper bounds, we potentially cut away negative offsets.
48 std::optional<std::vector<ValueType>> tmpOffsets;
49 if (Lower) {
50 tmpOffsets.emplace(offsets.size());
51 storm::utility::vector::applyPointwise(offsets, *tmpOffsets,
52 [](ValueType const& v) { return std::max<ValueType>(storm::utility::zero<ValueType>(), -v); });
53 } else if (std::any_of(offsets.begin(), offsets.end(), [](ValueType const& v) { return v < storm::utility::zero<ValueType>(); })) {
54 tmpOffsets.emplace(offsets.size());
55 storm::utility::vector::applyPointwise(offsets, *tmpOffsets, [](ValueType const& v) { return std::max(storm::utility::zero<ValueType>(), v); });
56 }
57 auto bounds = storm::modelchecker::helper::DsMpiMdpUpperRewardBoundsComputer<ValueType>(
58 matrix, tmpOffsets.has_value() ? tmpOffsets.value() : offsets, exitProbabilities)
59 .computeUpperBounds();
60 if (Lower) {
61 storm::utility::vector::applyPointwise(bounds, bounds, [](ValueType const& v) { return -v; });
62 }
63 return bounds;
64 }
65 }
66 STORM_LOG_ASSERT(!globalBound.empty(), "Could not compute lower or upper bound for total reward.");
67 return std::vector<ValueType>(matrix.getRowGroupCount(), *globalBound);
68}
