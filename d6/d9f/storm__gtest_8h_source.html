26// Annotate test cases that are too expensive to run in every CI run (identified via profiling) with STORM_EXPENSIVE_*.
27// These macros expand to gtest's DISABLED_ prefix, so the tests are skipped by default and can be re-enabled at run time
28// without reconfiguring the build, either via --gtest_also_run_disabled_tests or the GTEST_ALSO_RUN_DISABLED_TESTS=1
29// environment variable.
30#define STORM_EXPENSIVE_TEST(test_suite_name, test_name) TEST(test_suite_name, DISABLED_##test_name)
31#define STORM_EXPENSIVE_TEST_F(test_suite_name, test_name) TEST_F(test_suite_name, DISABLED_##test_name)
32#define STORM_EXPENSIVE_TEST_P(test_suite_name, test_name) TEST_P(test_suite_name, DISABLED_##test_name)
33#define STORM_EXPENSIVE_TYPED_TEST(test_suite_name, test_name) TYPED_TEST(test_suite_name, DISABLED_##test_name)

94template<typename T1, typename T2>
95 requires(storm::test::IsExtendedNumber<T1>::value || storm::test::IsExtendedNumber<T2>::value)
96AssertionResult DoubleNearPredFormat(const char* expr1, const char* expr2, const char* abs_error_expr, T1 const& val1, T2 const& val2,
97 typename storm::test::FiniteValueTypeOf<T1>::type const& absError) {
98 if (storm::test::isInfiniteValue(val1) || storm::test::isInfiniteValue(val2)) {
99 // An infinity is within any distance of itself and of nothing else.
100 if (val1 == val2) {
101 return AssertionSuccess();
102 }
103 return AssertionFailure() << "The difference between " << expr1 << " and " << expr2 << " is not within " << abs_error_expr << ", where\n"
104 << expr1 << " evaluates to " << val1 << ",\n"
105 << expr2 << " evaluates to " << val2 << ".";
106 }
107 return DoubleNearPredFormat(expr1, expr2, abs_error_expr, storm::test::finiteValue(val1), storm::test::finiteValue(val2), absError);
108}
109} // namespace internal
110} // namespace testing
