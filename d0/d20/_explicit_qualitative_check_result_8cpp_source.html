49void ExplicitQualitativeCheckResult<ValueType>::performLogicalOperation(ExplicitQualitativeCheckResult<ValueType>& first, QualitativeCheckResult const& second,
50 bool logicalAnd) {
51 STORM_LOG_THROW(second.isExplicitQualitativeCheckResult(), storm::exceptions::InvalidOperationException,
52 "Cannot perform logical 'and' on check results of incompatible type.");
53 ExplicitQualitativeCheckResult<ValueType> const& secondCheckResult = static_cast<ExplicitQualitativeCheckResult<ValueType> const&>(second);
54 STORM_LOG_THROW(first.states == secondCheckResult.states && first.truthValues.size() == secondCheckResult.truthValues.size(),
55 storm::exceptions::InvalidOperationException, "Cannot perform logical 'and' on check results of incompatible type.");
56 if (logicalAnd) {
57 first.truthValues &= secondCheckResult.truthValues;
58 } else {
59 first.truthValues |= secondCheckResult.truthValues;
60 }
61}

150template<typename ValueType>
151void ExplicitQualitativeCheckResult<ValueType>::filter(QualitativeCheckResult const& filter) {
152 STORM_LOG_THROW(filter.isExplicitQualitativeCheckResult(), storm::exceptions::InvalidOperationException,
153 "Cannot filter explicit check result with non-explicit filter.");
154 STORM_LOG_THROW(filter.isResultForAllStates(), storm::exceptions::InvalidOperationException, "Cannot filter check result with non-complete filter.");
155 ExplicitQualitativeCheckResult<ValueType> const& explicitFilter = filter.template asExplicitQualitativeCheckResult<ValueType>();
156 vector_type const& filterTruthValues = explicitFilter.getTruthValuesVector();
157
158 // Line the filter up with the states this result has truth values for. The two need not span the same range
159 // of states, e.g. if this result holds a truth value for a single state only.
160 uint64_t const numStates = std::max(filterTruthValues.size(), states.size());
161 storm::storage::BitVector available = states;
162 available.resize(numStates);
163 storm::storage::BitVector selected(filterTruthValues);
164 selected.resize(numStates);
165 STORM_LOG_THROW(selected.isSubsetOf(available), storm::exceptions::InvalidOperationException,
166 "The check result fails to contain some results referred to by the filter.");
167
168 truthValues = truthValues % (selected % available);
169 states = filterTruthValues;
170}

196 std::optional<storm::storage::sparse::Valuations> const& stateValuations = std::nullopt,
197 std::optional<storm::models::sparse::StateLabeling> const& stateLabels = std::nullopt) {
