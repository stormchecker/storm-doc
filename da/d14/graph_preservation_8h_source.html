84template<typename ValueType>
85storm::pars::analysis::GraphPreservationStatus establishGraphPreservation(std::shared_ptr<storm::models::ModelBase> const& model,
86 std::vector<storm::storage::ParameterRegion<ValueType>> const& regions,
87 bool graphPreservationRequired) {
88 bool const assumed = storm::settings::getModule<storm::settings::modules::RegionSettings>().isAssumeGraphPreservingSet();
89 bool const needsGraphPreservation = assumed || graphPreservationRequired;
90 auto const collector = collectConstraints<ValueType>(model);
91
92 storm::pars::analysis::GraphPreservationStatus status =
93 needsGraphPreservation ? storm::pars::analysis::GraphPreservationStatus::GraphPreserving : storm::pars::analysis::GraphPreservationStatus::WellDefined;
94 for (auto const& region : regions) {
95 auto const result = checkRegionConstraints<ValueType>(collector, region, !needsGraphPreservation);
96 if (result && result->verdict == storm::pars::analysis::ConstraintVerdict::Sat) {
97 continue;
98 }
99 bool const violated = result && result->verdict == storm::pars::analysis::ConstraintVerdict::Unsat;
100
101 if (!needsGraphPreservation) {
102 // Ill-defined regions are handled by the engine itself, so we do not warn.
103 if (violated) {
104 status = storm::pars::analysis::GraphPreservationStatus::None;
105 } else if (status != storm::pars::analysis::GraphPreservationStatus::None) {
106 status = storm::pars::analysis::GraphPreservationStatus::Unknown;
107 }
108 } else if (assumed) {
109 STORM_LOG_WARN_COND(!violated, "Region " << region << " does not preserve the graph structure of the parametric model: "
110 << describeViolation(*result) << ". Set --assume-graph-preserving false.");
111 } else {
112 STORM_LOG_THROW(!violated, storm::exceptions::InvalidSettingsException,
113 "Region " << region << " is provably not graph-preserving: " << describeViolation(*result)
114 << ". Pass --assume-graph-preserving true to proceed nevertheless.");
115 STORM_LOG_THROW(false, storm::exceptions::NotSupportedException,
116 "Cannot prove that region " << region
117 << " preserves the graph structure of the parametric model. This is only possible for sparse "
118 "parametric models with decidable (e.g. linear) constraints. Pass --assume-graph-preserving "
119 "true to skip this check.");
120 }
121 }
122 return status;
123}
