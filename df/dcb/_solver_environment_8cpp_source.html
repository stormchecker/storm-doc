204std::pair<boost::optional<storm::RationalNumber>, boost::optional<bool>> SolverEnvironment::getPrecisionOfLinearEquationSolver(
205 storm::solver::EquationSolverType const& solverType) const {
206 std::pair<boost::optional<storm::RationalNumber>, boost::optional<bool>> result;
207 switch (solverType) {
208 case storm::solver::EquationSolverType::Gmmxx:
209 result.first = gmmxx().getPrecision();
210 break;
211 case storm::solver::EquationSolverType::Eigen:
212 result.first = eigen().getPrecision();
213 break;
214 case storm::solver::EquationSolverType::Native:
215 result.first = native().getPrecision();
216 result.second = native().getRelativeTerminationCriterion();
217 break;
218 case storm::solver::EquationSolverType::Elimination:
219 break;
220 case storm::solver::EquationSolverType::Topological:
221 result = getPrecisionOfLinearEquationSolver(topological().getUnderlyingEquationSolverType());
222 break;
223 default:
224 STORM_LOG_THROW(false, storm::exceptions::UnexpectedException, "The selected solver type is unknown.");
225 }
226 return result;
227}

229void SolverEnvironment::setLinearEquationSolverPrecision(boost::optional<storm::RationalNumber> const& newPrecision,
230 boost::optional<bool> const& relativePrecision) {
231 // Assert that each solver type is handled in this method.
232 STORM_LOG_ASSERT(getLinearEquationSolverType() == storm::solver::EquationSolverType::Native ||
233 getLinearEquationSolverType() == storm::solver::EquationSolverType::Gmmxx ||
234 getLinearEquationSolverType() == storm::solver::EquationSolverType::Eigen ||
235 getLinearEquationSolverType() == storm::solver::EquationSolverType::Elimination ||
236 getLinearEquationSolverType() == storm::solver::EquationSolverType::Topological,
237 "The current solver type is not respected in this method.");
238 if (newPrecision) {
239 native().setPrecision(newPrecision.get());
240 gmmxx().setPrecision(newPrecision.get());
241 eigen().setPrecision(newPrecision.get());
242 // Elimination and Topological solver do not have a precision
243 }
244 if (relativePrecision) {
245 native().setRelativeTerminationCriterion(relativePrecision.get());
246 // gmm, eigen, elimination, and topological solvers do not have a precision
247 }
248}
