12// A distinct seed for every expression type, so that two expressions of different types with equal
13// operator and operands do not collide. Collisions would be harmless for correctness, but avoiding
14// the obvious ones keeps the distribution reasonable.

42boost::any HashVisitor::visit(IfThenElseExpression const& expression, boost::any const& data) {
43 std::size_t result = TypeIfThenElse;
44 boost::hash_combine(result, hashExpression(*expression.getCondition()));
45 boost::hash_combine(result, hashExpression(*expression.getThenExpression()));
46 boost::hash_combine(result, hashExpression(*expression.getElseExpression()));
47 return result;
48}

114boost::any HashVisitor::visit(PredicateExpression const& expression, boost::any const& data) {
115 std::size_t result = TypePredicate;
116 boost::hash_combine(result, hashValue(expression.getPredicateType()));
117 for (uint_fast64_t i = 0; i < expression.getArity(); ++i) {
118 boost::hash_combine(result, hashExpression(*expression.getOperand(i)));
119 }
120 return result;
121}
