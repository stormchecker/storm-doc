31 bool treatUnspecifiedChoiceAsDontCare) {
32 STORM_LOG_THROW(pomdp.isCanonic(), storm::exceptions::IllegalArgumentException, "Applying an FSC requires a canonic POMDP.");
33 STORM_LOG_THROW(
34 pomdp.getInitialStates().hasUniqueSetBit(), storm::exceptions::NotSupportedException,
35 "Applying an FSC requires exactly one initial state, but the POMDP contains " << pomdp.getInitialStates().getNumberOfSetBits() << " initial states.");
36 for (auto const& [rewardModelName, rewardModel] : pomdp.getRewardModels()) {
37 STORM_LOG_THROW(!rewardModel.hasTransitionRewards(), storm::exceptions::NotSupportedException,
38 "Applying an FSC does not support transition rewards, but reward model \"" << rewardModelName << "\" contains them.");
39 }
40 std::unordered_map<uint64_t, std::pair<uint64_t, uint64_t>> idToPomdpStateAndFscNode;
41 std::unordered_map<uint64_t, std::unordered_map<uint64_t, uint64_t>> pomdpStateAndFscNodeToNewStateId;
42 storm::storage::SparseMatrixBuilder<PomdpValueType> matrixBuilder;
43 std::unordered_map<std::string, std::vector<PomdpValueType>> stateRewards;
44
45 uint64_t initialStateId = 0ul;
46 uint64_t nextStateId = 1ul;
47 uint64_t initialPomdpState = pomdp.getInitialStates().getNextSetIndex(0ul);
48 idToPomdpStateAndFscNode[initialStateId] = {initialPomdpState, fsc.getInitialNodeId()};
49 pomdpStateAndFscNodeToNewStateId[initialPomdpState][fsc.getInitialNodeId()] = initialStateId;
50 STORM_LOG_DEBUG("Add initial state " << initialStateId << " (P: " << initialPomdpState << ", F: " << fsc.getInitialNodeId() << ")");
51
52 std::optional<std::unordered_map<uint64_t, uint64_t>> pomdpObservationIdToFscObservationId = std::nullopt;
53 std::optional<std::unordered_map<uint64_t, std::unordered_map<uint64_t, uint64_t>>> fscActionIdToPomdpActionIdMap = std::nullopt;
54 // If the POMDP and FSC have matching metadata, use it to create mappings between actions and observations.
55 // Otherwise, we assume that IDs correspond directly.
56 if (fsc.hasIdToObservationSignatureMap()) {
57 STORM_LOG_THROW(pomdp.hasObservationValuations(), storm::exceptions::UnexpectedException,
58 "The FSC uses observation signatures, but the POMDP has no observation valuations. Cannot apply the FSC robustly.");
59 std::unordered_map<uint64_t, uint64_t> observationMap;
60 for (uint64_t pomdpObservationId = 0; pomdpObservationId < pomdp.getNrObservations(); ++pomdpObservationId) {
61 auto const observationSignature = getObservationSignature(pomdp, pomdpObservationId);
62 auto fscObservationId = fsc.getObservationIdBySignature(observationSignature);
63 STORM_LOG_THROW(fscObservationId.has_value(), storm::exceptions::UnexpectedException,
64 "Could not map POMDP observation " << pomdpObservationId << " (" << observationSignatureToString(observationSignature)
65 << ") to an FSC observation.");
66 observationMap[pomdpObservationId] = *fscObservationId;
67 }
68 pomdpObservationIdToFscObservationId = std::move(observationMap);
69 }
70
71 if (fsc.hasIdToActionNameMap()) {
72 STORM_LOG_THROW(pomdp.hasChoiceLabeling(), storm::exceptions::UnexpectedException,
73 "The FSC uses action names, but the POMDP has no choice labeling. Cannot apply the FSC robustly.");
74 std::unordered_map<uint64_t, std::unordered_map<uint64_t, uint64_t>> actionMap;
75 auto const nrStates = pomdp.getNumberOfStates();
76 std::vector<uint64_t> representativeStates(pomdp.getNrObservations(), nrStates);
77 for (uint64_t state = 0; state < nrStates; ++state) {
78 auto& representativeState = representativeStates[pomdp.getObservation(state)];
79 if (representativeState == nrStates) {
80 representativeState = state;
81 }
82 }
83 for (uint64_t pomdpObservationId = 0; pomdpObservationId < pomdp.getNrObservations(); ++pomdpObservationId) {
84 auto const representativeState = representativeStates[pomdpObservationId];
85 if (representativeState == nrStates) {
86 continue;
87 }
88 uint64_t const pomdpRowGroupIndex = pomdp.getTransitionMatrix().getRowGroupIndices()[representativeState];
89 uint64_t const pomdpRowGroupSize = pomdp.getTransitionMatrix().getRowGroupSize(representativeState);
90 uint64_t const fscObservationId = pomdpObservationIdToFscObservationId && pomdpObservationIdToFscObservationId->contains(pomdpObservationId)
91 ? pomdpObservationIdToFscObservationId->at(pomdpObservationId)
92 : pomdpObservationId;
93 for (uint64_t localActionIndex = 0; localActionIndex < pomdpRowGroupSize; ++localActionIndex) {
94 // Numeric fallbacks are only valid for FSC actions without a name, regardless of the POMDP choice labels.
95 if (!fsc.hasActionName(fscObservationId, localActionIndex)) {
96 actionMap[fscObservationId].try_emplace(localActionIndex, localActionIndex);
97 }
98 auto const& labels = pomdp.getChoiceLabeling().getLabelsOfChoice(pomdpRowGroupIndex + localActionIndex);
99 if (labels.empty()) {
100 continue;
101 }
102 std::string const& actionName = *labels.begin();
103 if (auto fscActionId = fsc.getActionIdByName(fscObservationId, actionName)) {
104 actionMap[fscObservationId][*fscActionId] = localActionIndex;
105 }
106 }
107 }
108 fscActionIdToPomdpActionIdMap = std::move(actionMap);
109 }
110
111 if (!pomdpObservationIdToFscObservationId && !fscActionIdToPomdpActionIdMap) {
112 STORM_LOG_WARN(
113 "The POMDP or the FSC is missing naming information for actions or observations. Assuming that IDs correspond directly (this might lead to "
114 "unexpected results if they don't).");
115 }
116 std::queue<uint64_t> statesToExplore;
117 statesToExplore.push(initialStateId);
118 while (!statesToExplore.empty()) {
119 uint64_t currentStateId = statesToExplore.front();
120 auto [currentPomdpState, currentFscNode] = idToPomdpStateAndFscNode[currentStateId];
121 statesToExplore.pop();
122 STORM_LOG_DEBUG("Explore state " << currentStateId << " (P: " << currentPomdpState << ", F: " << currentFscNode << ")");
123 storm::storage::Distribution<FscValueType, uint64_t> actionDistribution;
124 uint64_t nextFSCNode;
125 uint64_t const pomdpObservationId = pomdp.getObservation(currentPomdpState);
126 uint64_t const fscObservationId = pomdpObservationIdToFscObservationId && pomdpObservationIdToFscObservationId->contains(pomdpObservationId)
127 ? pomdpObservationIdToFscObservationId->at(pomdpObservationId)
128 : pomdpObservationId;
129 bool const hasFscOutput = fsc.hasOutputForObservationInNode(currentFscNode, fscObservationId);
130 if (hasFscOutput) {
131 std::tie(actionDistribution, nextFSCNode) = fsc.getActionDistributionAndSuccessorForObservationInNode(currentFscNode, fscObservationId);
132 } else {
133 STORM_LOG_THROW(treatUnspecifiedChoiceAsDontCare, storm::exceptions::UnexpectedException,
134 "Expected FSC to have an output for observation " << fscObservationId << " in FSC node " << currentFscNode << ", but it does not.");
135 // If requested, pick POMDP local action zero and stay in the same FSC node.
136 nextFSCNode = currentFscNode;
137 actionDistribution = storm::storage::Distribution<FscValueType, uint64_t>();
138 actionDistribution.addProbability(0ul, storm::utility::one<FscValueType>());
139 }
140 for (auto const& [rewardModelName, rewardModel] : pomdp.getRewardModels()) {
141 auto& rewards = stateRewards[rewardModelName];
142 rewards.resize(currentStateId + 1, storm::utility::zero<PomdpValueType>());
143 if (rewardModel.hasStateRewards()) {
144 rewards[currentStateId] += rewardModel.getStateRewardVector()[currentPomdpState];
145 }
146 }
147 // Accumulate contributions from different actions to the same successor before inserting the row into the matrix.
148 // This avoids repeated matrix lookups and insertions while the builder keeps the row sorted.
149 std::unordered_map<uint64_t, PomdpValueType> successorProbabilities;
150 for (auto const& actionEntry : actionDistribution) {
151 auto const actionProbability = storm::utility::convertNumber<PomdpValueType>(actionEntry.second);
152 uint64_t pomdpActionId = actionEntry.first;
153 if (hasFscOutput && fscActionIdToPomdpActionIdMap) {
154 STORM_LOG_THROW(fscActionIdToPomdpActionIdMap->contains(fscObservationId) &&
155 fscActionIdToPomdpActionIdMap->at(fscObservationId).contains(actionEntry.first),
156 storm::exceptions::UnexpectedException,
157 "Could not map FSC action " << actionEntry.first << " for FSC observation " << fscObservationId << " to a POMDP action.");
158 pomdpActionId = fscActionIdToPomdpActionIdMap->at(fscObservationId).at(actionEntry.first);
159 }
160 for (auto const& [rewardModelName, rewardModel] : pomdp.getRewardModels()) {
161 // We take the expected reward over the action distribution, so not all properties are preserved.
162 if (rewardModel.hasStateActionRewards()) {
163 uint64_t globalActionIndex = pomdp.getTransitionMatrix().getRowGroupIndices()[currentPomdpState] + pomdpActionId;
164 stateRewards[rewardModelName][currentStateId] += rewardModel.getStateActionRewardVector()[globalActionIndex] * actionProbability;
165 }
166 }
167 for (auto const& transitionEntry : pomdp.getTransitionMatrix().getRow(currentPomdpState, pomdpActionId)) {
168 uint64_t successorPomdpState = transitionEntry.getColumn();
169 uint64_t successorNewStateId;
170 if (pomdpStateAndFscNodeToNewStateId.contains(successorPomdpState) &&
171 pomdpStateAndFscNodeToNewStateId[successorPomdpState].contains(nextFSCNode)) {
172 successorNewStateId = pomdpStateAndFscNodeToNewStateId[successorPomdpState][nextFSCNode];
173 } else {
174 successorNewStateId = nextStateId;
175 idToPomdpStateAndFscNode[successorNewStateId] = {
176 successorPomdpState,
177 nextFSCNode,
178 };
179 pomdpStateAndFscNodeToNewStateId[successorPomdpState][nextFSCNode] = successorNewStateId;
180 STORM_LOG_DEBUG("Add state " << successorNewStateId << " (P: " << successorPomdpState << ", F: " << nextFSCNode << ")");
181 ++nextStateId;
182 statesToExplore.push(successorNewStateId);
183 }
184 if (successorProbabilities.contains(successorNewStateId)) {
185 // Add transition from currentStateId to successorNewStateId with probability from POMDP
186 successorProbabilities[successorNewStateId] += transitionEntry.getValue() * actionProbability;
187 } else {
188 successorProbabilities[successorNewStateId] = transitionEntry.getValue() * actionProbability;
189 }
190 }
191 }
192 for (auto const& [successorStateId, probability] : successorProbabilities) {
193 matrixBuilder.addNextValue(currentStateId, successorStateId, probability);
194 }
195 }
196 storm::storage::sparse::ModelComponents<PomdpValueType> modelComponents;
197 modelComponents.transitionMatrix = matrixBuilder.build(nextStateId, nextStateId);
198
199 // Add state labeling
200 storm::models::sparse::StateLabeling stateLabeling(nextStateId);
201 // Eagerly add all labels to preserve labels that may be unreachable under the FSC, but relevant for a property
202 for (auto const& label : pomdp.getStateLabeling().getLabels()) {
203 stateLabeling.addLabel(label);
204 }
205 stateLabeling.addLabelToState("init", initialStateId);
206 for (uint64_t pomdpState = 0; pomdpState < pomdp.getNumberOfStates(); ++pomdpState) {
207 if (pomdpStateAndFscNodeToNewStateId.contains(pomdpState)) {
208 for (const auto& newStateId : pomdpStateAndFscNodeToNewStateId[pomdpState] | std::views::values) {
209 for (auto const& label : pomdp.getStateLabeling().getLabelsOfState(pomdpState)) {
210 if (label != "init") {
211 stateLabeling.addLabelToState(label, newStateId);
212 }
213 }
214 }
215 }
216 }
217 modelComponents.stateLabeling = stateLabeling;
218
219 // build reward models
220 for (auto const& [rewardModelName, rewardValues] : stateRewards) {
221 storm::models::sparse::StandardRewardModel<PomdpValueType> rewardModel(rewardValues);
222 modelComponents.rewardModels[rewardModelName] = rewardModel;
223 }
224
225 return storm::models::sparse::Dtmc<PomdpValueType>(modelComponents);
226}
