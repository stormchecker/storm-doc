89template<typename ValueType, SplitterRefinementMode Mode>
90void recomputeSilentStates(SplitterRefinementContext<ValueType, Mode>& context, auto const& candidates)
91 requires(Mode != SplitterRefinementMode::Strong)
92{
93 // A state can only ever go from silent to non-silent.
94 for (uint64_t const state : candidates) {
95 context.weakData->silentStates.set(state, isSilentState(context.model.getTransitionMatrix(), context.partition, state));
96 }

102template<typename ValueType, SplitterRefinementMode Mode>
103bool checkSilentStates(SplitterRefinementContext<ValueType, Mode> const& context)
104 requires(Mode != SplitterRefinementMode::Strong)
105{
106 for (uint64_t state = 0; state < context.partition.getNumberOfElements(); ++state) {
107 if (context.weakData->silentStates.get(state) != isSilentState(context.model.getTransitionMatrix(), context.partition, state)) {
108 return false;
109 }
110 }
111 return true;
112}

123template<typename ValueType, SplitterRefinementMode Mode>
124void enqueueSubBlocks(SplitterRefinementContext<ValueType, Mode>& context, storm::bisimulation::Partition::Block const splitBlock) {
125 bool const wasInQueue = context.queue.erase(splitBlock) > 0;
126 storm::bisimulation::Partition::Block largestSubBlock;
127 context.partition.forEachSubBlock(splitBlock, [&context, &largestSubBlock](auto const& subBlock) {
128 context.queue.insert(subBlock);
129 if (subBlock.size() > largestSubBlock.size()) {
130 largestSubBlock = subBlock;
131 }
132 });
133 if (Mode == SplitterRefinementMode::Strong && !wasInQueue) {
134 context.queue.erase(largestSubBlock);
135 }

143template<typename ValueType, SplitterRefinementMode Mode>
144void refineBlockStrong(SplitterRefinementContext<ValueType, Mode>& context, storm::bisimulation::Partition::Block const predecessorBlockToSplit) {
145 auto& predecessorToSplitterProbabilities = context.cache.predecessorToSplitterProbabilities;
146
147 // First split the block by whether a state is a predecessor of the splitter block or not
148 // We do this by either iterating over the splitterPredecessors or the predecessorBlockToSplit, depending on what is shorter.
149 auto [noPredecessors, predecessors] =
150 predecessorToSplitterProbabilities.getNonDefaultStates().size() < predecessorBlockToSplit.size()
151 ? context.partition.splitBlockByRange(predecessorBlockToSplit, predecessorToSplitterProbabilities.getNonDefaultStates())
152 : context.partition.splitBlockByPredicate(
153 predecessorBlockToSplit, [&predecessorToSplitterProbabilities](
154 auto const& state) { return !storm::utility::isZero(predecessorToSplitterProbabilities.getValues()[state]); });
155
156 STORM_LOG_ASSERT(!predecessors.empty(), "The predecessor block should contain at least one predecessor state.");
157 bool wasSplit = noPredecessors.size() > 0;
158
159 // Splitting with interval probabilities is not trivial: it is not clear whether the entire toSplitterProbs interval (which might be the sum of several
160 // transitions) is feasible.
161 static_assert(!storm::IsIntervalType<ValueType>, "Interval-valued splitter probabilities are not supported.");
162 auto const& toSplitterProbs = predecessorToSplitterProbabilities.getValues();
163 auto const less = [&toSplitterProbs](uint64_t const state1, uint64_t const state2) { return toSplitterProbs[state1] < toSplitterProbs[state2]; };
164 if (storm::utility::isZero(context.tolerance)) {
165 // Attention: Do not short circuit, i.e., wasSplit = wasSplit || foo() might not execute foo()
166 wasSplit |= context.partition.splitBlockByOrder(predecessors, less);
167 } else {
168 auto const lessTolerance = [&toSplitterProbs, &context](uint64_t const state1, uint64_t const state2) {
169 return toSplitterProbs[state1] + context.tolerance < toSplitterProbs[state2];
170 };
171 // Attention: Do not short circuit
172 wasSplit |= context.partition.splitBlockByOrder(predecessors, less, lessTolerance);
173 }
174
175 if (wasSplit) {
176 enqueueSubBlocks(context, predecessorBlockToSplit);
177 }
178}

187template<typename ValueType>
188void refineBlockWeak(SplitterRefinementContext<ValueType, SplitterRefinementMode::WeakDiscreteTime>& context, storm::bisimulation::Partition::Block const block,
189 storm::bisimulation::Partition::Block const splitterBlock) {
190 STORM_LOG_ASSERT(!context.weakData->isDivergent(block), "Assumed a non-divergent block as a predecessor block of the splitter.");
191
192 // Step 1: Gather the non-silent states of the given block. If the block is large, it is usually faster to iterate over the non-silent states
193 auto& frontierStates = context.cache.frontierStates;
194 auto const& silentStates = context.weakData->silentStates;
195 frontierStates.clear();
196 if (block.size() * 64 > silentStates.size()) {
197 for (uint64_t nonSilentState = silentStates.getNextUnsetIndex(0); nonSilentState < silentStates.size();
198 nonSilentState = silentStates.getNextUnsetIndex(nonSilentState + 1)) {
199 if (context.partition.contains(block, nonSilentState)) {
200 frontierStates.push_back(nonSilentState);
201 }
202 }
203 } else {
204 for (uint64_t const state : block) {
205 if (!silentStates.get(state)) {
206 frontierStates.push_back(state);
207 }
208 }
209 }
210
211 // Step 2: Compute conditional escape probability at frontier states
212 auto& conditionalValues = context.cache.conditionalValues;
213 auto const& toSplitterProbs = context.cache.predecessorToSplitterProbabilities.getValues();
214 for (uint64_t const state : frontierStates) {
215 if (storm::utility::isZero(toSplitterProbs[state])) {
216 conditionalValues[state] = storm::utility::zero<ValueType>(); // the state has no transition into the splitter
217 continue;
218 }
219 auto const row = context.model.getTransitionMatrix().getRow(state);
220 STORM_LOG_ASSERT(std::any_of(row.begin(), row.end(),
221 [&context, &splitterBlock](auto const& entry) {
222 return !storm::utility::isZero(entry.getValue()) && context.partition.contains(splitterBlock, entry.getColumn());
223 }),
224 "Expected a transition into a splitter, but none was found.");
225 ValueType escapeValue = storm::utility::zero<ValueType>();
226 bool leavesOnlyToSplitter = true;
227 for (auto const& entry : row) {
228 if (storm::utility::isZero(entry.getValue()) || context.partition.isSameBlock(state, entry.getColumn())) {
229 continue; // moves within the own block are unobservable
230 }
231 escapeValue += entry.getValue();
232 leavesOnlyToSplitter = leavesOnlyToSplitter && context.partition.contains(splitterBlock, entry.getColumn());
233 }
234 STORM_LOG_ASSERT(!storm::utility::isZero(escapeValue), "A non-silent state must be able to leave its block.");
235
236 // If there are only transitions to the splitter, we set the conditional probability to 1 explicitly.
237 // This avoids numerical issues where toSplitterProbs[state] and escapeValue do not match exactly.
238 conditionalValues[state] = leavesOnlyToSplitter ? storm::utility::one<ValueType>() : toSplitterProbs[state] / escapeValue;
239 }
240 // Every state of a non-divergent block can leave the block; the first state that is non-silent on such a path witnesses this.
241 STORM_LOG_ASSERT(!frontierStates.empty(), "A non-divergent block must contain a non-silent state.");
242
243 // Step 3: Divide the frontier states into equivalence classes based on their conditional values
244 auto& stateClasses = context.cache.temporaryStateClasses;
245 uint64_t constexpr Unclassified = std::numeric_limits<uint64_t>::max(); // Indicates that no class has been assigned to a state
246 uint64_t constexpr MultipleFrontiers =
247 std::numeric_limits<uint64_t>::max() - 1; // Indicates that multiple frontier states with different conditional values can be reached
248
249 std::sort(frontierStates.begin(), frontierStates.end(),
250 [&conditionalValues](uint64_t const state1, uint64_t const state2) { return conditionalValues[state1] < conditionalValues[state2]; });
251 {
252 // true if state1 and state2 should be in different classes. Assumes conditionalValues[state1] <= conditionalValues[state2].
253 auto const haveDifferentClass = [&conditionalValues, &context](uint64_t const state1, uint64_t const state2) {
254 // Whether a state can move to the splitter at all is not subject to the tolerance
255 if (storm::utility::isZero(context.tolerance) || storm::utility::isZero(conditionalValues[state1]) ||
256 storm::utility::isZero(conditionalValues[state2])) {
257 return conditionalValues[state1] != conditionalValues[state2];
258 }
259 return conditionalValues[state1] + context.tolerance < conditionalValues[state2];
260 };
261
262 uint64_t currClassRepresentative = frontierStates.front();
263 stateClasses[currClassRepresentative] = currClassRepresentative;
264 for (uint64_t const state : frontierStates | std::views::drop(1)) {
265 if (haveDifferentClass(currClassRepresentative, state)) {
266 currClassRepresentative = state;
267 }
268 stateClasses[state] = currClassRepresentative;
269 }
270 }
271
272 // Step 4: Catch the trivial case where all states have the same class
273 if (stateClasses[frontierStates.front()] == stateClasses[frontierStates.back()]) {
274 // Just a single class, nothing to split.
275 // Clean-up the touched classes before return.
276 for (uint64_t const state : frontierStates) {
277 stateClasses[state] = Unclassified;
278 }
279 return;
280 }
281
282 // Step 5: Classify the remaining states. Two states state1 and state2 are in the same class iff
283 // - a) all frontier states reachable from state1 or from state2 are in the same class, or
284 // - b) both, state1 and state2 each reach multiple frontier states with distinct classes.
285 // We do a backwards search from the frontier states, propagating their class labels to their predecessors.
286 // A BFS is used (rather than a DFS) to detect case b) more quickly.
287 auto& bfsQueue = context.cache.bfsQueue;
288 bfsQueue.clear();
289 for (auto const state : frontierStates) {
290 bfsQueue.push_back(state);
291 }
292 // We also collect candidates of states that might lose their silence after refinement: the states for which case b) applies that have a case a) successor.
293 std::vector<uint64_t>& nonSilentCandidates = context.cache.nonSilentCandidates;
294 nonSilentCandidates.clear();
295 while (!bfsQueue.empty()) {
296 uint64_t const currentState = bfsQueue.front();
297 bfsQueue.pop_front();
298 uint64_t const currentClass = stateClasses[currentState];
299 STORM_LOG_ASSERT(currentClass != Unclassified, "The current state must have a class assigned.");
300 for (auto const& predecessorEntry : context.backwardTransitions.getRow(currentState)) {
301 uint64_t const predecessor = predecessorEntry.getColumn();
302 uint64_t& predecessorClass = stateClasses[predecessor];
303 if (predecessorClass == Unclassified) {
304 // Ignore predecessors outside of the block.
305 if (context.partition.isBlockOfElement(block, predecessor)) {
306 STORM_LOG_ASSERT(silentStates.get(predecessor), "An unclassified state must be silent.");
307 predecessorClass = currentClass; // propagate the current class
308 bfsQueue.push_back(predecessor);
309 }
310 } else if (predecessorClass == currentClass) {
311 // The class of predecessor does not change. Nothing to do.
312 } else if (predecessorClass == MultipleFrontiers) {
313 // case b) applies for the predecessor, but case a) potentially applies for the current state.
314 nonSilentCandidates.push_back(predecessor);
315 } else if (silentStates.get(predecessor)) {
316 // So far, we assumed that case a) applies for the predecessor, but we have now found a successor in a different class. Hence, Case b) applies.
317 predecessorClass = MultipleFrontiers;
318 bfsQueue.push_back(predecessor);
319 nonSilentCandidates.push_back(predecessor);
320 } else {
321 // The last case only applies for a non-silent (i.e. frontier) predecessor. Hence, the class of the predecessor does not change. Nothing to do.
322 }
323 }
324 }
325
326 // Step 6: Split the block by the computed classes and enqueue the subblocks
327 context.partition.splitBlockByOrder(block,
328 [&stateClasses](uint64_t const state1, uint64_t const state2) { return stateClasses[state1] < stateClasses[state2]; });
329 STORM_LOG_ASSERT(context.partition.isProperSuperBlock(block), "As there are multiple different frontier states, the block must be split.");
330 enqueueSubBlocks(context, block);
331
332 // Step 7: update silent states and clear cached data.
333 recomputeSilentStates(context, nonSilentCandidates);
334 for (uint64_t const state : block) {
335 stateClasses[state] = Unclassified;
336 }
337}

339template<typename ValueType, SplitterRefinementMode Mode>
340void refinePartitionBasedOnSplitter(SplitterRefinementContext<ValueType, Mode>& context, storm::bisimulation::Partition::Block const splitterBlock) {
341 auto& predecessorToSplitterProbabilities = context.cache.predecessorToSplitterProbabilities;
342 auto& predecessorBlocks = context.cache.predecessorBlocks;
343
344 for (auto currentState : splitterBlock) {
345 // Compute probability to enter splitter block for each predecessor
346 for (const auto& predecessorEntry : context.backwardTransitions.getRow(currentState)) {
347 auto predecessorState = predecessorEntry.getColumn();
348 auto predecessorBlock = context.partition.getBlockOfElement(predecessorState);
349 if (predecessorBlock.size() == 1) {
350 continue; // No need to try to split singleton blocks
351 }
352 if constexpr (SplitterRefinementContext<ValueType, Mode>::isWeak) {
353 // The states of a divergent block can never leave it, so they all exhibit the same (unobservable) behavior and must not be split.
354 if (context.weakData->divergentStates.get(predecessorState)) {
355 continue;
356 }
357 // For weak bisimulation, the moves of a non-step-sensitive state within its own block are unobservable.
358 // Hence, the splitter must not split itself.
359 if (context.partition.isEqualBlock(predecessorBlock, splitterBlock) && !context.isStepSensitive(predecessorState)) {
360 continue;
361 }
362 }
363 predecessorToSplitterProbabilities.addValue(predecessorState, predecessorEntry.getValue());
364 predecessorBlocks.insert(predecessorBlock);
365 }
366 }
367
368 while (!predecessorBlocks.empty()) {
369 auto const predecessorBlockToSplit = predecessorBlocks.pop();
370 if constexpr (Mode == SplitterRefinementMode::WeakDiscreteTime) {
371 if (!context.isStepSensitive(predecessorBlockToSplit.front())) {
372 refineBlockWeak(context, predecessorBlockToSplit, splitterBlock);
373 continue;
374 }
375 }
376 refineBlockStrong(context, predecessorBlockToSplit);
377 }
378
379 // Reset the predecessorToSplitterProbabilities for the next iteration.
380 predecessorToSplitterProbabilities.clear();
381}

385template<typename ValueType, SplitterRefinementMode Mode>
386void performSplitterBasedRefinement(storm::models::sparse::Model<ValueType> const& model, storm::storage::SparseMatrix<ValueType> const& backwardTransitions,
387 storm::bisimulation::Partition& partition, ValueType const tolerance, storm::OptionalRef<WeakBisimulationData> weakData) {
388 static_assert(!storm::IsIntervalType<ValueType>, "Interval types are not supported for splitter-based refinement.");
389 // Refinement for interval models requires limiting signatures to feasible intervals. This is rather difficult in a splitter-based setting.
390 STORM_LOG_THROW(!model.isNondeterministicModel(), storm::exceptions::InvalidArgumentException,
391 "Splitter-based refinement is only supported for deterministic models.");
392 STORM_LOG_THROW((storm::utility::isZero(tolerance) || !std::is_same_v<ValueType, storm::RationalFunction>), storm::exceptions::InvalidArgumentException,
393 "Splitter-based refinement with non-zero tolerance does not apply to parametric models.");
394 detail::SplitterRefinementContext<ValueType, Mode> context(model, backwardTransitions, partition, tolerance, weakData);
395 if constexpr (Mode != SplitterRefinementMode::Strong) {
396 STORM_LOG_ASSERT(weakData->checkBlockHomogeneity(partition), "Partition is not homogeneous with respect to the weak bisimulation data.");
397 STORM_LOG_ASSERT(detail::checkSilentStates(context), "The given silent states do not match the given partition.");
398 }
399 // Initially, add all current blocks to the queue.
400 partition.forEachBlock([&context](auto const& block) { context.queue.insert(block); });
401
402 // Perform the splitting until there are no more splitters.
403 while (!context.queue.empty()) {
404 // take the smallest block from the queue
405 auto const splitterBlock = *context.queue.begin();
406 context.queue.erase(context.queue.begin());
407 STORM_LOG_ASSERT(!partition.isProperSuperBlock(splitterBlock), "Broken invariant: the queue should not contain blocks that have been split.");
408 // Split the predecessor blocks and add them to the queue
409 detail::refinePartitionBasedOnSplitter(context, splitterBlock);
410 }
411
412 if constexpr (Mode != SplitterRefinementMode::Strong) {
413 // The refinement itself only keeps the silent states of the blocks refined through refineBlockWeak up to date.
414 if (Mode == SplitterRefinementMode::WeakDiscreteTime) {
415 detail::recomputeSilentStates(context, context.weakData->stepSensitiveStates);
416 } else {
417 detail::recomputeSilentStates(context, context.weakData->silentStates);
418 }
419 STORM_LOG_ASSERT(detail::checkSilentStates(context), "The silent states do not match the final partition.");
420 }
421}
