52 BeliefClipping clipping = clipBeliefToGrid(belief, observationResolutions[belief.observation()]);
53 if (clipping.isClippable) {
54 BeliefValueType a = (storm::utility::one<BeliefValueType>() - clipping.delta) * probabilityFactor;
55 BeliefValueType b = clipping.delta * probabilityFactor;
56 if (extremalRewardValues.has_value()) {
57 // We compute the reward adjustment necessary for clipping (see https://doi.org/10.48550/arXiv.2201.08772)
58 // Because we don't add clipped beliefs into the abstraction MDP, we compute the influence of the clipping transition (based on the expected
59 // reward value in the implicit clipped belief) here such that we can simply add the value to the state-action reward of the transition
60 auto rewardAdjustment = storm::utility::zero<BeliefValueType>();
61 for (auto const& [state, deltaValue] : clipping.deltaValues) {
62 rewardAdjustment += deltaValue * extremalRewardValues->at(state);
63 }
64 rewardAdjustment = probabilityFactor * rewardAdjustment;
65 callback(std::move(clipping.targetBelief), std::move(a), std::move(b), std::move(rewardAdjustment));
66 } else {
67 callback(std::move(clipping.targetBelief), std::move(a), std::move(b), std::nullopt);
68 }
69 } else {
70 // Belief on Grid
71 callback(std::move(belief), std::move(probabilityFactor), std::nullopt, std::nullopt);
72 }
