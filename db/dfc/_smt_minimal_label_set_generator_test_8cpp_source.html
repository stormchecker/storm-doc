47// The generator reasons symbolically about which edges can enable each other. Since the location of an automaton is
48// not part of the variable valuation, that reasoning has to take the source and target locations of the edges into
49// account. Models that keep their control flow in locations rather than in variables (as, e.g., models exported by
50// Modest do) used to make the generator fail with a 'map::at: key not found' error.
51TEST(SmtMinimalLabelSetGeneratorTest, DtmcWithMultipleLocations) {
52 // The automaton moves from l0 either to l1 (probability 0.9) or to l2 (probability 0.1) and only reaches the
53 // target state s=3 via l2. All guards are trivially true, so the location is the only thing distinguishing the
54 // edges. The minimal counterexample consists of the initial edge and the edge in l2, but not the one in l1.
55 auto result = computeJaniCounterexample(STORM_TEST_RESOURCES_DIR "/dtmc/locations_counterexample.jani", "P<0.05 [ F s=3 ]");
56 ASSERT_TRUE(result.has_value());
57 ASSERT_EQ(1ull, result->getNumberOfAutomata());
58 ASSERT_EQ(2ull, getNumberOfEdges(result.value()));
59
60 storm::jani::Automaton const& automaton = result->getAutomaton(0);
61 EXPECT_EQ(automaton.getLocationIndex("l0"), automaton.getEdge(0).getSourceLocationIndex());
62 EXPECT_EQ(automaton.getLocationIndex("l2"), automaton.getEdge(1).getSourceLocationIndex());
63}

65TEST(SmtMinimalLabelSetGeneratorTest, MdpWithMultipleLocationsAndInitialLocations) {
66 // Automaton A is as in the DTMC above, but its first edge synchronizes with automaton B, which has two initial
67 // locations. As the property value is the maximum over the initial states, it suffices to keep the edge of B that
68 // is enabled in one of the two initial locations. The edge of A in l1 is not needed.
69 auto result = computeJaniCounterexample(STORM_TEST_RESOURCES_DIR "/mdp/locations_counterexample.jani", "P<0.5 [ F s=3 ]");
70 ASSERT_TRUE(result.has_value());
71
72 ASSERT_EQ(2ull, result->getNumberOfAutomata());
73 EXPECT_EQ(2ull, result->getAutomaton("A").getNumberOfEdges());
74 // Either edge of B yields a counterexample, so we only check how many of them are kept.
75 EXPECT_EQ(1ull, result->getAutomaton("B").getNumberOfEdges());
76}
