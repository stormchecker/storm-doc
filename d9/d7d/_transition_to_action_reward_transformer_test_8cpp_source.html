24template<typename TransitionValueType, typename RewardValueType>
25std::shared_ptr<storm::models::sparse::Dtmc<TransitionValueType, storm::models::sparse::StandardRewardModel<RewardValueType>>> buildDtmcWithTransitionRewards(
26 RewardValueType const& firstReward, RewardValueType const& secondReward) {
27 storm::storage::SparseMatrixBuilder<TransitionValueType> transitionBuilder(3, 3, 3);
28 transitionBuilder.addNextValue(0, 2, storm::utility::one<TransitionValueType>());
29 transitionBuilder.addNextValue(1, 2, storm::utility::one<TransitionValueType>());
30 transitionBuilder.addNextValue(2, 2, storm::utility::one<TransitionValueType>());
31
32 storm::storage::SparseMatrixBuilder<RewardValueType> rewardBuilder(3, 3, 2);
33 rewardBuilder.addNextValue(0, 2, firstReward);
34 rewardBuilder.addNextValue(1, 2, secondReward);
35
36 storm::models::sparse::StateLabeling labeling(3);
37 labeling.addLabel("init");
38 labeling.addLabelToState("init", 0);
39 storm::storage::sparse::ModelComponents<TransitionValueType, storm::models::sparse::StandardRewardModel<RewardValueType>> components(
40 transitionBuilder.build(), std::move(labeling));
41 std::optional<storm::storage::SparseMatrix<RewardValueType>> transitionRewards(rewardBuilder.build());
42 components.rewardModels.emplace("rew",
43 storm::models::sparse::StandardRewardModel<RewardValueType>(std::nullopt, std::nullopt, std::move(transitionRewards)));
44 return std::make_shared<storm::models::sparse::Dtmc<TransitionValueType, storm::models::sparse::StandardRewardModel<RewardValueType>>>(
45 std::move(components));
46}

48template<typename TransitionValueType, typename RewardValueType>
49void checkTransformedDtmc(storm::transformer::TransitionToActionRewardTransformerReturnType<
50 TransitionValueType, storm::models::sparse::StandardRewardModel<RewardValueType>> const& transformed,
51 RewardValueType const& firstReward, RewardValueType const& secondReward) {
52 ASSERT_EQ(5ull, transformed.model->getNumberOfStates());
53 EXPECT_EQ((std::vector<uint64_t>{0, 1, 4}), transformed.originalToNewStateIndices);
54
55 auto const& transitionMatrix = transformed.model->getTransitionMatrix();
56 auto checkTransition = [&transitionMatrix](uint64_t row, uint64_t column) {
57 auto const rowEntries = transitionMatrix.getRow(row);
58 ASSERT_EQ(1ull, rowEntries.getNumberOfEntries());
59 auto const& entry = *rowEntries.begin();
60 EXPECT_EQ(column, entry.getColumn());
61 EXPECT_EQ(storm::utility::one<TransitionValueType>(), entry.getValue());
62 };
63 checkTransition(0, 2);
64 checkTransition(1, 3);
65 checkTransition(2, 4);
66 checkTransition(3, 4);
67 checkTransition(4, 4);
68
69 auto const& rewardModel = transformed.model->getRewardModel("rew");
70 ASSERT_TRUE(rewardModel.hasStateActionRewards());
71 ASSERT_FALSE(rewardModel.hasTransitionRewards());
72 auto const& actionRewards = rewardModel.getStateActionRewardVector();
73 ASSERT_EQ(5ull, actionRewards.size());
74 EXPECT_EQ(firstReward, actionRewards[2]);
75 EXPECT_EQ(secondReward, actionRewards[3]);
76}
