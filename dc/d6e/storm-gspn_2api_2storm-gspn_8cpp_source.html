80 }
81 expressionParser.setIdentifierMapping(identifierMapping);
82 expressionParser.setAcceptDoubleLiterals(false);
83
84 std::unordered_map<std::string, uint64_t> map;
85
86 std::ifstream stream;
87 storm::io::openFile(filename, stream);
88
89 std::string line;
90 while (storm::io::getline(stream, line)) {
91 std::vector<std::string> strs;
92 boost::split(strs, line, boost::is_any_of("\t "));
93 STORM_LOG_THROW(strs.size() == 2, storm::exceptions::WrongFormatException, "Expect key value pairs.");
94 storm::expressions::Expression expr = expressionParser.parseFromString(strs[1]);
95 if (!gspn.getConstantsSubstitution().empty()) {
96 expr = expr.substitute(gspn.getConstantsSubstitution());
97 }
98 STORM_LOG_THROW(!expr.containsVariables(), storm::exceptions::WrongFormatException,
99 "The capacity expression '" << strs[1] << "' still contains undefined constants.");
100 map[strs[0]] = expr.evaluateAsInt();
101 }
102 storm::io::closeFile(stream);
103 return map;
