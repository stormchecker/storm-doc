19TEST_F(HashVisitorTest, SyntacticallyEqualExpressionsHashEqually) {
20 auto manager = std::make_shared<storm::expressions::ExpressionManager>();
21 auto x = manager->declareBooleanVariable("x");
22 auto p = manager->declareRationalVariable("p");
23 auto q = manager->declareRationalVariable("q");
24
25 // Each of these pairs is syntactically equal, so the hashes must agree.
26 EXPECT_EQ(hash(x && x), hash(x && x));
27 EXPECT_EQ(hash((x && !x) || x), hash((x && !x) || x));
28 EXPECT_EQ(hash(p <= q), hash(p <= q));
29 EXPECT_EQ(hash(manager->integer(3) + manager->integer(4)), hash(manager->integer(3) + manager->integer(4)));
30 EXPECT_EQ(hash(manager->rational(3.5)), hash(manager->rational(3.5)));
31 EXPECT_EQ(hash(manager->boolean(true)), hash(manager->boolean(true)));
32
33 // Syntactic equality is insensitive to expression identity: two separately built but
34 // structurally identical expressions must still hash equally.
35 auto left = (x && x);
36 auto right = (x && x);
37 ASSERT_TRUE(left.isSyntacticallyEqual(right));
38 EXPECT_EQ(hash(left), hash(right));
39}

42 auto manager = std::make_shared<storm::expressions::ExpressionManager>();

46 auto q = manager->declareRationalVariable("q");
47
48 // Distinct variables.
49 EXPECT_NE(hash(x), hash(y));
50
51 // Distinct operators over the same operands.
52 EXPECT_NE(hash(x && x), hash(x || x));
53

55 EXPECT_NE(hash(x && x), hash(x && y));
56
57 // Different literals of the same type.
58 EXPECT_NE(hash(manager->integer(3)), hash(manager->integer(4)));
59 EXPECT_NE(hash(manager->boolean(true)), hash(manager->boolean(false)));
60 EXPECT_NE(hash(manager->rational(3.5)), hash(manager->rational(4.5)));
61
62 // Different types that could otherwise coincide: an integer and a rational literal, and a
63 // variable against a literal.
64 EXPECT_NE(hash(manager->integer(2)), hash(manager->rational(2.0)));
65 EXPECT_NE(hash(x), hash(manager->boolean(true)));
66
67 // Same shape but a different relation type.
68 EXPECT_NE(hash(p <= q), hash(p >= q));
69
70 // The two arguments of a relation are not interchangeable, so the order matters.

126TEST_F(HashVisitorTest, PredicatesNestedInExpressionsAreHashed) {
127 auto manager = std::make_shared<storm::expressions::ExpressionManager>();
128 auto x = manager->declareBooleanVariable("x");
129 auto y = manager->declareBooleanVariable("y");
130 auto z = manager->declareBooleanVariable("z");
131 auto p = manager->declareRationalVariable("p");
132
133 // A predicate that is reached through the recursion instead of being the root of the traversal must
134 // be hashed as well, otherwise the hash would still reach the unimplemented base visitor here.
135 EXPECT_EQ(hash(x && storm::expressions::atLeastOneOf({x, y})), hash(x && storm::expressions::atLeastOneOf({x, y})));
136 EXPECT_EQ(hash((x && storm::expressions::atLeastOneOf({x, y})) || z), hash((x && storm::expressions::atLeastOneOf({x, y})) || z));
137
138 // The nested predicate keeps contributing its type, its operands and their order.
139 EXPECT_NE(hash(x && storm::expressions::atLeastOneOf({x, y})), hash(x && storm::expressions::atMostOneOf({x, y})));
140 EXPECT_NE(hash(x && storm::expressions::atLeastOneOf({x, y})), hash(x && storm::expressions::atLeastOneOf({y, x})));
141 EXPECT_NE(hash(x && storm::expressions::atLeastOneOf({x, y})), hash(x && storm::expressions::atLeastOneOf({x, z})));
142
143 // Higher arity.
144 EXPECT_NE(hash(x && storm::expressions::atLeastOneOf({x, y, z})), hash(x && storm::expressions::atLeastOneOf({x, y})));
145
146 // A predicate nested inside a predicate is not expressible, because the factories only accept plain boolean
147 // operands. What matters here is that the operands are hashed recursively, which the arithmetic operand
148 // already covers.
149 EXPECT_EQ(hash(storm::expressions::atLeastOneOf({x && (p <= manager->integer(1)), y})),
150 hash(storm::expressions::atLeastOneOf({x && (p <= manager->integer(1)), y})));
151 EXPECT_NE(hash(storm::expressions::atLeastOneOf({x && (p <= manager->integer(1)), y})),
152 hash(storm::expressions::atLeastOneOf({x && (p <= manager->integer(2)), y})));
153}
