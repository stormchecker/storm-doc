25void testGoalStateMerger(std::string const& prismModelFile, std::string const& formulaString) {
26 storm::prism::Program program = storm::parser::PrismParser::parse(prismModelFile, true);
27 auto formulas = storm::api::extractFormulasFromProperties(storm::api::parsePropertiesForPrismProgram(formulaString, program));
28 auto model = storm::api::buildSparseModel<double>(program, formulas);
29 auto task = storm::api::createTask<double>(formulas.front(), false);
30 ASSERT_EQ(1ull, model->getInitialStates().getNumberOfSetBits()) << "For model " << prismModelFile << ".";
31
32 auto computeValue = [&task](auto const& m) {
33 auto result = storm::api::verifyWithSparseEngine(m, task);
34 return result->template asExplicitQuantitativeCheckResult<double>()[*m->getInitialStates().begin()];
35 };
36 auto const modelVal = computeValue(model);
37
38 storm::transformer::GoalStateMerger<double> merger(*model);
39 for (bool dropUnreachableFromInit : {true, false}) {
40 auto mergerResult = merger.mergeForFormula(*formulas.front(), dropUnreachableFromInit);
41 ASSERT_TRUE(mergerResult.has_value()) << "Merging goal states failed for model " << prismModelFile << " and formula " << *formulas.front()
42 << " (dropUnreachableFromInit=" << dropUnreachableFromInit << ").";
43 ASSERT_EQ(1ull, mergerResult->model->getInitialStates().getNumberOfSetBits())
44 << "For merged model of " << prismModelFile << " (dropUnreachableFromInit=" << dropUnreachableFromInit << ").";
45
46 // The original (unmodified) formula must still be valid on the merged model and induce the same value at the initial state.
47 auto const mergedVal = computeValue(mergerResult->model);
48 bool const relativeDifferenceOk = std::abs(mergedVal - modelVal) <= 1e-6 * std::abs(modelVal);
49 EXPECT_TRUE(relativeDifferenceOk) << "Relative difference between original model result (" << modelVal << ") and merged model result (" << mergedVal
50 << ") is too high.\nFailed for model " << prismModelFile << " and formula " << *formulas.front()
51 << " (dropUnreachableFromInit=" << dropUnreachableFromInit << ").";
52 }
53}
