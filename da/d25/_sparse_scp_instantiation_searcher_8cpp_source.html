31 storm::Environment const& env, std::shared_ptr<storm::pars::FeasibilitySynthesisTask const> const& task) {
32 // Also validated here because ScpOptions can be constructed by library callers.
33 STORM_LOG_THROW(options.trustRegionInitial > 1.0, storm::exceptions::NotSupportedException, "ScpOptions::trustRegionInitial must be > 1.");
34 STORM_LOG_THROW(options.trustRegionFactor > 1.0, storm::exceptions::NotSupportedException, "ScpOptions::trustRegionFactor must be > 1.");
35 STORM_LOG_THROW(options.trustRegionMax > 1.0, storm::exceptions::NotSupportedException, "ScpOptions::trustRegionMax must be > 1.");
36 STORM_LOG_THROW(options.trustRegionMinExcess >= 0.0, storm::exceptions::NotSupportedException, "ScpOptions::trustRegionMinExcess must be >= 0.");
37 STORM_LOG_THROW(options.maxIterations >= 1, storm::exceptions::NotSupportedException, "ScpOptions::maxIterations must be >= 1.");
38
39 STORM_LOG_THROW(task->getFormula().isProbabilityOperatorFormula() || task->getFormula().isRewardOperatorFormula(), storm::exceptions::NotSupportedException,
40 "SCP currently only supports probability- and reward operator formulas.");
41 bool const isReward = task->getFormula().isRewardOperatorFormula();
42 STORM_LOG_THROW(task->getFormula().asOperatorFormula().getSubformula().isEventuallyFormula(), storm::exceptions::NotSupportedException,
43 "SCP currently only supports unbounded reachability (F) properties.");
44 STORM_LOG_THROW(task->isBoundSet(), storm::exceptions::NotSupportedException, "SCP requires an explicitly given bound.");
45
46 storm::logic::Bound const& bound = task->getBound();
47 ThresholdDirection direction = bound.isLowerBound() ? ThresholdDirection::AtLeast : ThresholdDirection::AtMost;
48 double threshold = bound.threshold.evaluateAsDouble();
49
50 // Robust reading: AtLeast is checked against Pmin/Rmin, AtMost against Pmax/Rmax.
51 storm::solver::OptimizationDirection schedulerDirection =
52 (direction == ThresholdDirection::AtLeast) ? storm::solver::OptimizationDirection::Minimize : storm::solver::OptimizationDirection::Maximize;
53
54 // Target states, via a plain propositional check of the (label-based) subformula of "F ...".
55 auto const& targetSubformula = task->getFormula().asOperatorFormula().getSubformula().asEventuallyFormula().getSubformula();
56 storm::modelchecker::SparsePropositionalModelChecker<SparseModelType> propositionalChecker(model);
57 storm::modelchecker::CheckTask<storm::logic::Formula, ParametricType> targetCheckTask(targetSubformula);
58 storm::storage::BitVector targetStates =
59 propositionalChecker.check(targetCheckTask)->template asExplicitQualitativeCheckResult<ParametricType>().getTruthValuesVector();
60 storm::storage::BitVector allStates(model.getNumberOfStates(), true);
61
62 // Probability property: zeroStates/oneStates are Prob0/Prob1. Reward property: zeroStates are the
63 // target states, oneStates stays empty, and infiniteStates have infinite expected reward. All sets
64 // are computed for schedulerDirection.
65 storm::storage::BitVector zeroStates, oneStates, infiniteStates;
66 storm::storage::BitVector excludedRows(model.getTransitionMatrix().getRowCount(), false);
67 if constexpr (std::is_same_v<SparseModelType, storm::models::sparse::Mdp<ParametricType>>) {
68 if (isReward) {
69 zeroStates = targetStates;
70 oneStates = storm::storage::BitVector(model.getNumberOfStates(), false);
71 infiniteStates = storm::solver::minimize(schedulerDirection)
72 ? storm::utility::graph::performProb1E(model.getTransitionMatrix(), model.getTransitionMatrix().getRowGroupIndices(),
73 model.getBackwardTransitions(), allStates, targetStates)
74 : storm::utility::graph::performProb1A(model.getTransitionMatrix(), model.getTransitionMatrix().getRowGroupIndices(),
75 model.getBackwardTransitions(), allStates, targetStates);
76 infiniteStates.complement();
77 // A state with finite value can still have actions leading into an infinite-value state;
78 // those rows are excluded individually.
79 storm::storage::BitVector maybeStates = ~(zeroStates | infiniteStates);
80 excludedRows = ~model.getTransitionMatrix().getRowFilter(maybeStates, ~infiniteStates);
81 } else {
82 auto probs = storm::solver::maximize(schedulerDirection) ? storm::utility::graph::performProb01Max(model, allStates, targetStates)
83 : storm::utility::graph::performProb01Min(model, allStates, targetStates);
84 zeroStates = std::move(probs.first);
85 oneStates = std::move(probs.second);
86 infiniteStates = storm::storage::BitVector(model.getNumberOfStates(), false);
87 }
88 } else {
89 static_assert(std::is_same_v<SparseModelType, storm::models::sparse::Dtmc<ParametricType>>,
90 "SparseScpInstantiationSearcher only supports Mdp and Dtmc models.");
91 if (isReward) {
92 zeroStates = targetStates;
93 oneStates = storm::storage::BitVector(model.getNumberOfStates(), false);
94 infiniteStates = storm::utility::graph::performProb1(model.getBackwardTransitions(), allStates, targetStates);
95 infiniteStates.complement();
96 // A DTMC state with an infinite successor is itself infinite, so no row exclusion is needed.
97 } else {
98 auto probs = storm::utility::graph::performProb01(model, allStates, targetStates);
99 zeroStates = std::move(probs.first);
100 oneStates = std::move(probs.second);
101 infiniteStates = storm::storage::BitVector(model.getNumberOfStates(), false);
102 }
103 }
104
105 storm::models::sparse::StandardRewardModel<ParametricType> const* rewardModel = nullptr;
106 if (isReward) {
107 auto const& rewardFormula = task->getFormula().asRewardOperatorFormula();
108 if (rewardFormula.hasRewardModelName()) {
109 STORM_LOG_THROW(model.hasRewardModel(rewardFormula.getRewardModelName()), storm::exceptions::NotSupportedException,
110 "The model has no reward model named '" << rewardFormula.getRewardModelName() << "'.");
111 rewardModel = &model.getRewardModel(rewardFormula.getRewardModelName());
112 } else {
113 STORM_LOG_THROW(model.hasUniqueRewardModel(), storm::exceptions::NotSupportedException,
114 "The reward formula does not name a reward model, and the model does not have a unique reward model either.");
115 rewardModel = &model.getUniqueRewardModel();
116 }
117 }
118
119 AffinePMdpDecomposition decomposition(model, rewardModel);
120
121 // Use a real region in both cases: the caller's, or [0,1]^n.
122 storm::storage::ParameterRegion<ParametricType> region;
123 if (task->isRegionSet()) {
124 region = task->getRegion();
125 for (auto const& var : decomposition.getParameters()) {
126 STORM_LOG_THROW(region.getVariables().count(var) > 0, storm::exceptions::NotSupportedException,
127 "The given region does not cover parameter '" << var << "', which occurs in the model.");
128 }
129 } else {
130 Valuation lower, upper;
131 for (auto const& var : decomposition.getParameters()) {
132 lower[var] = storm::utility::zero<CoefficientType>();
133 upper[var] = storm::utility::one<CoefficientType>();
134 }
135 region = storm::storage::ParameterRegion<ParametricType>(std::move(lower), std::move(upper));
136 }
137
138 std::map<VariableType, std::pair<double, double>> parameterBounds;
139 for (auto const& var : decomposition.getParameters()) {
140 parameterBounds[var] = {storm::utility::convertNumber<double>(region.getLowerBoundary(var)),
141 storm::utility::convertNumber<double>(region.getUpperBoundary(var))};
142 }
143 Valuation initialValuation = region.getCenterPoint();
144
145 STORM_LOG_THROW(model.getInitialStates().getNumberOfSetBits() == 1, storm::exceptions::NotSupportedException,
146 "SCP requires a model with a unique initial state.");
147 uint64_t initialState = model.getInitialStates().getNextSetIndex(0);
148
149 if (zeroStates.get(initialState) || oneStates.get(initialState) || infiniteStates.get(initialState)) {
150 // The initial state's value is fixed by the graph structure; ScpLpEncoder requires a state
151 // with an unknown value.
152 storm::utility::ExtendedValueType<ConstantType> constantValue = oneStates.get(initialState) ? storm::utility::one<ConstantType>()
153 : infiniteStates.get(initialState) ? storm::utility::positiveInfinity<ConstantType>()
154 : storm::utility::zero<ConstantType>();
155 return ScpResult<ConstantType>{{constantValue, initialValuation}, bound.isSatisfied(constantValue) ? ScpStatus::Feasible : ScpStatus::NotFound, 0};
156 }
157
158 std::vector<uint64_t> rowGroupIndices(model.getTransitionMatrix().getRowGroupIndices().begin(), model.getTransitionMatrix().getRowGroupIndices().end());
159 PropertyKind propertyKind = isReward ? PropertyKind::Reward : PropertyKind::Probability;
160 ScpLpEncoder encoder(decomposition, rowGroupIndices, zeroStates, oneStates, infiniteStates, excludedRows, initialState, parameterBounds, threshold,
161 direction, propertyKind, options);
162
163 storm::modelchecker::CheckTask<storm::logic::Formula, ParametricType> checkTask(task->getFormula(), /*onlyInitialStatesRelevant=*/false);
164
165 if constexpr (std::is_same_v<SparseModelType, storm::models::sparse::Mdp<ParametricType>>) {
166 if (checkTask.isOptimizationDirectionSet()) {
167 // An explicit Pmax/Pmin/Rmax/Rmin must agree with the direction implied by the threshold.
168 STORM_LOG_THROW(checkTask.getOptimizationDirection() == schedulerDirection, storm::exceptions::NotSupportedException,
169 "SCP requires the formula's optimization direction (if explicitly given, e.g. Pmax=?/Pmin=?) to match the "
170 "robust reading implied by the threshold direction (AtLeast -> Pmin, AtMost -> Pmax); mismatched combinations "
171 "like 'Pmax >= lambda' are not supported.");
172 } else {
173 checkTask.setOptimizationDirection(schedulerDirection);
174 }
175 storm::modelchecker::SparseMdpInstantiationModelChecker<SparseModelType, ConstantType> instantiationChecker(env, model, /*produceScheduler=*/false);
176 instantiationChecker.setInstantiationsAreGraphPreserving(true);
177 instantiationChecker.specifyFormula(checkTask);
178 return runLoop(env, instantiationChecker, encoder, bound, direction, initialState, initialValuation);
179 } else {
180 storm::modelchecker::SparseDtmcInstantiationModelChecker<SparseModelType, ConstantType> instantiationChecker(env, model);
181 instantiationChecker.setInstantiationsAreGraphPreserving(true);
182 instantiationChecker.specifyFormula(checkTask);
183 return runLoop(env, instantiationChecker, encoder, bound, direction, initialState, initialValuation);
184 }

187template<typename SparseModelType, typename ConstantType>
188template<typename InstantiationChecker>
189ScpResult<ConstantType> SparseScpInstantiationSearcher<SparseModelType, ConstantType>::runLoop(storm::Environment const& env,
190 InstantiationChecker& instantiationChecker,
191 ScpLpEncoder const& encoder, storm::logic::Bound const& bound,
192 ThresholdDirection direction, uint64_t initialState,
193 Valuation const& initialValuation) const {
194 // Returns the model-checked value vector at ConstantType precision. Only the LP below uses double.
195 auto modelCheck = [&](Valuation const& valuation) -> std::vector<ConstantType> {
196 auto result = instantiationChecker.check(env, valuation);
197 return result->template asExplicitQuantitativeCheckResult<ConstantType>().getFiniteValueVector();
198 };
199 auto toDouble = [](Valuation const& v) -> std::map<VariableType, double> {
200 std::map<VariableType, double> result;
201 for (auto const& [var, val] : v) {
202 result[var] = storm::utility::convertNumber<double>(val);
203 }
204 return result;
205 };
206 auto toDoubleVec = [](std::vector<ConstantType> const& v) -> std::vector<double> {
207 std::vector<double> result(v.size());
208 for (uint64_t i = 0; i < v.size(); ++i) {
209 result[i] = storm::utility::convertNumber<double>(v[i]);
210 }
211 return result;
212 };
213
214 Valuation vHat = initialValuation;
215 std::vector<ConstantType> pHat = modelCheck(vHat);
216 ConstantType betaStar = pHat.at(initialState);
217
218 ScpResult<ConstantType> best{{betaStar, vHat}, ScpStatus::NotFound, 0};
219 if (bound.isSatisfied(betaStar)) {
220 best.status = ScpStatus::Feasible;
221 return best;
222 }
223
224 double delta = options.trustRegionInitial;
225
226 uint64_t iteration = 0;
227 for (; iteration < options.maxIterations; ++iteration) {
228 // Stops on --timeout, Ctrl-C and other termination requests.
229 if (storm::utility::resources::isTerminate()) {
230 // best.iterations keeps the iteration at which `best` was established.
231 best.status = ScpStatus::Timeout;
232 return best;
233 }
234 if (delta - 1.0 <= options.trustRegionMinExcess) {
235 best.status = ScpStatus::NotFound;
236 return best;
237 }
238
239 EncodedLp lp = encoder.encode(env, toDouble(vHat), toDoubleVec(pHat), delta);
240 lp.solver->optimize();
241 if (!lp.solver->isOptimal()) {
242 delta = (delta - 1.0) / options.trustRegionFactor + 1.0;
243 continue;
244 }
245
246 Valuation candidate = initialValuation;
247 for (auto const& [var, lpVar] : lp.parameterVariables) {
248 candidate[var] = storm::utility::convertNumber<CoefficientType>(lp.solver->getContinuousValue(lpVar));
249 }
250
251 std::vector<ConstantType> valueVector = modelCheck(candidate);
252 ConstantType beta = valueVector.at(initialState);
253
254 if (bound.isSatisfied(beta)) {
255 return ScpResult<ConstantType>{{beta, candidate}, ScpStatus::Feasible, iteration + 1};
256 }
257
258 bool improved = (direction == ThresholdDirection::AtLeast) ? (beta > betaStar) : (beta < betaStar);
259 if (improved) {
260 vHat = candidate;
261 pHat = std::move(valueVector);
262 betaStar = beta;
263 best = ScpResult<ConstantType>{{betaStar, vHat}, ScpStatus::NotFound, iteration + 1};
264 delta = std::min(options.trustRegionMax, (delta - 1.0) * options.trustRegionFactor + 1.0);
265 } else {
266 delta = (delta - 1.0) / options.trustRegionFactor + 1.0;
267 }
268 }
269
270 best.status = ScpStatus::MaxIterationsReached;
271 return best;
272}
