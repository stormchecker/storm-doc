36template<typename ValueType>
37bool ReachabilityRewardCertificate<ValueType>::checkValidity(storm::models::Model<ValueType> const& model) const {
38 if (!model.isSparseModel()) {
39 STORM_LOG_WARN("Certificate invalid because the given model is not sparse.");
40 return false;
41 }
42 if (model.isOfType(storm::models::ModelType::Ctmc)) {
43 if (auto const* ctmcPtr = dynamic_cast<storm::models::sparse::Ctmc<ValueType> const*>(&model)) {
44 return checkValidity(ctmcPtr->computeProbabilityMatrix());
45 }
46 } else if (model.isOfType(storm::models::ModelType::Dtmc) || model.isOfType(storm::models::ModelType::Mdp) ||
47 model.isOfType(storm::models::ModelType::MarkovAutomaton)) {
48 if (auto const* sparseModelPtr = dynamic_cast<storm::models::sparse::Model<ValueType> const*>(&model)) {
49 return checkValidity(sparseModelPtr->getTransitionMatrix());
50 }
51 }
52 STORM_LOG_WARN("Certificate invalid because the given model is of unsupported type.");
53 return false;
54}

56template<typename ValueType, OptimizationDirection Dir>
57bool checkUpperBoundCertificate(storm::storage::SparseMatrix<ValueType> const& transitionProbabilityMatrix, storm::storage::BitVector const& targetStates,
58 std::vector<ValueType> const& stateActionRewardVector, std::vector<storm::utility::ExtendedValueType<ValueType>> const& values,
59 std::vector<RankingType> const& ranks) {
60 for (uint64_t state = 0; state < targetStates.size(); ++state) {
61 if (storm::utility::isNegativeInfinity(values[state])) {
62 STORM_LOG_WARN("Certificate invalid because state " << state << " has negative infinity as value.");
63 return false;
64 }
65 // Handle states that do not almost surely reach target
66 if (storm::utility::isFinite(values[state]) && ranks[state] == InfRank) {
67 STORM_LOG_WARN("Certificate invalid because state " << state << " has infinite rank but finite value " << values[state] << ".");
68 return false;
69 }
70 if (targetStates.get(state)) {
71 // Handle target state
72 if (values[state] < storm::utility::zero<ValueType>()) {
73 STORM_LOG_WARN("Certificate invalid because target state " << state << " has negative upper bound " << values[state]);
74 return false;
75 }
76 // The rank of a target state must be non-negative, too.
77 // However, a negative rank (which is an unsigned integer) is not possible, anyway.
78 // So, we skip checking for the rank
79 continue;
80 }
81
82 // Apply Bellman and Distance operator for non-target states
83 // Following Propositions 8 and 9 of https://doi.org/10.1007/978-3-031-90653-4_7 (see also the arXiv version), the rank only considers
84 // the x-decreasing (i.e. inductive) choices if we minimize and all choices if we maximize.
85 auto stateValue = storm::solver::maximize(Dir) ? storm::utility::negativeInfinity<ValueType>() : storm::utility::positiveInfinity<ValueType>();
86 storm::utility::Extremum<Dir, RankingType> stateRankMinusOne;
87 for (auto choice : transitionProbabilityMatrix.getRowGroupIndices(state)) {
88 auto const currentChoiceValue =
89 detail::multiplyRowWithVectorAddNumber(transitionProbabilityMatrix, choice, values, stateActionRewardVector[choice]);
90 stateValue = storm::solver::maximize(Dir) ? std::max(stateValue, currentChoiceValue) : std::min(stateValue, currentChoiceValue);
91 // Compute rank for this choice if we're maximizing or if the choice is inductive
92 if (storm::solver::maximize(Dir) || currentChoiceValue <= values[state]) {
93 storm::utility::Minimum<RankingType> choiceRank;
94 for (auto const& entry : transitionProbabilityMatrix.getRow(choice)) {
95 if (storm::utility::isZero(entry.getValue())) {
96 continue;
97 }
98 choiceRank &= ranks[entry.getColumn()];
99 }
100 STORM_LOG_ASSERT(!choiceRank.empty(),
101 "Ranking operator failed to compute rank for state " << state << " since the row " << choice << " is empty.");
102 stateRankMinusOne &= *choiceRank;
103 }
104 }
105 if (values[state] < stateValue) {
106 double const approxDiff = storm::utility::convertNumber<double>(stateValue - values[state]);
107 STORM_LOG_WARN("Certificate invalid because upper bound is not inductive. At state " << state << " the Bellman operator yields " << stateValue
108 << " but the certificate has smaller value " << values[state]
109 << ". Approx. diff is " << approxDiff << ".");
110 return false;
111 }
112 if (stateRankMinusOne.empty()) {
113 STORM_LOG_WARN("Certificate invalid because there is no valid (i.e. decreasing) action at state " << state << ", where the certificate has rank "
114 << ranks[state] << ".");
115 return false;
116 }
117 RankingType const stateRank = *stateRankMinusOne == InfRank ? InfRank : *stateRankMinusOne + 1; // avoid integer overflow!
118 if (stateRank > ranks[state]) {
119 STORM_LOG_WARN("Certificate invalid because ranks for upper bound is not inductive. At state "
120 << state << " the rank operator yields " << stateRank << " but the certificate has smaller rank " << ranks[state] << ".");
121 return false;
122 }
123 }
124 return true;
125}

127template<typename ValueType, OptimizationDirection Dir>
128bool checkLowerBoundCertificate(storm::storage::SparseMatrix<ValueType> const& transitionProbabilityMatrix, storm::storage::BitVector const& targetStates,
129 std::vector<ValueType> const& stateActionRewardVector, std::vector<storm::utility::ExtendedValueType<ValueType>> const& values,
130 std::vector<RankingType> const& ranks) {
131 for (uint64_t state = 0; state < targetStates.size(); ++state) {
132 // Handle target state
133 if (targetStates.get(state)) {
134 if (!storm::utility::isFinite(values[state]) || !storm::utility::isZero(storm::utility::getFinite(values[state]))) {
135 STORM_LOG_WARN("Certificate invalid because target state " << state << " has non-zero lower bound " << values[state] << ".");
136 return false;
137 }
138 if (ranks[state] != InfRank) {
139 STORM_LOG_WARN("Certificate invalid because target state " << state << " has finite rank" << ranks[state] << ".");
140 return false;
141 }
142 continue;
143 }
144 if (storm::utility::isInfinity(values[state]) && ranks[state] == InfRank) {
145 STORM_LOG_WARN("Certificate invalid because state " << state << " has infinite rank and value.");
146 return false;
147 }
148 // Apply ranking- and Bellman operators for non-target states
149 // Following Proposition 7 of https://doi.org/10.1007/978-3-031-90653-4_7 (see also the arXiv version), the (complementary distance) rank
150 // considers all choices, no matter whether we minimize or maximize. Unlike for lower bounds on maximal reachability probabilities, there is no
151 // restriction to inductive choices.
152 auto stateValue = storm::solver::maximize(Dir) ? storm::utility::negativeInfinity<ValueType>() : storm::utility::positiveInfinity<ValueType>();
153 storm::utility::Extremum<storm::solver::invert(Dir), RankingType> stateRank;
154
155 for (auto choice : transitionProbabilityMatrix.getRowGroupIndices(state)) {
156 auto const currentChoiceValue =
157 detail::multiplyRowWithVectorAddNumber(transitionProbabilityMatrix, choice, values, stateActionRewardVector[choice]);
158 stateValue = storm::solver::maximize(Dir) ? std::max(stateValue, currentChoiceValue) : std::min(stateValue, currentChoiceValue);
159 // Compute rank for this choice
160 storm::utility::Minimum<RankingType> minSuccessorRank;
161 bool allSuccessorsEqual = true;
162 for (auto const& entry : transitionProbabilityMatrix.getRow(choice)) {
163 if (storm::utility::isZero(entry.getValue())) {
164 continue;
165 }
166 if (!minSuccessorRank.empty() && ranks[entry.getColumn()] != *minSuccessorRank) {
167 allSuccessorsEqual = false;
168 }
169 minSuccessorRank &= ranks[entry.getColumn()];
170 }
171 STORM_LOG_ASSERT(!minSuccessorRank.empty(),
172 "Ranking operator failed to compute rank for state " << state << " since the row " << choice << " is empty.");
173 auto choiceRank = *minSuccessorRank;
174 if (choiceRank != InfRank && !allSuccessorsEqual) {
175 ++choiceRank;
176 }
177 stateRank &= choiceRank;
178 }
179
180 if (stateValue < values[state]) {
181 double const approxDiff = storm::utility::convertNumber<double>(values[state] - stateValue);
182 STORM_LOG_WARN("Certificate invalid because lower bound is not inductive. At state " << state << " the Bellman operator yields " << stateValue
183 << " but the certificate has larger value " << values[state]
184 << ". Approx. diff is " << approxDiff << ".");
185 return false;
186 }
187 STORM_LOG_ASSERT(!stateRank.empty(), "Failed to compute ranks for state " << state << " since the row group is empty.");
188 if (*stateRank > ranks[state]) {
189 STORM_LOG_WARN("Certificate invalid because ranks for lower bound is not inductive. At state "
190 << state << " the rank operator yields " << *stateRank << " but the certificate has smaller rank " << ranks[state] << ".");
191 return false;
192 }
193 }
194 return true;
195}

197template<typename ValueType>
198bool ReachabilityRewardCertificate<ValueType>::checkValidity(storm::storage::SparseMatrix<ValueType> const& transitionProbabilityMatrix) const {
199 if (!dir.has_value() && !transitionProbabilityMatrix.hasTrivialRowGrouping()) {
200 STORM_LOG_WARN("Certificate invalid because Matrix has non-trivial row grouping but no optimization direction given.");
201 return false;
202 }
203 if (transitionProbabilityMatrix.getRowGroupCount() != targetStates.size() || transitionProbabilityMatrix.getColumnCount() != targetStates.size()) {
204 STORM_LOG_WARN("Certificate invalid because Matrix has invalid dimensions (target states).");
205 return false;
206 }
207 if (transitionProbabilityMatrix.getRowCount() != stateActionRewardVector.size()) {
208 STORM_LOG_WARN("Certificate invalid because Matrix has invalid dimensions (reward vector).");
209 return false;
210 }
211
212 if (dir.has_value() && storm::solver::maximize(*dir)) {
213 return checkValidityInternal<storm::OptimizationDirection::Maximize>(transitionProbabilityMatrix);
214 } else {
215 return checkValidityInternal<storm::OptimizationDirection::Minimize>(transitionProbabilityMatrix);
216 }
217}

220template<storm::OptimizationDirection Dir>
221bool ReachabilityRewardCertificate<ValueType>::checkValidityInternal(storm::storage::SparseMatrix<ValueType> const& transitionProbabilityMatrix) const {
222 if (hasUpperBoundsCertificate() && !checkUpperBoundCertificate<ValueType, Dir>(transitionProbabilityMatrix, targetStates, stateActionRewardVector,
223 upperBoundsCertificate.values, upperBoundsCertificate.ranks)) {
224 STORM_LOG_WARN("Certificate invalid because upper bound certificate is violated.");
225 return false;
226 }
227 if (hasLowerBoundsCertificate() && !checkLowerBoundCertificate<ValueType, Dir>(transitionProbabilityMatrix, targetStates, stateActionRewardVector,
228 lowerBoundsCertificate.values, lowerBoundsCertificate.ranks)) {
229 STORM_LOG_WARN("Certificate invalid because lower bound certificate is violated.");
230 return false;
231 }
232 return true;

256template<typename ValueType>
257std::string ReachabilityRewardCertificate<ValueType>::summaryString(storm::storage::BitVector const& relevantStates) const {
258 std::stringstream ss;
259 auto const numStates = relevantStates.getNumberOfSetBits();
260 STORM_LOG_THROW(numStates > 0, storm::exceptions::InvalidOperationException,
261 "Tried to summarize reachability reward certificate but no state is relevant.");
262
263 auto getLowerUpperBound = [this](uint64_t state) -> std::pair<ExtendedValueType, ExtendedValueType> {
264 return {hasLowerBoundsCertificate() ? lowerBoundsCertificate.values[state] : storm::utility::negativeInfinity<ValueType>(),
265 hasUpperBoundsCertificate() ? upperBoundsCertificate.values[state] : storm::utility::positiveInfinity<ValueType>()};
266 };
267 auto const [lowerBound, upperBound] = getLowerUpperBound(relevantStates.getNextSetIndex(0));
268 ss << "Certified reachability reward: [" << lowerBound << ", " << upperBound << "]";
269 if (storm::NumberTraits<ValueType>::IsExact) {
270 ss << " (approx. [" << storm::utility::convertNumber<double>(lowerBound) << ", " << storm::utility::convertNumber<double>(upperBound) << "])";
271 }
272 if (numStates > 1) {
273 ss << " (only showing first of " << numStates << " relevant states)";
274 }
275 ExtendedValueType const zero(storm::utility::zero<ValueType>());
276 ExtendedValueType maxAbsDiff(zero), maxRelDiff(zero);
277 for (auto const state : relevantStates) {
278 auto const [lower, upper] = getLowerUpperBound(state);
279 ExtendedValueType absDiff = (upper == lower) ? zero : (upper - lower); // inf-inf undefined!
280 ExtendedValueType relDiff;
281 if (absDiff == zero) {
282 relDiff = zero;
283 } else if (lower == zero || storm::utility::isInfinity(absDiff)) {
284 relDiff = storm::utility::positiveInfinity<ValueType>();
285 } else {
286 ValueType relDiffValue = storm::utility::getFinite(absDiff) / storm::utility::getFinite(lower);
287 relDiff = ExtendedValueType(relDiffValue < storm::utility::zero<ValueType>() ? -relDiffValue : relDiffValue);
288 }
289 maxAbsDiff = std::max(maxAbsDiff, absDiff);
290 maxRelDiff = std::max(maxRelDiff, relDiff);
291 }
292 auto printIntervalWidth([&](auto const& value, auto const& name) {
293 ss << "\n\t" << name << " interval width: " << value;
294 if (storm::NumberTraits<ValueType>::IsExact) {
295 ss << " (approx. " << storm::utility::convertNumber<double>(value) << ")";
296 }
297 if (numStates > 1) {
298 ss << " (showing maximum over " << numStates << " relevant states)";
299 }
300 });
301 printIntervalWidth(maxAbsDiff, "Absolute");
302 printIntervalWidth(maxRelDiff, "Relative");
303 return ss.str();
304}
