15// Creates variables with a random layout: Variables are placed consecutively (so many of them cross bucket boundaries)
16storm::generator::VariableInformation createRandomLayout(storm::expressions::ExpressionManager& manager, std::mt19937_64& rng, uint64_t targetNumberOfBits) {
17 storm::generator::VariableInformation info;
18 uint64_t offset = 0;
19 uint64_t index = 0;
20 while (offset < targetNumberOfBits) {
21 auto const name = "v" + std::to_string(index++);
22 switch (rng() % 3) {
23 case 0: {
24 info.booleanVariables.emplace_back(manager.declareBooleanVariable(name), offset, false, true);
25 offset += 1;
26 break;
27 }
28 case 1: {
29 // Width 0 is possible for locations with a single value.
30 uint64_t const width = rng() % 9;
31 info.locationVariables.emplace_back(manager.declareIntegerVariable(name), (uint64_t(1) << width) - 1, offset, width, true);
32 offset += width;
33 break;
34 }
35 default: {
36 // Widths up to 52 bits are exactly representable by the (double based) evaluator
37 uint64_t const width = 1 + rng() % 52;
38 int64_t const lowerBound = static_cast<int64_t>(rng() % 20) - 10;
39 info.integerVariables.emplace_back(manager.declareIntegerVariable(name), lowerBound,
40 lowerBound + static_cast<int64_t>((uint64_t(1) << width) - 1), offset, width);
41 offset += width;
42 break;
43 }
44 }
45 }
46 info.totalBitOffset = offset;
47 return info;
48}

58// Makes a state that agrees with the given one except for a few random bits
59storm::generator::CompressedState perturbState(storm::generator::CompressedState state, std::mt19937_64& rng, uint64_t numberOfFlips) {
60 for (uint64_t i = 0; i < numberOfFlips; ++i) {
61 uint64_t const bit = rng() % state.size();
62 state.set(bit, !state.get(bit));
63 }
64 return state;
65}
