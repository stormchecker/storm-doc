168 std::unordered_map<BeliefObservationType, BeliefValueType> computeSuccessorObservations(BeliefType const& belief, uint64_t localActionIndex,
169 BeliefActionObservationType actionObservation) {
170 std::unordered_map<BeliefObservationType, BeliefValueType> successorObservations;
171 belief.forEach([&localActionIndex, &successorObservations, &actionObservation, this](BeliefStateType const& state, BeliefValueType const& beliefValue) {
172 for (auto const& pomdpTransition : pomdp.getTransitionMatrix().getRow(state, localActionIndex)) {
173 if (!storm::utility::isZero(pomdpTransition.getValue())) {
174 auto const obs = (pomdp.getNrObservations() * actionObservation) + pomdp.getObservation(pomdpTransition.getColumn());
175 STORM_LOG_ASSERT(obs < InvalidObservation, "Encoded belief observation " << obs << " exceeds the valid observation range.");
176 BeliefValueType const val = beliefValue * storm::utility::convertNumber<BeliefValueType>(pomdpTransition.getValue());
177 if (auto [insertionIt, inserted] = successorObservations.emplace(static_cast<BeliefObservationType>(obs), val); !inserted) {
178 insertionIt->second += val;
179 }
180 }
181 }
182 });
183
184 // Adjust the distribution to diminish numerical inaccuracies a bit
185 if constexpr (!storm::NumberTraits<BeliefValueType>::IsExact || !storm::NumberTraits<typename PomdpType::ValueType>::IsExact) {
186 if (successorObservations.size() == 1) {
187 successorObservations.begin()->second = storm::utility::one<BeliefValueType>();
188 }
189 }
190 return successorObservations;
191 }

193 template<typename... CallBackArgs>
194 void computeSuccessorBeliefs(BeliefType const& belief, uint64_t localActionIndex, BeliefValueType const& transitionProbability,
195 BeliefActionObservationType actionObservation, CallBackArgs const&... additionalCallbackArgs) {
196 // For each successor observation we build the successor belief
197 auto const successorObservations = computeSuccessorObservations(belief, localActionIndex, actionObservation);
198 for (auto const& successorObsValue : successorObservations) {
199 BeliefBuilder<BeliefType> builder;
200 builder.setObservation(successorObsValue.first);
201 belief.forEach([&builder, &localActionIndex, &successorObsValue, this](BeliefStateType const& state, BeliefValueType const& beliefValue) {
202 for (auto const& pomdpTransition : pomdp.getTransitionMatrix().getRow(state, localActionIndex)) {
203 if (!storm::utility::isZero(pomdpTransition.getValue()) &&
204 pomdp.getObservation(pomdpTransition.getColumn()) == (successorObsValue.first % pomdp.getNrObservations())) {
205 BeliefValueType const prob =
206 beliefValue * storm::utility::convertNumber<BeliefValueType>(pomdpTransition.getValue()) / successorObsValue.second;
207 STORM_LOG_ASSERT(prob > storm::utility::zero<BeliefValueType>(), "Invalid belief probability " << prob << ".");
208 builder.addValue(pomdpTransition.getColumn(), prob);
209 }
210 }
211 });
212 if constexpr (!storm::NumberTraits<BeliefValueType>::IsExact || !storm::NumberTraits<typename PomdpType::ValueType>::IsExact) {
213 if (successorObservations.size() == 1) {
214 // The observation probability was forced to one, so normalize the successor belief separately.
215 builder.normalize();
216 }
217 }
218 applyPostAbstraction(builder.build(), static_cast<BeliefValueType>(successorObsValue.second * transitionProbability),
219 std::forward<CallBackArgs const>(additionalCallbackArgs)...);
220 }
221 }
222
223 template<typename... CallBackArgs>
224 void applyPostAbstraction(BeliefType&& belief, BeliefValueType&& transitionProbability, CallBackArgs const&... additionalCallbackArgs) {
225 if constexpr (isNoAbstraction<PostAbstractionType>) {
226 discoverCallback(std::move(belief), std::move(transitionProbability), additionalCallbackArgs...);
227 } else {
228 postAbstraction.abstract(
229 std::move(belief), std::move(transitionProbability),
230 [this, &additionalCallbackArgs...](BeliefType&& postBel, BeliefValueType&& postVal, auto&&... additionalPostAbstractionArgs) {
231 discoverCallback(std::move(postBel), std::move(postVal), std::forward<CallBackArgs const>(additionalCallbackArgs)...,
232 std::forward<decltype(additionalPostAbstractionArgs)>(additionalPostAbstractionArgs)...);
233 });
234 }
