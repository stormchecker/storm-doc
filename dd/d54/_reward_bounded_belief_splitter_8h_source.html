47 belief.forEach([&localActionIndex, &splitBeliefs, this](BeliefStateType const& state, BeliefValueType const& beliefValue) {
48 auto const globalActionIndex = pomdp.getTransitionMatrix().getRowGroupIndices()[state] + localActionIndex;
49 auto const& rewVector = actionRewardVectors[globalActionIndex];
50 splitBeliefs[rewVector].addValue(state, beliefValue);

52
53 for (auto& [rewardVector, builder] : splitBeliefs) {
54 // Index 0 shares the original POMDP observation encoding used by the initial belief.
55 // The initial occurrence has no incoming reward; generated successors carry their reward vector
56 // on the transition. Do not infer an initial reward from the observation value.
57 BeliefActionObservationType const freshIndex = rewardVectorToIndex.size();
58 auto const rewVectorIndex = rewardVectorToIndex.emplace(rewardVector, freshIndex).first->second;
59 builder.setObservation(belief.observation());
60 if (splitBeliefs.size() == 1u) {
61 // Fix the distribution to diminish numerical issues a bit
62 callback(builder.build(), storm::utility::one<BeliefValueType>(), rewVectorIndex, rewardVector);
63 } else {
64 auto val = builder.normalize();
65 callback(builder.build(), std::move(val), rewVectorIndex, rewardVector);
66 }
67 }
68 }
