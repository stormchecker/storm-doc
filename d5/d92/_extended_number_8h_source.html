410 if constexpr (storm::NumberTraits<ValueType>::HasInfinity) {
411 // A NaN is not infinite, but it is not a finite value either: the guards built on this predicate are there to

664TargetType convertNumber(ExtendedNumber<SourceType> const& number) {
665 if (number.isFinite()) {

680template<typename TargetType, typename SourceType>
681 requires(detail::IsExtendedNumber<TargetType>::value && !detail::IsExtendedNumber<SourceType>::value)
682TargetType convertNumber(SourceType const& number) {
683 if constexpr (storm::NumberTraits<SourceType>::HasInfinity) {
684 // The source brings its own infinity, which the underlying conversion has no way of expressing.
685 if (storm::utility::isInfinity(number)) {
686 return TargetType::infinity();
687 }
688 if (storm::utility::isInfinity(SourceType(-number))) {
689 return TargetType::negativeInfinity();
690 }
691 }
692 return TargetType(convertNumber<FiniteValueType<TargetType>, SourceType>(number));
