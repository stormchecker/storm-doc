20// A single term of a polynomial: a rational coefficient given as a numerator and a denominator, multiplied by a
21// product of variables, where a repeated variable stands for a power.
22struct Term {
23 std::int64_t numerator;
24 std::int64_t denominator;
25 std::vector<Variable> factors;
26};
27
28using Terms = std::vector<Term>;

30// Creating a rational function variable always yields a new one, even for a name that is already in use. The tests
31// need the same variable everywhere, so these are created only once.
32Variable p() {
33 static Variable const variable = storm::createRFVariable("p");
34 return variable;
35}

37Variable q() {
38 static Variable const variable = storm::createRFVariable("q");
39 return variable;

42// An integer as an exact rational number. The cast keeps the constructor call unambiguous.
43storm::RationalNumber rational(std::int64_t value) {
44 return storm::RationalNumber{static_cast<long>(value)};
45}

47// The coefficient of the given term as carl uses it.
48Coeff coefficientOf(Term const& term) {
49 return Coeff(term.numerator) / Coeff(term.denominator);
50}
51
52// Builds the polynomial described by the given terms. This constructs exactly the polynomial type that
53// ConstraintCollector hands to polynomialToExpression, so no part of the value is lost on the way in.
54storm::RawPolynomial buildPolynomial(Terms const& terms) {
55 storm::RawPolynomial result;
56 for (auto const& term : terms) {
57 if (term.numerator == 0) {
58 continue;
59 }
60 storm::RawPolynomial partial{coefficientOf(term)};
61 for (auto const& factor : term.factors) {
62 partial *= factor;
63 }
64 result += partial;
65 }
66 return result;
67}

69// Converts the given polynomial to an expression and evaluates that expression at the given values.
70storm::RationalNumber evaluateConverted(storm::RawPolynomial const& polynomial, storm::RationalNumber const& pValue, storm::RationalNumber const& qValue) {
71 auto manager = std::make_shared<storm::expressions::ExpressionManager>();
72 auto expression = storm::expressions::polynomialToExpression(polynomial, manager);
73
74 storm::expressions::ExpressionEvaluator<storm::RationalNumber> evaluator(*manager);
75 if (manager->hasVariable("p")) {
76 evaluator.setRationalValue(manager->getVariable("p"), pValue);
77 }
78 if (manager->hasVariable("q")) {
79 evaluator.setRationalValue(manager->getVariable("q"), qValue);
80 }
81
82 return evaluator.asRational(expression);
83}

85// Evaluates the terms described by the given list directly, without any conversion, which yields the value that the
86// conversion has to reproduce.
87storm::RationalNumber evaluateDirectly(Terms const& terms, storm::RationalNumber const& pValue, storm::RationalNumber const& qValue) {
88 storm::RationalNumber result{0};
89 for (auto const& term : terms) {
90 if (term.numerator == 0) {
91 continue;
92 }
93 storm::RationalNumber value = rational(term.numerator);
94 value /= rational(term.denominator);
95 for (auto const& factor : term.factors) {
96 value *= factor == p() ? pValue : qValue;
97 }
98 result += value;
99 }
100 return result;
101}

118// Checks that converting the polynomial described by the given terms to an expression preserves its value.
119void expectValuesPreserved(Terms const& terms) {
120 auto polynomial = buildPolynomial(terms);
121
122 for (auto const& values : sampleValues()) {
123 auto expected = evaluateDirectly(terms, values.first, values.second);
124 auto actual = evaluateConverted(polynomial, values.first, values.second);
125
126 EXPECT_EQ(expected, actual) << "at p = " << values.first << " and q = " << values.second;
127 }
128}
