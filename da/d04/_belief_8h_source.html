67 SummandsType getWeightedSum(std::vector<SummandsType> const& summands) const {
68 auto sum = storm::utility::zero<SummandsType>();
69 forEach([&sum, &summands](auto const& state, auto const& val) {
70 STORM_LOG_ASSERT(state < summands.size(), "State " << state << " is out of range for the given summands.");
71 // If SummandsType is the same as ValueType, we can directly multiply the value, otherwise convert
72 if constexpr (std::is_same_v<SummandsType, ValueType>) {
73 sum += summands[state] * val;

75 sum += summands[state] * storm::utility::convertNumber<SummandsType>(val);

123 static_assert(BeliefFlatMapIsOrdered);
124 auto const zero = storm::utility::zero<ValueType>();
125 auto it2 = other.data.cbegin();
126 auto const it2End = other.data.cend();
127 if (considerOnlyThisSupport) {
128 for (auto const& [state1, value1] : data) {
129 while (it2 != it2End && it2->first < state1) {
130 ++it2;
131 }
132 if (it2 == it2End || it2->first > state1) {
133 f(state1, value1, zero);
134 } else {
135 STORM_LOG_ASSERT(state1 == it2->first, "Unexpected state.");
136 f(state1, value1, it2->second);
137 }
138 }
139 } else {
140 auto it1 = data.cbegin();
141 auto const it1End = data.cend();
142 while (it1 != it1End && it2 != it2End) {
143 auto const state1 = it1->first;
144 auto const state2 = it2->first;
145 if (state1 == state2) {
146 f(state1, it1->second, it2->second);
147 ++it1;
148 ++it2;
149 } else if (state1 < state2) {
150 f(state1, it1->second, zero);
151 ++it1;
152 } else {
153 STORM_LOG_ASSERT(state2 < state1, "unexpected states.");
154 f(state2, zero, it2->second);
155 ++it2;
156 }
157 }
158 for (; it1 != it1End; ++it1) {
159 f(it1->first, it1->second, zero);
160 }
161 for (; it2 != it2End; ++it2) {
162 f(it2->first, zero, it2->second);
