190 template<typename InfoType, typename NextStateHandleType>
191 bool performExploration(InfoType& info, NextStateHandleType&& exploreNextStates, TerminalBeliefCallback const& terminalBeliefCallback,
192 TerminationCallback const& terminationCallback) {
193 while (info.queue.hasNext()) {
194 // Check if we terminate prematurely
195 if ((terminationCallback && terminationCallback()) || storm::utility::resources::isTerminate()) {
196 STORM_LOG_ASSERT(storm::utility::resources::isTerminate() || info.queue.getContents() == info.getFrontierBeliefs(),
197 "Frontier beliefs inconsistent.");
198 return false; // Terminate prematurely
199 }
200
201 // Get the next belief to explore and perform some checks
202 auto const currentBeliefId = info.queue.popNext();
203 STORM_LOG_ASSERT(info.discoveredBeliefs.containsId(currentBeliefId), "Unknown belief id");
204 STORM_LOG_ASSERT(info.exploredBeliefs.count(currentBeliefId) == 0, "Belief #" << currentBeliefId << " already explored.");
205 STORM_LOG_ASSERT(info.terminalBeliefValues.count(currentBeliefId) == 0, "Belief #" << currentBeliefId << " already found to be terminal.");
206 // do not take the current belief as reference since it will be invalidated when collecting more beliefs
207 auto const currentBelief = info.discoveredBeliefs.getBeliefFromId(currentBeliefId);
208 STORM_LOG_TRACE("Explore belief " << currentBeliefId << " : " << currentBelief.toString());
209 // Check if the current belief is terminal
210 if (terminalBeliefCallback) {
211 if (auto terminal = terminalBeliefCallback(currentBelief); terminal.has_value()) {
212 info.terminalBeliefValues.emplace(currentBeliefId, std::move(terminal.value()));
213 continue;
214 }
215 }
216
217 // Explore for each action the successors of the current belief with that action. Potentially also add rewards.
218 info.exploredBeliefs.emplace(currentBeliefId, info.matrix.groups());
219 auto const numActions = firstStateNextStateGenerator.getBeliefNumberOfActions(currentBelief);
220 for (uint64_t localActionIndex = 0; localActionIndex < numActions; ++localActionIndex) {
221 exploreNextStates(currentBelief, localActionIndex);
222 info.matrix.endCurrentRow();
223 if (firstStateNextStateGenerator.hasRewardModel()) {
224 info.actionRewards.emplace_back(
225 storm::utility::convertNumber<BeliefMdpValueType>(firstStateNextStateGenerator.getBeliefActionReward(currentBelief, localActionIndex)));
226 }
227 if (info.generateChoiceLabeling) {
228 info.matrix.choiceLabels.push_back(firstStateNextStateGenerator.getBeliefActionChoiceLabels(currentBelief, localActionIndex));
229 }
230 }
231 info.matrix.endCurrentRowGroup();
232 }
233 return true;
234 }
