61storm::models::sparse::Pomdp<double> buildSingleStatePomdpWithObservationValuations(bool reverseObservableOrder) {
62 storm::storage::SparseMatrixBuilder<double> matrixBuilder(0, 0, 0, false, true);
63 matrixBuilder.newRowGroup(0);
64 matrixBuilder.addNextValue(0, 0, 1.0);
65
66 storm::models::sparse::StateLabeling stateLabeling(1);
67 stateLabeling.addLabel("init");
68 stateLabeling.addLabelToState("init", 0);
69
70 auto manager = std::make_shared<storm::expressions::ExpressionManager>();
71 auto const enabled = manager->declareBooleanVariable("enabled");
72 auto const mode = manager->declareIntegerVariable("mode");
73 storm::storage::sparse::ValuationDescriptionBuilder valuationBuilder(manager);
74 if (reverseObservableOrder) {
75 valuationBuilder.addIntegerVariable(mode, 0, 1);

82 observationValuations.getStorage().resize(1);
83 observationValuations.getStorage().writeValue(0, enabled, true);
84 observationValuations.getStorage().writeValue(0, mode, int64_t{1});
85
86 storm::storage::sparse::ModelComponents<double> components(matrixBuilder.build(), std::move(stateLabeling));
87 components.observabilityClasses = std::vector<uint32_t>{0};
88 components.observationValuations = std::move(observationValuations);
