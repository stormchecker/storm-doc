16// The initial state branches to hidden states 1 and 2 with probabilities 1/4 and 3/4.
17// Action a reaches goal 3 with probabilities 1/5 and 4/5; action b reverses these.
18// All other transitions go to absorbing failure state 4. The optimal value is 13/20.
19Pomdp makePomdp(bool reverseActions = false, bool hiddenTarget = false) {
20 storm::storage::SparseMatrixBuilder<double> matrix(0, 0, 0, false, true);
21 std::vector<std::string> actionLabels;
22 uint64_t row = 0;
23 auto addAction = [&](std::string const& label, std::initializer_list<std::pair<uint64_t, double>> transitions) {
24 for (auto const& [state, probability] : transitions) {
25 matrix.addNextValue(row, state, probability);
26 }
27 actionLabels.push_back(label);
28 ++row;
29 };
30 matrix.newRowGroup(row);
31 addAction("start", {{1, 0.25}, {2, 0.75}});
32 matrix.newRowGroup(row);
33 addAction("a", {{3, 0.2}, {4, 0.8}});
34 addAction("b", {{3, 0.8}, {4, 0.2}});
35 matrix.newRowGroup(row);
36 if (reverseActions) {
37 addAction("b", {{3, 0.2}, {4, 0.8}});
38 addAction("a", {{3, 0.8}, {4, 0.2}});
39 } else {
40 addAction("a", {{3, 0.8}, {4, 0.2}});
41 addAction("b", {{3, 0.2}, {4, 0.8}});
42 }
43 matrix.newRowGroup(row);
44 addAction("a", {{3, 1.0}});
45 addAction("b", {{3, 1.0}});
46 matrix.newRowGroup(row);
47 addAction("stay", {{4, 1.0}});
48
49 storm::models::sparse::StateLabeling labels(5);
50 labels.addLabel("init");
51 labels.addLabelToState("init", 0);
52 labels.addLabel("goal");
53 labels.addLabelToState("goal", 3);
54 labels.addLabel("safe");
55 labels.addLabelToState("safe", 0);
56 labels.addLabelToState("safe", 1);
57 storm::models::sparse::ChoiceLabeling choices(row);
58 for (uint64_t choice = 0; choice < row; ++choice) {
59 if (!choices.containsLabel(actionLabels[choice])) {
60 choices.addLabel(actionLabels[choice]);
61 }
62 choices.addLabelToChoice(actionLabels[choice], choice);
63 }
64 storm::storage::sparse::ModelComponents<double> components(matrix.build(), std::move(labels));
65 components.choiceLabeling = std::move(choices);
66 components.observabilityClasses = hiddenTarget ? std::vector<uint32_t>{0, 1, 1, 1, 2} : std::vector<uint32_t>{0, 1, 1, 2, 3};
67 return Pomdp(std::move(components), !reverseActions);
68}

110 auto pomdp = makePomdp();

189TEST(PomdpVerificationClippingTest, RefinesTargetsAndRemapsResolutionsWithoutChangingInput) {
190#if !defined(STORM_HAVE_Z3) || !defined(STORM_HAVE_GLPK)
191 GTEST_SKIP() << "Clipping test requires Z3 preprocessing and GLPK.";
192#endif
193 auto pomdp = makePomdp(false, true);
194 ASSERT_EQ(pomdp.getNrObservations(), 3ul);
195 ASSERT_EQ(pomdp.getObservation(1), pomdp.getObservation(3));
196 auto const observations = pomdp.getObservations();
197 storm::parser::FormulaParser parser;
198 auto formula = parser.parseSingleFormulaFromString("Pmax=? [F \"goal\"]");
199 storm::Environment env;
200 env.solver().setLpSolverType(storm::solver::LpSolverType::Glpk);
201 Options options;
202 options.maxExplorationSize = 1;
203 options.useClipping = true;
204 // The hidden belief (1/4, 3/4) is off the resolution-3 grid, forcing clipping.
205 options.clippingResolutions = {2, 3, 5};
206
207 auto const result = storm::pomdp::api::verifyWithBeliefUnfolding(env, pomdp, *formula, options);
208
209 // Explicitly separate the target, preserving the original observation IDs for other states.
210 // The fresh observation 3 must inherit resolution 3 from original observation 1.
211 auto refined = pomdp;
212 refined.updateObservations({0, 1, 1, 3, 2}, true);
213 auto refinedOptions = options;
214 refinedOptions.clippingResolutions = {2, 3, 5, 3};
215 auto const expected = storm::pomdp::api::verifyWithBeliefUnfolding(env, refined, *formula, refinedOptions);
216 EXPECT_NEAR(result.value, 0.65, 1e-8);
217 EXPECT_NEAR(result.value, expected.value, 1e-8);
218 EXPECT_EQ(result.statistics.discoveredBeliefs, expected.statistics.discoveredBeliefs);
219 EXPECT_GT(result.statistics.exploredBeliefs, 1ul);
220 EXPECT_EQ(pomdp.getObservations(), observations);
221 ASSERT_TRUE(options.clippingResolutions.has_value());
222 EXPECT_EQ(*options.clippingResolutions, (std::vector<uint64_t>{2, 3, 5}));
223 EXPECT_EQ(options.maxExplorationSize, 1ul);
224}
