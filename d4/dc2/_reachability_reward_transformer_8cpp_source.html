22template<typename ValueType>
23TotalRewardSpecification<ValueType> ReachabilityRewardTransformer<ValueType>::transform(StateSet const& targetStates,
24 std::vector<ValueType> const& actionRewards) {
25 STORM_LOG_ASSERT(transitionMatrix.hasTrivialRowGrouping(), "Transition matrix has non-trivial row grouping but no optimization direction is given.");
26
27 // As usual for reachability rewards, the reward is infinite if the target is not reached with probability one (even if all rewards are zero).
28 StateSet allStates(transitionMatrix.getRowGroupCount(), true);
29 auto infStates = storm::utility::graph::performProb1(backwardTransitionCache.get(), allStates, targetStates);
30 infStates.complement();
31 return TotalRewardSpecification<ValueType>{.terminalStates = targetStates | infStates,
32 .terminalStateValues = {{TerminalStateValue<ValueType>::getPlusInfinity(), std::move(infStates)}},
33 .terminalStatesUniversallyAlmostSurelyReached = true,
34 .actionRewards = actionRewards};
35}

37template<typename ValueType>
38TotalRewardSpecification<ValueType> ReachabilityRewardTransformer<ValueType>::transform(std::optional<storm::OptimizationDirection> dir,
39 StateSet const& targetStates,
40 std::vector<ValueType> const& actionRewards) {
41 if (!dir.has_value()) {
42 return transform(targetStates, actionRewards);
43 }
44 // As usual for reachability rewards, the reward is infinite if the target is not reached with probability one (even if all rewards are zero).
45 // For minimizing (maximizing), this considers whether there is a scheduler (whether all schedulers) that reach the target with probability one.
46 StateSet allStates(transitionMatrix.getRowGroupCount(), true);
47 auto infStates = storm::solver::minimize(*dir) ? storm::utility::graph::performProb1E(transitionMatrix, transitionMatrix.getRowGroupIndices(),
48 backwardTransitionCache.get(), allStates, targetStates)
49 : storm::utility::graph::performProb1A(transitionMatrix, transitionMatrix.getRowGroupIndices(),
50 backwardTransitionCache.get(), allStates, targetStates);
51 infStates.complement();
52 return TotalRewardSpecification<ValueType>{.terminalStates = targetStates | infStates,
53 .terminalStateValues = {{TerminalStateValue<ValueType>::getPlusInfinity(), std::move(infStates)}},
54 .terminalStatesUniversallyAlmostSurelyReached = storm::solver::maximize(*dir),
55 .actionRewards = actionRewards};
56}
