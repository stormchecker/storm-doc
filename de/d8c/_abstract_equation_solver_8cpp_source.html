291template<typename SolutionType>
292void AbstractEquationSolver<SolutionType>::setSolutionBoundsFromPrecision(std::vector<SolutionType> const& x, SolutionType const& precision,
293 bool relative) const {
294 if constexpr (std::is_same_v<SolutionType, storm::RationalFunction>) {
295 STORM_LOG_THROW(false, storm::exceptions::InvalidOperationException, "Cannot derive solution bounds from a precision for rational functions.");
296 } else {
297 STORM_LOG_ASSERT(!relative || precision < storm::utility::one<SolutionType>(), "A relative precision of one or more carries no information.");
298 SolutionBounds<SolutionType> bounds;
299 std::vector<SolutionType>& lower = bounds.lower.emplace(x.size());
300 std::vector<SolutionType>& upper = bounds.upper.emplace(x.size());
301 for (uint64_t i = 0; i < x.size(); ++i) {
302 // For the relative criterion the guarantee reads |x_i - s_i| <= precision * |s_i| for the exact solution s.
303 // That gives |s_i| <= |x_i| / (1 - precision) and hence the deviation below, which does not refer to s.
304 SolutionType const deviation =
305 relative ? precision * storm::utility::abs<SolutionType>(x[i]) / (storm::utility::one<SolutionType>() - precision) : precision;
306 lower[i] = x[i] - deviation;
307 upper[i] = x[i] + deviation;
308 }
309 this->setSolutionBounds(std::move(bounds));
310 }
311}

313template<typename SolutionType>
314void AbstractEquationSolver<SolutionType>::finalizeSolutionBounds(std::vector<SolutionType> const& x) const {
315 if constexpr (std::is_same_v<SolutionType, storm::RationalFunction>) {
316 // Rational functions are not ordered, so there is no tighter of two bounds to pick.
317 return;
318 } else {
319 // Bounds that were known beforehand hold for the solution, so they tighten whatever was computed.
320 if (this->hasLowerBound()) {
321 bool const hadBound = solutionBounds.hasLower();
322 std::vector<SolutionType>& bound = hadBound ? *solutionBounds.lower : solutionBounds.lower.emplace(x.size());
323 STORM_LOG_ASSERT(bound.size() == x.size(), "Computed lower bound does not match the size of the solution.");
324 for (uint64_t i = 0; i < x.size(); ++i) {
325 bound[i] = hadBound ? std::max(bound[i], this->getLowerBound(i)) : this->getLowerBound(i);
326 }
327 }
328 if (this->hasUpperBound()) {
329 bool const hadBound = solutionBounds.hasUpper();
330 std::vector<SolutionType>& bound = hadBound ? *solutionBounds.upper : solutionBounds.upper.emplace(x.size());
331 STORM_LOG_ASSERT(bound.size() == x.size(), "Computed upper bound does not match the size of the solution.");
332 for (uint64_t i = 0; i < x.size(); ++i) {
333 bound[i] = hadBound ? std::min(bound[i], this->getUpperBound(i)) : this->getUpperBound(i);
334 }
335 }
336 // Every step between establishing a bound and handing back a solution, the extraction of a scheduler among
337 // them, can move a value a rounding error past it. Following the values keeps the bounds sound.
338 solutionBounds.widenTo(x);
339 }

420template<typename SolutionType>
421void AbstractEquationSolver<SolutionType>::reportStatus(SolverStatus status, boost::optional<uint64_t> const& iterations) const {
422 if (iterations) {
423 switch (status) {
424 case SolverStatus::Converged:
425 STORM_LOG_TRACE("Iterative solver converged after " << iterations.get() << " iterations.");
426 break;
427 case SolverStatus::TerminatedEarly:
428 STORM_LOG_TRACE("Iterative solver terminated early after " << iterations.get() << " iterations.");
429 break;
430 case SolverStatus::MaximalIterationsExceeded:
431 STORM_LOG_WARN("Iterative solver did not converge after " << iterations.get() << " iterations.");
432 break;
433 case SolverStatus::Aborted:
434 STORM_LOG_WARN("Iterative solver was aborted after " << iterations.get() << " iterations.");
435 break;
436 default:
437 STORM_LOG_THROW(false, storm::exceptions::InvalidStateException, "Iterative solver terminated unexpectedly.");
438 }
439 } else {
440 switch (status) {
441 case SolverStatus::Converged:
442 STORM_LOG_TRACE("Solver converged.");
443 break;
444 case SolverStatus::TerminatedEarly:
445 STORM_LOG_TRACE("Solver terminated early.");
446 break;
447 case SolverStatus::MaximalIterationsExceeded:
448 STORM_LOG_ASSERT(false, "Non-iterative solver should not exceed maximal number of iterations.");
449 STORM_LOG_WARN("Solver did not converge.");
450 break;
451 case SolverStatus::Aborted:
452 STORM_LOG_WARN("Solver was aborted.");
453 break;
454 default:
455 STORM_LOG_THROW(false, storm::exceptions::InvalidStateException, "Solver terminated unexpectedly.");
456 }
457 }

461SolverStatus AbstractEquationSolver<SolutionType>::updateStatus(SolverStatus status, bool earlyTermination, uint64_t iterations,
462 uint64_t maximalNumberOfIterations) const {
463 if (status != SolverStatus::Converged) {
