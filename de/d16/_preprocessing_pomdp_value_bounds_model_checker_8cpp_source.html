44template<typename PomdpType>
45std::pair<std::vector<typename PreprocessingPomdpValueBoundsModelChecker<PomdpType>::PomdpValueType>,
46 storm::storage::Scheduler<typename PreprocessingPomdpValueBoundsModelChecker<PomdpType>::PomdpValueType>>
47PreprocessingPomdpValueBoundsModelChecker<PomdpType>::computeValuesForGuessedScheduler(
48 storm::Environment const& env, std::vector<PomdpValueType> const& stateValues, std::vector<PomdpValueType>* actionBasedRewards,
49 storm::logic::Formula const& formula, storm::pomdp::analysis::FormulaInformation const& info,
50 std::shared_ptr<storm::models::sparse::Mdp<PomdpValueType>> underlyingMdp, PomdpValueType const& scoreThreshold, bool relativeScore) {
51 // Create some positional scheduler for the POMDP
52 storm::storage::Scheduler<PomdpValueType> pomdpScheduler(pomdp.getNumberOfStates());
53 // For each state, we heuristically find a good distribution over output actions.
54 auto choiceValues = getChoiceValues(stateValues, actionBasedRewards);
55 auto const& choiceIndices = pomdp.getTransitionMatrix().getRowGroupIndices();
56 std::vector<storm::storage::Distribution<PomdpValueType, uint_fast64_t>> choiceDistributions(pomdp.getNrObservations());
57 for (uint64_t state = 0; state < pomdp.getNumberOfStates(); ++state) {
58 auto& choiceDistribution = choiceDistributions[pomdp.getObservation(state)];
59 PomdpValueType const& stateValue = stateValues[state];
60 STORM_LOG_ASSERT(stateValue >= storm::utility::zero<PomdpValueType>(), "State value expected non-negative.");
61 for (auto choice = choiceIndices[state]; choice < choiceIndices[state + 1]; ++choice) {
62 PomdpValueType const& choiceValue = choiceValues[choice];
63 STORM_LOG_ASSERT(choiceValue >= storm::utility::zero<PomdpValueType>(), "Choice value expected non-negative.");
64 // Rate this choice by considering the relative difference between the choice value and the (optimal) state value
65 // A high score shall mean that the choice is "good"
66 if (storm::utility::isInfinity(stateValue)) {
67 // For infinity states, we simply distribute uniformly.
68 // This case could be handled a bit more sensible
69 choiceDistribution.addProbability(choice - choiceIndices[state], scoreThreshold);
70 } else {
71 PomdpValueType choiceScore = info.minimize() ? (choiceValue - stateValue) : (stateValue - choiceValue);
72 if (relativeScore) {
73 PomdpValueType avg = (stateValue + choiceValue) / storm::utility::convertNumber<PomdpValueType, uint64_t>(2);
74 if (!storm::utility::isZero(avg)) {
75 choiceScore /= avg;
76 }
77 }
78 choiceScore = storm::utility::one<PomdpValueType>() - choiceScore;
79 if (choiceScore >= scoreThreshold) {
80 choiceDistribution.addProbability(choice - choiceIndices[state], choiceScore);
81 }
82 }
83 }
84 // If the distribution is empty, i.e. no choice has had a suitable score, distribute uniformly
85 if (choiceDistribution.size() == 0) {
86 for (auto choice = choiceIndices[state]; choice < choiceIndices[state + 1]; ++choice) {
87 choiceDistribution.addProbability(choice - choiceIndices[state], scoreThreshold);
88 }
89 }
90 STORM_LOG_ASSERT(choiceDistribution.size() > 0, "Empty choice distribution.");
91 }
92 // Normalize all distributions
93 for (auto& choiceDistribution : choiceDistributions) {
94 choiceDistribution.normalize();
95 }
96 // Set the scheduler for all states
97 for (uint64_t state = 0; state < pomdp.getNumberOfStates(); ++state) {
98 pomdpScheduler.setChoice(choiceDistributions[pomdp.getObservation(state)], state);
99 }
100 STORM_LOG_ASSERT(!pomdpScheduler.isPartialScheduler(), "Expected a fully defined scheduler.");
101 auto scheduledModel = underlyingMdp->applyScheduler(pomdpScheduler, false);
102
103 auto resultPtr =
104 storm::api::verifyWithSparseEngine<PomdpValueType>(env, scheduledModel, storm::api::createTask<PomdpValueType>(formula.asSharedPointer(), false));
105 STORM_LOG_THROW(resultPtr, storm::exceptions::UnexpectedException, "No check result obtained.");
106 STORM_LOG_THROW(resultPtr->isExplicitQuantitativeCheckResult(), storm::exceptions::UnexpectedException, "Unexpected Check result Type.");
107 std::vector<PomdpValueType> pomdpSchedulerResult = resultPtr->template asExplicitQuantitativeCheckResult<PomdpValueType>().getSentinelValueVector();
108 return std::make_pair(pomdpSchedulerResult, pomdpScheduler);
109}

114PreprocessingPomdpValueBoundsModelChecker<PomdpType>::computeValuesForRandomFMPolicy(storm::Environment const& env, storm::logic::Formula const& formula,
115 storm::pomdp::analysis::FormulaInformation const& info,
116 uint64_t memoryBound) {
117 // Consider memoryless policy on memory-unfolded POMDP
118 storm::storage::Scheduler<PomdpValueType> pomdpScheduler(pomdp.getNumberOfStates() * memoryBound);
119
120 STORM_LOG_DEBUG("Computing the unfolding for memory bound " << memoryBound);
121 storm::storage::PomdpMemory memory = storm::storage::PomdpMemoryBuilder().build(storm::storage::PomdpMemoryPattern::Full, memoryBound);
122 storm::transformer::PomdpMemoryUnfolder<PomdpValueType> memoryUnfolder(pomdp, memory);
123 // We keep unreachable states to not mess with the state ordering and capture potential better choices
124 auto memPomdp = memoryUnfolder.transform(false);
125
126 // Determine an observation-based policy by choosing any of the enabled actions uniformly at random
127 std::vector<uint64_t> obsChoiceVector(memPomdp->getNrObservations());
128 std::random_device rd;
129 auto engine = std::mt19937(rd());
130 for (uint64_t obs = 0; obs < memPomdp->getNrObservations(); ++obs) {
131 uint64_t nrChoices = memPomdp->getNumberOfChoices(memPomdp->getStatesWithObservation(obs).front());
132 std::uniform_int_distribution<uint64_t> uniform_dist(0, nrChoices - 1);
133 obsChoiceVector[obs] = uniform_dist(engine);
134 }
135 for (uint64_t state = 0; state < memPomdp->getNumberOfStates(); ++state) {
136 pomdpScheduler.setChoice(obsChoiceVector[memPomdp->getObservation(state)], state);
137 }
138
139 // Model check the DTMC resulting from the policy
140 auto underlyingMdp = std::make_shared<storm::models::sparse::Mdp<PomdpValueType>>(memPomdp->getTransitionMatrix(), memPomdp->getStateLabeling(),
141 memPomdp->getRewardModels());
142 auto scheduledModel = underlyingMdp->applyScheduler(pomdpScheduler, false);
143 auto resultPtr =
144 storm::api::verifyWithSparseEngine<PomdpValueType>(env, scheduledModel, storm::api::createTask<PomdpValueType>(formula.asSharedPointer(), false));
145 STORM_LOG_THROW(resultPtr, storm::exceptions::UnexpectedException, "No check result obtained.");
146 STORM_LOG_THROW(resultPtr->isExplicitQuantitativeCheckResult(), storm::exceptions::UnexpectedException, "Unexpected Check result Type.");
147 std::vector<PomdpValueType> pomdpSchedulerResult = resultPtr->template asExplicitQuantitativeCheckResult<PomdpValueType>().getSentinelValueVector();
148
149 // Take the optimal value in ANY of the unfolded states for a POMDP state as the resulting state value
150 std::vector<PomdpValueType> res(pomdp.getNumberOfStates(), storm::utility::zero<PomdpValueType>());
151 storm::storage::BitVector hasValue(pomdp.getNumberOfStates(), false);
152 for (uint64_t memPomdpState = 0; memPomdpState < pomdpSchedulerResult.size(); ++memPomdpState) {
153 uint64_t modelState = memPomdpState / memoryBound;
154 if (!hasValue.get(modelState) || (info.minimize() && pomdpSchedulerResult[memPomdpState] < res[modelState]) ||
155 (!info.minimize() && pomdpSchedulerResult[memPomdpState] > res[modelState])) {
156 res[modelState] = pomdpSchedulerResult[memPomdpState];
157 hasValue.set(modelState);
158 }
159 }
160 return std::make_pair(res, pomdpScheduler);
161}

198template<typename PomdpType>
199PreprocessingPomdpValueBoundsModelChecker<PomdpType>::ValueBounds PreprocessingPomdpValueBoundsModelChecker<PomdpType>::getValueBounds(
200 storm::Environment const& env, storm::logic::Formula const& formula, storm::pomdp::analysis::FormulaInformation const& info) {
201 STORM_LOG_THROW(
202 !formula.isOperatorFormula() || !formula.asOperatorFormula().hasBound(), storm::exceptions::InvalidPropertyException,
203 "The threshold in '" << formula << "' is not supported by quantitative POMDP analysis. Use a Pmin=?/Pmax=? or Rmin=?/Rmax=? query instead.");
204 STORM_LOG_THROW(info.isNonNestedReachabilityProbability() || info.isNonNestedExpectedRewardFormula(), storm::exceptions::NotSupportedException,
205 "The property type is not supported for this analysis.");
206
207 // Compute the values on the fully observable MDP
208 // We need an actual MDP so that we can apply schedulers below.
209 // Also, the api call in the next line will require a copy anyway.
210 auto underlyingMdp =
211 std::make_shared<storm::models::sparse::Mdp<PomdpValueType>>(pomdp.getTransitionMatrix(), pomdp.getStateLabeling(), pomdp.getRewardModels());
212 auto resultPtr =
213 storm::api::verifyWithSparseEngine<PomdpValueType>(env, underlyingMdp, storm::api::createTask<PomdpValueType>(formula.asSharedPointer(), false));
214 STORM_LOG_THROW(resultPtr, storm::exceptions::UnexpectedException, "No check result obtained.");
215 STORM_LOG_THROW(resultPtr->isExplicitQuantitativeCheckResult(), storm::exceptions::UnexpectedException, "Unexpected Check result Type.");
216 std::vector<PomdpValueType> fullyObservableResult = resultPtr->template asExplicitQuantitativeCheckResult<PomdpValueType>().getSentinelValueVector();
217
218 std::vector<PomdpValueType> actionBasedRewards;
219 std::vector<PomdpValueType>* actionBasedRewardsPtr = nullptr;
220 if (info.isNonNestedExpectedRewardFormula()) {
221 actionBasedRewards = pomdp.getRewardModel(info.getRewardModelName()).getTotalRewardVector(pomdp.getTransitionMatrix());
222 actionBasedRewardsPtr = &actionBasedRewards;
223 }
224 std::vector<std::vector<PomdpValueType>> guessedSchedulerValues;
225 std::vector<storm::storage::Scheduler<PomdpValueType>> guessedSchedulers;
226 std::shared_ptr<std::pair<std::vector<PomdpValueType>, storm::storage::Scheduler<PomdpValueType>>> guessedSchedulerPair;
227 std::vector<std::pair<double, bool>> guessParameters({{0.875, false}, {0.875, true}, {0.75, false}, {0.75, true}});
228 for (auto const& pars : guessParameters) {
229 guessedSchedulerPair = std::make_shared<std::pair<std::vector<PomdpValueType>, storm::storage::Scheduler<PomdpValueType>>>(
230 computeValuesForGuessedScheduler(env, fullyObservableResult, actionBasedRewardsPtr, formula, info, underlyingMdp,
231 storm::utility::convertNumber<PomdpValueType>(pars.first), pars.second));
232 guessedSchedulerValues.push_back(guessedSchedulerPair->first);
233 guessedSchedulers.push_back(guessedSchedulerPair->second);
234 }
235
236 // compute the 'best' guess and do a few iterations on it
237 uint64_t bestGuess = 0;
238 PomdpValueType bestGuessSum =
239 std::accumulate(guessedSchedulerValues.front().begin(), guessedSchedulerValues.front().end(), storm::utility::zero<PomdpValueType>());
240 for (uint64_t guess = 1; guess < guessedSchedulerValues.size(); ++guess) {
241 PomdpValueType guessSum =
242 std::accumulate(guessedSchedulerValues[guess].begin(), guessedSchedulerValues[guess].end(), storm::utility::zero<PomdpValueType>());
243 if ((info.minimize() && guessSum < bestGuessSum) || (info.maximize() && guessSum > bestGuessSum)) {
244 bestGuess = guess;
245 bestGuessSum = guessSum;
246 }
247 }
248 guessedSchedulerPair = std::make_shared<std::pair<std::vector<PomdpValueType>, storm::storage::Scheduler<PomdpValueType>>>(
249 computeValuesForGuessedScheduler(env, guessedSchedulerValues[bestGuess], actionBasedRewardsPtr, formula, info, underlyingMdp,
250 storm::utility::convertNumber<PomdpValueType>(guessParameters[bestGuess].first), guessParameters[bestGuess].second));
251 guessedSchedulerValues.push_back(guessedSchedulerPair->first);
252 guessedSchedulers.push_back(guessedSchedulerPair->second);
253 guessedSchedulerPair = std::make_shared<std::pair<std::vector<PomdpValueType>, storm::storage::Scheduler<PomdpValueType>>>(
254 computeValuesForGuessedScheduler(env, guessedSchedulerValues.back(), actionBasedRewardsPtr, formula, info, underlyingMdp,
255 storm::utility::convertNumber<PomdpValueType>(guessParameters[bestGuess].first), guessParameters[bestGuess].second));
256 guessedSchedulerValues.push_back(guessedSchedulerPair->first);
257 guessedSchedulers.push_back(guessedSchedulerPair->second);
258 guessedSchedulerPair = std::make_shared<std::pair<std::vector<PomdpValueType>, storm::storage::Scheduler<PomdpValueType>>>(
259 computeValuesForGuessedScheduler(env, guessedSchedulerValues.back(), actionBasedRewardsPtr, formula, info, underlyingMdp,
260 storm::utility::convertNumber<PomdpValueType>(guessParameters[bestGuess].first), guessParameters[bestGuess].second));
261 guessedSchedulerValues.push_back(guessedSchedulerPair->first);
262 guessedSchedulers.push_back(guessedSchedulerPair->second);
263
264 // Check if one of the guesses is worse than one of the others (and potentially delete it)
265 // Avoid deleting entries during the loop to ensure that indices remain valid
266 storm::storage::BitVector keptGuesses(guessedSchedulerValues.size(), true);
267 for (uint64_t i = 0; i < guessedSchedulerValues.size() - 1; ++i) {
268 if (!keptGuesses.get(i)) {
269 continue;
270 }
271 for (uint64_t j = i + 1; j < guessedSchedulerValues.size(); ++j) {
272 if (!keptGuesses.get(j)) {
273 continue;
274 }
275 if (storm::utility::vector::compareElementWise(guessedSchedulerValues[i], guessedSchedulerValues[j], std::less_equal<PomdpValueType>())) {
276 if (info.minimize()) {
277 // In this case we are guessing upper bounds (and smaller upper bounds are better)
278 keptGuesses.set(j, false);
279 } else {
280 // In this case we are guessing lower bounds (and larger lower bounds are better)
281 keptGuesses.set(i, false);
282 break;
283 }
284 } else if (storm::utility::vector::compareElementWise(guessedSchedulerValues[j], guessedSchedulerValues[i], std::less_equal<PomdpValueType>())) {
285 if (info.minimize()) {
286 keptGuesses.set(i, false);
287 break;
288 } else {
289 keptGuesses.set(j, false);
290 }
291 }
292 }
293 }
294 STORM_LOG_INFO("Keeping scheduler guesses " << keptGuesses);
295 storm::utility::vector::filterVectorInPlace(guessedSchedulerValues, keptGuesses);
296 std::vector<storm::storage::Scheduler<PomdpValueType>> filteredSchedulers;
297 for (uint64_t i = 0; i < guessedSchedulers.size(); ++i) {
298 if (keptGuesses[i]) {
299 filteredSchedulers.push_back(guessedSchedulers[i]);
300 }
301 }
302
303 // Finally prepare the result
304 ValueBounds result;
305 if (info.minimize()) {
306 result.lower.push_back(std::move(fullyObservableResult));
307 result.upper = std::move(guessedSchedulerValues);
308 result.upperSchedulers = filteredSchedulers;
309 } else {
310 result.lower = std::move(guessedSchedulerValues);
311 result.upper.push_back(std::move(fullyObservableResult));
312 result.lowerSchedulers = filteredSchedulers;
313 }
314#ifndef NDEBUG
315 bool boundsValid = true;
316 auto maxDifference = storm::utility::zero<PomdpValueType>();
317 for (auto const& lower : result.lower) {
318 for (auto const& upper : result.upper) {
319 for (uint64_t state = 0; state < pomdp.getNumberOfStates(); ++state) {
320 if (storm::utility::min<PomdpValueType>(upper.at(state), lower.at(state)) == upper.at(state) && upper.at(state) != lower.at(state)) {
321 boundsValid = false;
322 maxDifference = storm::utility::max<PomdpValueType>(maxDifference, lower.at(state) - upper.at(state));
323 STORM_LOG_TRACE("Lower bound " << lower.at(state) << " at state " << state << " is larger than upper bound " << upper.at(state));
324 }
325 }
326 }
327 }
328
329 STORM_LOG_WARN_COND_DEBUG(boundsValid, "At least one lower bound is not smaller than an upper bound (max. difference: "
330 << maxDifference
331 << "). This might be due to floating point imprecisions. Enable TRACE output for more details.");
332#endif
333
334 return result;
335}

352 storm::Environment const& env, storm::logic::Formula const& formula, storm::pomdp::analysis::FormulaInformation const& info) {
353 STORM_LOG_THROW(!formula.isOperatorFormula() || !formula.asOperatorFormula().hasBound(), storm::exceptions::InvalidPropertyException,

355 STORM_LOG_THROW(info.isNonNestedExpectedRewardFormula(), storm::exceptions::NotSupportedException, "The property type is not supported for this analysis.");

357 // Compute the values for the opposite direction on the fully observable MDP

364 newFormula.setOptimalityType(storm::solver::OptimizationDirection::Maximize);
365 }
366 auto formulaPtr = std::make_shared<storm::logic::RewardOperatorFormula>(newFormula);
367 auto underlyingMdp =
368 std::make_shared<storm::models::sparse::Mdp<PomdpValueType>>(pomdp.getTransitionMatrix(), pomdp.getStateLabeling(), pomdp.getRewardModels());

374 if (info.minimize()) {
375 res.min = false;
376 } else {
377 res.min = true;
378 }
379 res.isInfinite = storm::utility::vector::filterInfinity(resultVec);
380 res.values = std::move(resultVec);
381 return res;
382}
