16std::shared_ptr<storm::models::sparse::Dtmc<RationalFunction>> BinaryDtmcTransformer::transform(storm::models::sparse::Dtmc<RationalFunction> const& dtmc,
17 bool keepStateValuations) const {
18 auto data = transformTransitions(dtmc);
19 storm::storage::sparse::ModelComponents<RationalFunction> components;
20 components.stateLabeling = transformStateLabeling(dtmc, data);
21 for (auto const& rewModel : dtmc.getRewardModels()) {

28

50std::shared_ptr<RawPolynomialCache> cacheOf(Polynomial const& polynomial) {
51 auto cache = polynomial.pCache();
52 if (!cache) {
53 // The polynomial was never factored, so there is nothing to share. This is only expected for constant
54 // polynomials.
55 STORM_LOG_ASSERT(polynomial.isConstant(), "Expected a non-constant polynomial to carry a cache.");
56 return std::make_shared<RawPolynomialCache>();
57 }
58 return cache;
59}

70RationalFunction toRationalFunction(RawPolynomial const& numerator, Polynomial const& denominator, std::shared_ptr<RawPolynomialCache> const& cache) {
71 STORM_LOG_ASSERT(!denominator.pCache() || denominator.pCache() == cache, "Expected the denominator to share the cache of the numerator.");

95typename BinaryDtmcTransformer::TransformationData BinaryDtmcTransformer::transformTransitions(
96 storm::models::sparse::Dtmc<RationalFunction> const& dtmc) const {
97 auto const& matrix = dtmc.getTransitionMatrix();
98
99 // All polynomials of the model share a single factorization cache, which the polynomials created below reuse.
100 auto const cache = cacheOfModel(dtmc);
101
102 // Initialize a FIFO Queue that stores the start and the end of each row
103 std::queue<StateWithRow> queue;
104 for (uint64_t state = 0; state < matrix.getRowCount(); ++state) {
105 std::vector<storage::MatrixEntry<uint64_t, RationalFunction>> diyRow;
106 for (auto const& entry : matrix.getRow(state)) {
107 diyRow.push_back(entry);
108 }
109 queue.emplace(StateWithRow{state, diyRow});
110 }
111
112 storm::storage::SparseMatrixBuilder<RationalFunction> builder;
113 uint64_t currRow = 0;
114 uint64_t currAuxState = queue.size();
115 std::vector<uint64_t> origStates;
116
117 while (!queue.empty()) {
118 auto stateWithRow = std::move(queue.front());
119 queue.pop();
120
121 std::set<RationalFunctionVariable> variablesInRow;
122
123 for (auto const& entry : stateWithRow.row) {
124 for (auto const& variable : entry.getValue().gatherVariables()) {
125 variablesInRow.emplace(variable);
126 }
127 }
128
129 if (variablesInRow.size() == 0) {
130 // Insert the row directly
131 for (auto const& entry : stateWithRow.row) {
132 builder.addNextValue(currRow, entry.getColumn(), entry.getValue());
133 }
134 ++currRow;
135 } else if (variablesInRow.size() == 1) {
136 auto parameter = *variablesInRow.begin();
137 auto parameterPol = RawPolynomial(parameter);
138 auto oneMinusParameter = RawPolynomial(1) - parameterPol;
139
140 std::vector<storage::MatrixEntry<uint64_t, RationalFunction>> outgoing;
141 // p * .. state
142 std::vector<storage::MatrixEntry<uint64_t, RationalFunction>> newStateLeft;
143 // (1-p) * .. state
144 std::vector<storage::MatrixEntry<uint64_t, RationalFunction>> newStateRight;
145
146 RationalFunction sumOfLeftBranch;
147 RationalFunction sumOfRightBranch;
148
149 for (auto const& entry : stateWithRow.row) {
150 if (entry.getValue().isConstant()) {
151 outgoing.push_back(entry);
152 }
153 auto nominator = entry.getValue().nominator();
154 auto denominator = entry.getValue().denominator();
155 auto byP = RawPolynomial(nominator).divideBy(parameterPol);
156 if (byP.remainder.isZero()) {
157 auto probability = toRationalFunction(byP.quotient, denominator, cache);
158 newStateLeft.push_back(storage::MatrixEntry<uint64_t, RationalFunction>(entry.getColumn(), probability));
159 sumOfLeftBranch += probability;
160 continue;
161 }
162 auto byOneMinusP = RawPolynomial(nominator).divideBy(oneMinusParameter);
163 if (byOneMinusP.remainder.isZero()) {
164 auto probability = toRationalFunction(byOneMinusP.quotient, denominator, cache);
165 newStateRight.push_back(storage::MatrixEntry<uint64_t, RationalFunction>(entry.getColumn(), probability));
166 sumOfRightBranch += probability;
167 continue;
168 }
169 STORM_LOG_ERROR("Invalid transition!");
170 }
171 sumOfLeftBranch.simplify();
172 sumOfRightBranch.simplify();
173 for (auto& entry : newStateLeft) {
174 entry.setValue(entry.getValue() / sumOfLeftBranch);
175 }
176 for (auto& entry : newStateRight) {
177 entry.setValue(entry.getValue() / sumOfRightBranch);
178 }
179
180 auto parameterPolynomial = Polynomial(RawPolynomial(parameter), cache);
181
182 queue.push(StateWithRow{currAuxState, newStateLeft});
183 outgoing.push_back(storage::MatrixEntry<uint64_t, RationalFunction>(currAuxState, (sumOfLeftBranch)*RationalFunction(parameterPolynomial)));
184 ++currAuxState;
185 queue.push(StateWithRow{currAuxState, newStateRight});
186 outgoing.push_back(storage::MatrixEntry<uint64_t, RationalFunction>(
187 currAuxState, (sumOfRightBranch) * (utility::one<RationalFunction>() - RationalFunction(parameterPolynomial))));
188 ++currAuxState;
189
190 for (auto const& entry : outgoing) {
191 builder.addNextValue(currRow, entry.getColumn(), entry.getValue());
192 }
193 ++currRow;
194 } else {
195 STORM_LOG_ERROR("More than one variable in row " << currRow << "!");
196 }
197 origStates.push_back(stateWithRow.state);
198 }
199 TransformationData result;
200 result.simpleMatrix = builder.build(currRow, currAuxState, currAuxState);
201 result.simpleStateToOriginalState = std::move(origStates);
202 return result;
203}

205storm::models::sparse::StateLabeling BinaryDtmcTransformer::transformStateLabeling(storm::models::sparse::Dtmc<RationalFunction> const& dtmc,
206 TransformationData const& data) const {
207 storm::models::sparse::StateLabeling labeling(data.simpleMatrix.getRowCount());
208 for (auto const& labelName : dtmc.getStateLabeling().getLabels()) {
209 storm::storage::BitVector newStates = dtmc.getStateLabeling().getStates(labelName);
210 newStates.resize(data.simpleMatrix.getRowCount(), false);
211 if (labelName != "init") {
212 for (uint64_t newState = dtmc.getNumberOfStates(); newState < data.simpleMatrix.getRowCount(); ++newState) {
213 newStates.set(newState, newStates[data.simpleStateToOriginalState[newState]]);
214 }
215 }
216 labeling.addLabel(labelName, std::move(newStates));
217 }
218 return labeling;
219}
