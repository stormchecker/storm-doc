65template<typename BeliefType>
66bool BeliefBuilder<BeliefType>::assertBelief() {
67 using ValueType = typename BeliefType::ValueType;
68 if (obs == InvalidObservation) {
69 STORM_LOG_ERROR("Observation of Belief not set.");
70 return false;
71 }
72 auto sum = storm::utility::zero<ValueType>();
73 for (auto const& [state, value] : data) {
74 if (value <= storm::utility::zero<ValueType>()) {
75 STORM_LOG_ERROR("Non-positive belief value " << value << " at state " << state << ".");
76 return false;
77 }
78 if (!BeliefNumerics<ValueType>::lessOrEqual(value, storm::utility::one<ValueType>())) {
79 STORM_LOG_ERROR("Invalid belief value " << value << " at state " << state << ".");
80 return false;
81 }
82 sum += value;
83 }
84 if (!BeliefNumerics<ValueType>::lessOrEqual(sum, storm::utility::one<ValueType>())) {
85 STORM_LOG_ERROR("belief value sum " << sum << " is larger than 1 (sum-1=" << sum - storm::utility::one<ValueType>() << ").");
86 return false;
87 }
88 if (!BeliefNumerics<ValueType>::lessOrEqual(storm::utility::one<ValueType>(), sum)) {
89 STORM_LOG_ERROR("belief value sum " << sum << " is smaller than 1 (1-sum=" << storm::utility::one<ValueType>() - sum << ").");
90 return false;
91 }
92 return true;
93}
