35void validateInput(std::shared_ptr<Model> const& model, std::shared_ptr<storm::logic::Formula const> const& formula,
36 std::optional<Engine> const& regionEngine = std::nullopt) {
37 STORM_LOG_THROW(model && formula, storm::exceptions::InvalidArgumentException, "A model and formula are required.");
38 STORM_LOG_THROW(model->isOfType(storm::models::ModelType::Dtmc) || model->isOfType(storm::models::ModelType::Mdp), storm::exceptions::NotSupportedException,
39 "Parameter synthesis supports sparse DTMCs and MDPs only.");
40 STORM_LOG_THROW(model->getInitialStates().getNumberOfSetBits() == 1, storm::exceptions::NotSupportedException,
41 "Parameter synthesis requires exactly one initial state.");
42 STORM_LOG_THROW((formula->isProbabilityOperatorFormula() || formula->isRewardOperatorFormula()) && formula->asOperatorFormula().hasBound(),
43 storm::exceptions::NotSupportedException, "Parameter synthesis requires a bounded probability or reward operator formula.");
44 if (regionEngine && *regionEngine != Engine::RobustParameterLifting) {
45 STORM_LOG_THROW(storm::utility::parameterlifting::validateParameterLiftingSound(*model, *formula, *regionEngine),
46 storm::exceptions::NotSupportedException, "Could not establish that parameter lifting is sound for this model and formula.");
47 }
48 if (regionEngine == Engine::RobustParameterLifting) {
49 STORM_LOG_THROW(model->isOfType(storm::models::ModelType::Dtmc), storm::exceptions::NotSupportedException, "Robust PLA supports DTMCs only.");
50 auto const& subformula = formula->asOperatorFormula().getSubformula();
51 STORM_LOG_THROW(!subformula.isBoundedUntilFormula() && !subformula.isCumulativeRewardFormula(), storm::exceptions::NotSupportedException,
52 "Robust PLA does not support step-bounded probability or cumulative reward formulas.");
53 if (formula->isRewardOperatorFormula()) {
54 auto const& rewardFormula = formula->asRewardOperatorFormula();
55 auto const& rewardModel =
56 rewardFormula.hasRewardModelName() ? model->getRewardModel(rewardFormula.getRewardModelName()) : model->getUniqueRewardModel();
57 for (auto const& reward : rewardModel.getTotalRewardVector(model->getTransitionMatrix())) {
58 STORM_LOG_THROW(storm::utility::isConstant(reward), storm::exceptions::NotSupportedException,
59 "Robust PLA does not support nonconstant total rewards.");
60 }
61 }
62 }

65std::pair<Region, GraphStatus> resolveRegion(Model const& model, std::optional<Region> const& region, std::optional<Coefficient> const& regionBound,
66 bool requireGraphPreservation) {
67 STORM_LOG_THROW(region.has_value() != regionBound.has_value(), storm::exceptions::InvalidArgumentException,
68 "Provide exactly one of region or region bound.");
69 auto const parameters = getModelParameters<storm::RationalFunction>(model);
70 Region resolved;
71 if (region) {
72 resolved = *region;
73 } else {
74 Coefficient const one = storm::utility::one<Coefficient>();
75 STORM_LOG_THROW(*regionBound >= storm::utility::zero<Coefficient>() && *regionBound <= one / 2, storm::exceptions::InvalidArgumentException,
76 "The region bound must be between 0 and 1/2, inclusive.");
77 Region::Valuation lower, upper;
78 for (auto const& parameter : parameters) {
79 lower.emplace(parameter, *regionBound);
80 upper.emplace(parameter, one - *regionBound);
81 }
82 resolved = Region(std::move(lower), std::move(upper));
83 }
84 for (auto const& variable : parameters) {
85 STORM_LOG_THROW(resolved.getVariables().count(variable), storm::exceptions::InvalidArgumentException,
86 "The region does not cover model parameter '" << variable.name() << "'.");
87 }
88 if (!requireGraphPreservation) {
89 return {std::move(resolved), GraphStatus::None};
90 }
91 storm::analysis::ConstraintCollector collector(model);
92 storm::pars::analysis::RegionConstraintChecker checker(collector);
93 STORM_LOG_THROW(checker.isGraphPreserving(resolved), storm::exceptions::NotSupportedException,
94 "This engine requires a well-defined, graph-preserving region. Provide an explicit graph-preserving region or use robust PLA for a DTMC. "
95 "The region's endpoints are not removed automatically.");
96 return {std::move(resolved), GraphStatus::GraphPreserving};
97}

99std::shared_ptr<storm::logic::Formula> normalizedFormula(Model const& model, storm::logic::Formula const& formula) {
100 auto result = formula.clone();
101 auto& op = result->asOperatorFormula();
102 // Removing the bound must not lose the default universal scheduler quantification of a threshold property.
103 if (model.isOfType(storm::models::ModelType::Mdp) && !op.hasOptimalityType()) {
104 op.setOptimalityType(op.getBound().isLowerBound() ? storm::solver::OptimizationDirection::Minimize : storm::solver::OptimizationDirection::Maximize);
105 }
106 return result;
107}

134Result<double> checkFeasibilityImpl(storm::Environment const& env, std::shared_ptr<Model> const& model,
135 std::shared_ptr<storm::logic::Formula const> const& formula, std::optional<Region> const& region,
136 std::optional<Coefficient> const& regionBound, storm::pars::FeasibilityMethod method, Engine regionEngine) {
137 try {
138 switch (method) {
139 case storm::pars::FeasibilityMethod::PLA: {
140 validateInput(model, formula, regionEngine);
141 auto const [consideredRegion, graphStatus] = resolveRegion(*model, region, regionBound, regionEngine != Engine::RobustParameterLifting);
142 auto normalized = normalizedFormula(*model, *formula);
143 std::set<Region::VariableType> discreteVariables;
144 RefinementOptions<storm::RationalFunction> settings(model, *normalized, regionEngine, storm::modelchecker::RegionSplittingStrategy(),
145 MonotonicitySetting(), discreteVariables, true, graphStatus);
146 auto checker = initializeRegionRefinementChecker(env, settings);
147 auto const& bound = normalized->asOperatorFormula().getBound();
148 auto acceptGlobalBound = [&](auto const&, auto const& globalBound) { return !bound.isSatisfied(globalBound); };
149 auto rejectInstance = [&](auto const& value) { return bound.isSatisfied(value); };
150 auto direction = bound.isLowerBound() ? storm::solver::OptimizationDirection::Maximize : storm::solver::OptimizationDirection::Minimize;
151 auto result = checker->computeExtremalValueHelper(env, consideredRegion, direction, acceptGlobalBound, rejectInstance);
152 if (bound.isSatisfied(result.first)) {
153 return {Status::ExistsSat,
154 storm::pars::FeasibilityResult<double>{storm::utility::convertNumber<double>(result.first), std::move(result.second)}};
155 }
156 return {Status::AllViolated, std::nullopt};
157 }
158 case storm::pars::FeasibilityMethod::GD: {
159 validateInput(model, formula);
160 STORM_LOG_THROW(model->isOfType(storm::models::ModelType::Dtmc), storm::exceptions::NotSupportedException,
161 "Gradient descent supports DTMCs only.");
162 auto const [consideredRegion, graphStatus] = resolveRegion(*model, region, regionBound, true);
163 if (std::all_of(consideredRegion.getVariables().begin(), consideredRegion.getVariables().end(),
164 [&](auto const& variable) { return storm::utility::isZero(consideredRegion.getDifference(variable)); })) {
165 auto valuation = consideredRegion.getCenterPoint();
166 storm::modelchecker::SparseDtmcInstantiationModelChecker<Dtmc, double> checker(env, *model->as<Dtmc>());
167 checker.specifyFormula(createTask<storm::RationalFunction>(quantitativeFormula(*model, *formula), true));
168 auto value = checker.check(env, valuation)->asExplicitQuantitativeCheckResult<double>()[model->getInitialStates().getNextSetIndex(0)];
169 if (!formula->asOperatorFormula().getBound().isSatisfied(value)) {
170 return {Status::AllViolated, std::nullopt};
171 }
172 return {Status::ExistsSat, storm::pars::FeasibilityResult<double>{value, std::move(valuation)}};
173 }
174 auto task = createFeasibilityTask(*model, *formula, consideredRegion);
175 storm::derivative::GradientDescentInstantiationSearcher<storm::RationalFunction, double> searcher(
176 env, *model->as<Dtmc>(), storm::derivative::GradientDescentMethod::ADAM, 0.1, 0.9, 0.999, 32, 1e-6, consideredRegion.getCenterPoint(),
177 storm::derivative::GradientDescentConstraintMethod::PROJECT_WITH_GRADIENT, consideredRegion);
178 searcher.setup(env, task);
179 auto result = searcher.gradientDescent();
180 if (!formula->asOperatorFormula().getBound().isSatisfied(result.value)) {
181 return {};
182 }
183 return {Status::ExistsSat, std::move(result)};
184 }
185 case storm::pars::FeasibilityMethod::SCP: {
186 validateInput(model, formula);
187 auto const& op = formula->asOperatorFormula();
188 if (model->isOfType(storm::models::ModelType::Mdp) && op.hasOptimalityType()) {
189 auto schedulerDirection =
190 op.getBound().isLowerBound() ? storm::solver::OptimizationDirection::Minimize : storm::solver::OptimizationDirection::Maximize;
191 STORM_LOG_THROW(op.getOptimalityType() == schedulerDirection, storm::exceptions::NotSupportedException,
192 "SCP requires universal scheduler quantification on MDPs: lower bounds use min and upper bounds use max.");
193 }
194 auto const [consideredRegion, graphStatus] = resolveRegion(*model, region, regionBound, true);
195 auto task = createFeasibilityTask(*model, *formula, consideredRegion);
196 storm::pars::scp::ScpResult<double> result;
197 if (model->isOfType(storm::models::ModelType::Dtmc)) {
198 result = storm::pars::scp::SparseScpInstantiationSearcher<Dtmc, double>(*model->as<Dtmc>()).run(env, task);
199 } else {
200 STORM_LOG_ASSERT(model->isOfType(storm::models::ModelType::Mdp), "Expected an MDP.");
201 result = storm::pars::scp::SparseScpInstantiationSearcher<Mdp, double>(*model->as<Mdp>()).run(env, task);
202 }
203 if (result.status != storm::pars::scp::ScpStatus::Feasible) {
204 return {};
205 }
206 return {Status::ExistsSat, storm::pars::FeasibilityResult<double>{std::move(result.value), std::move(result.valuation)}};
207 }
208 }
209 STORM_LOG_THROW_UNCONDITIONALLY(storm::exceptions::InvalidArgumentException, "Unknown feasibility method.");
210 } catch (storm::exceptions::AbortException const&) {
211 return {};
212 }
