422TEST(MinMaxLinearEquationSolverTest, SoundMethodsTerminateEarly) {
423 // State 0 has value 1 but converges slowly due to its self-loop, state 1 has value 0.5.
424 // The thresholds are already met after the first iteration, so every sound method should stop far from convergence.
425 storm::storage::SparseMatrixBuilder<double> builder(0, 0, 0, false, true);
426 builder.newRowGroup(0);
427 builder.addNextValue(0, 0, 0.99);
428 builder.newRowGroup(1);
429 builder.addNextValue(1, 1, 0.5);
430 storm::storage::SparseMatrix<double> A = builder.build(2, 2, 2);
431 std::vector<double> b = {0.01, 0.25};
432 storm::storage::BitVector filter(2, std::vector<uint64_t>{0});
433
434 for (auto method : {storm::solver::MinMaxMethod::SoundValueIteration, storm::solver::MinMaxMethod::IntervalIteration,
435 storm::solver::MinMaxMethod::GuessingValueIteration}) {
436 for (bool useLowerBound : {true, false}) {
437 SCOPED_TRACE(storm::solver::toString(method) + (useLowerBound ? " with lower bound" : " with upper bound"));
438 storm::Environment env;
439 env.solver().minMax().setMethod(method);
440 env.solver().setForceSoundness(true);
441 env.solver().minMax().setPrecision(storm::utility::convertNumber<storm::RationalNumber>(1e-6));
442
443 auto solver = storm::solver::GeneralMinMaxLinearEquationSolverFactory<double>().create(env, A);
444 solver->setHasUniqueSolution(true);
445 solver->setHasNoEndComponents(true);
446 solver->setBounds(0.0, 10.0);
447 if (useLowerBound) {
448 solver->setTerminationCondition(

450 } else {
451 solver->setTerminationCondition(std::make_unique<storm::solver::TerminateIfFilteredExtremumBelowThreshold<double>>(filter, false, 9.95, true));
452 }
453 std::vector<double> x(2);
454 ASSERT_NO_THROW(solver->solveEquations(env, storm::OptimizationDirection::Minimize, x, b));
455 // Without early termination, x[0] would be within the precision of 1.
456 EXPECT_GT(std::abs(x[0] - 1.0), 1e-3) << "x[0] = " << x[0];
457 }
458 }

461TEST(IntervalIterationJustPreciseEnough, MinMax) {
462 storm::storage::SparseMatrixBuilder<double> builder(0, 0, 0, false, true);
463 builder.newRowGroup(0);
464 builder.addNextValue(0, 0, 0.9);
465 builder.addNextValue(1, 0, 0.9);
466 auto matrix = builder.build();
467 std::vector<double> b = {0.099, 0.05}; // Fixed points 0.99 and 0.5.
468 double const precision = 1e-3;
469
470 for (bool relative : {false, true}) {
471 for (auto dir : {storm::OptimizationDirection::Minimize, storm::OptimizationDirection::Maximize}) {
472 SCOPED_TRACE(::testing::Message() << "relative=" << relative << ", maximize=" << storm::solver::maximize(dir));
473 auto env = DoubleIntervalIterationEnvironment::createEnvironment();
474 env.solver().minMax().setPrecision(storm::utility::convertNumber<storm::RationalNumber>(precision));
475 env.solver().minMax().setRelativeTerminationCriterion(relative);
476 auto solver = storm::solver::GeneralMinMaxLinearEquationSolverFactory<double>().create(env, matrix);
477 solver->setHasUniqueSolution(true);
478 solver->setHasNoEndComponents(true);
479 solver->setBounds(0.0, 1.0);
480 std::vector<double> x(1);
481 ASSERT_TRUE(solver->solveEquations(env, dir, x, b));
482 ASSERT_TRUE(solver->hasSolutionLowerBounds());
483 ASSERT_TRUE(solver->hasSolutionUpperBounds());
484 double const lower = solver->getSolutionLowerBounds()[0];
485 double const upper = solver->getSolutionUpperBounds()[0];
486 double const exact = storm::solver::maximize(dir) ? 0.99 : 0.5;
487 double const tolerance = precision * (relative ? lower : 1.0);
488 EXPECT_LE(lower, exact);
489 EXPECT_GE(upper, exact);
490 EXPECT_DOUBLE_EQ(x[0], (lower + upper) / 2.0);
491 EXPECT_NEAR(x[0], exact, precision * (relative ? exact : 1.0));
492 EXPECT_LE(upper - lower, 2.0 * tolerance);
493 // With this slow contraction, stopping at half this width takes unnecessary iterations.
494 EXPECT_GT(upper - lower, tolerance);
495 }
