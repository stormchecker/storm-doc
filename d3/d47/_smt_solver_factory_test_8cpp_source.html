20TEST(SmtSolverFactory, EnvironmentIsSeededFromSettings) {
21 storm::Environment env;

23 // The environment picks up the SMT solver that is selected in the core settings.

41TEST(SmtSolverFactory, EnvironmentOverridesSmtSolver) {

56 storm::Environment env;

59 // The environment is the single place where the core setting is turned into a solver selection, so
