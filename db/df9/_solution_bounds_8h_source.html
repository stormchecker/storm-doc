50 bool encloses(std::vector<ValueType> const& values) const {
51 auto const isBoundedBy = [&values](std::optional<std::vector<ValueType>> const& bound, bool fromBelow) {
52 if (!bound.has_value()) {
53 return true;
54 }
55 if (bound->size() != values.size()) {
56 return false;
57 }
58 for (uint64_t i = 0; i < values.size(); ++i) {
59 if (fromBelow ? (*bound)[i] > values[i] : (*bound)[i] < values[i]) {
60 return false;
61 }
62 }
63 return true;
64 };
65 return isBoundedBy(lower, true) && isBoundedBy(upper, false);
66 }

78 bool isWithinPrecision(std::vector<ValueType> const& values, ValueType const& precision, bool relative) const {
79 if (!hasLower() || !hasUpper()) {
80 return false;
81 }
82 STORM_LOG_THROW(lower->size() == values.size() && upper->size() == values.size(), storm::exceptions::InvalidArgumentException,
83 "The given values do not match the size of the bounds.");
84 ValueType const zero = storm::utility::zero<ValueType>();
85 for (uint64_t i = 0; i < values.size(); ++i) {
86 ValueType const& low = (*lower)[i];
87 ValueType const& up = (*upper)[i];
88 ValueType allowed = precision;
89 if (relative) {
90 // The enclosed value is at least as far from zero as the bound closer to it, unless the two
91 // straddle zero, where only coinciding bounds say anything.
92 allowed = (low <= zero && up >= zero) ? zero : precision * std::min(storm::utility::abs(low), storm::utility::abs(up));
93 }
94 // Negated, so that a distance that is not a number is rejected.
95 if (!(values[i] - low <= allowed) || !(up - values[i] <= allowed)) {
96 return false;
97 }
98 }
99 return true;
100 }
