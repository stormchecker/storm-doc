75TEST_F(GraphConditionsTest, PolynomialDenominators) {
76 auto dtmc = buildParametricDtmc(STORM_TEST_RESOURCES_DIR "/pdtmc/only_p.pm", "P=? [F s=1]");
77
78 storm::analysis::ConstraintCollector collector(*dtmc);
79
80 // The only parameter must be reported.
81 std::vector<storm::RationalFunctionVariable> variables(collector.getVariables().begin(), collector.getVariables().end());
82 ASSERT_EQ(1ul, variables.size());
83 EXPECT_EQ("p", variables.front().name());
84
85 // The transitions are the symbolic values p and 1 - p, and both denominators are positive constants. We
86 // therefore expect the constraints 0 <= p <= 1 and 0 <= 1 - p <= 1. Note that the row sums to 1 identically,
87 // so no row-sum constraint is collected.
88 //
89 // Relations are normalized so that the polynomial never occurs on the greater side, and the upper bound is
90 // phrased as a relation of the difference to zero. Consequently, the constraints derived from p and from
91 // 1 - p coincide pairwise and are collected only once each.
92 std::vector<std::string> expectedWellformed{"((-1 + p) <= 0)", "(-(p) <= 0)"};
93 EXPECT_EQ(expectedWellformed, asSortedStrings(collector.getWellformedConstraints()));
94
95 // Both transitions must stay non-zero, which for p and 1 - p amounts to p != 0 and 1 - p != 0.
96 std::vector<std::string> expectedGraphPreserving{"((1 - p) != 0)", "(p != 0)"};
97 EXPECT_EQ(expectedGraphPreserving, asSortedStrings(collector.getGraphPreservingConstraints()));
98}

100TEST_F(GraphConditionsTest, SymbolicDenominators) {
101 auto dtmc = buildParametricDtmc(STORM_TEST_RESOURCES_DIR "/pdtmc/only_rational_denominator.pm", "P=? [F s=2]");
102
103 storm::analysis::ConstraintCollector collector(*dtmc);
104
105 std::vector<storm::RationalFunctionVariable> variables(collector.getVariables().begin(), collector.getVariables().end());
106 ASSERT_EQ(2ul, variables.size());
107
108 // Here both denominators are the symbolic value 1 - p + q, so non-negativity cannot be reduced to a sign
109 // constraint on the nominator. Instead, the denominator must be constrained to be non-zero, and the sign of
110 // each nominator must follow the sign of the denominator.
111 std::vector<std::string> expectedWellformed{"((((-1 + p) - q) < 0) ? ((-1 + p) <= 0) : ((1 - p) <= 0))", "((((-1 + p) - q) < 0) ? (-(q) <= 0) : (q <= 0))",
112 "(((1 - p) + q) != 0)"};
113 EXPECT_EQ(expectedWellformed, asSortedStrings(collector.getWellformedConstraints()));
114
115 std::vector<std::string> expectedGraphPreserving{"((1 - p) != 0)", "(q != 0)"};
116 EXPECT_EQ(expectedGraphPreserving, asSortedStrings(collector.getGraphPreservingConstraints()));
117}

136TEST_F(GraphConditionsTest, ConstraintsAreOrderedDeterministically) {
137 auto dtmc = buildParametricDtmc(STORM_TEST_RESOURCES_DIR "/pdtmc/only_p.pm", "P=? [F s=1]");
138
139 // Collecting the constraints repeatedly from equal models must yield the same order.
140 storm::analysis::ConstraintCollector firstCollector(*dtmc);
141 storm::analysis::ConstraintCollector secondCollector(*dtmc);
142
143 EXPECT_EQ(asStringsInDiscoveryOrder(firstCollector.getWellformedConstraints()), asStringsInDiscoveryOrder(secondCollector.getWellformedConstraints()));
144 EXPECT_EQ(asStringsInDiscoveryOrder(firstCollector.getGraphPreservingConstraints()),
145 asStringsInDiscoveryOrder(secondCollector.getGraphPreservingConstraints()));
146
147 // The constraints are collected in the order in which the model traversal discovers them, which is not their
148 // lexicographic order. Pinning the expected order keeps this test from passing vacuously.
149 std::vector<std::string> expectedWellformed{"(-(p) <= 0)", "((-1 + p) <= 0)"};
150 std::vector<std::string> expectedGraphPreserving{"(p != 0)", "((1 - p) != 0)"};
151
152 EXPECT_EQ(expectedWellformed, asStringsInDiscoveryOrder(firstCollector.getWellformedConstraints()));
153 EXPECT_EQ(expectedGraphPreserving, asStringsInDiscoveryOrder(firstCollector.getGraphPreservingConstraints()));
154
155 // Guard the guard: if the discovery order ever happened to coincide with the lexicographic one, the assertions
156 // above could no longer detect a collector that reorders or sorts its constraints.
157 auto sortedWellformed = asSortedStrings(firstCollector.getWellformedConstraints());
158 EXPECT_NE(expectedWellformed, sortedWellformed);
159}

161TEST_F(GraphConditionsTest, ExportToFile) {
162 auto dtmc = buildParametricDtmc(STORM_TEST_RESOURCES_DIR "/pdtmc/only_rational_denominator.pm", "P=? [F s=2]");
163
164 storm::analysis::ConstraintCollector collector(*dtmc);
165
166 // Use one of the model's symbolic transition values as the solution function.
167 storm::RationalFunction resultValue = storm::utility::zero<storm::RationalFunction>();
168 for (auto const& entry : dtmc->getTransitionMatrix()) {
169 if (!storm::utility::isConstant(entry.getValue())) {
170 resultValue = entry.getValue();
171 break;
172 }
173 }
174
175 std::string temporaryFile = "graph-conditions-export.txt";
176 storm::pars::api::exportParametricResultToFile<storm::RationalFunction>(resultValue, collector, temporaryFile);
177
178 std::ifstream input(temporaryFile);
179 ASSERT_TRUE(input.good());
180 std::stringstream buffer;
181 buffer << input.rdbuf();
182 std::string content = buffer.str();
183 storm::io::closeFile(input);
184
185 // The export must mention the parameters, the result, and both kinds of constraints.
186 EXPECT_NE(std::string::npos, content.find("$Parameters: p; q; \n"));
187 EXPECT_NE(std::string::npos, content.find("$Result: "));
188 EXPECT_NE(std::string::npos, content.find("$Well-formed Constraints: \n"));
189 EXPECT_NE(std::string::npos, content.find("$Graph-preserving Constraints: \n"));
190
191 // The export must list exactly the collected constraints. The collector returns them in the order in which it
192 // discovered them, so both sides are sorted before comparing.
193 std::vector<std::string> wellformedLines;
194 std::string currentSection;
195 std::istringstream contentStream(content);
196 std::string line;
197 while (std::getline(contentStream, line)) {
198 if (line == "$Well-formed Constraints: ") {
199 currentSection = "wellformed";
200 } else if (line == "$Graph-preserving Constraints: ") {
201 currentSection = "graphPreserving";
202 } else if (!line.empty() && currentSection == "wellformed") {
203 wellformedLines.push_back(line);
204 }
205 }
206 std::sort(wellformedLines.begin(), wellformedLines.end());
207 EXPECT_EQ(asSortedStrings(collector.getWellformedConstraints()), wellformedLines);
208
209 std::remove(temporaryFile.c_str());
210}
