13QuotientData<ValueType>::QuotientData(storm::bisimulation::Partition const& partition,
14 storm::OptionalRef<storm::storage::BitVector const> preferredRepresentatives) {
15 uint64_t constexpr Undef = std::numeric_limits<uint64_t>::max();
16 toQuotientState.assign(partition.getNumberOfElements(), Undef);
17 toRepresentativeState.reserve(partition.getNumberOfBlocks());
18 // Number the quotient states in the order of the smallest state of their block, so that the quotient resembles the order of the original states.
19 for (uint64_t state = 0; state < partition.getNumberOfElements(); ++state) {
20 if (toQuotientState[state] != Undef) {
21 continue;
22 }
23 auto const block = partition.getBlockOfElement(state);
24 uint64_t const quotientState = toRepresentativeState.size();
25 for (auto const s : block) {
26 toQuotientState[s] = quotientState;
27 }
28 // Unless a preferred representative is available, the first state of the block is the representative. Approximative signature-based refinement
29 // relies on this: that state is the anchor that all other states of the block were compared with, cf. Signatures::extendQuotientData.
30 uint64_t representativeState = block.front();
31 if (preferredRepresentatives) {
32 if (auto const it =
33 std::find_if(block.begin(), block.end(), [&preferredRepresentatives](auto const s) { return preferredRepresentatives->get(s); });
34 it != block.end()) {
35 representativeState = *it;
36 }
37 }
38 toRepresentativeState.push_back(representativeState);
39 }
40}
